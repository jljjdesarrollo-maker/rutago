   497→   646→   760→   760→// Lógica de precios (tipo semántico asignado por el API):
   498→   647→   761→   761→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   499→   648→   762→   762→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   500→   649→   763→   763→//
   501→   650→   764→   764→// El API asigna dirección según el origen:
   502→   651→   765→   765→//   Si origen = Loja →  "ida" (alejándose)
   503→   652→   766→   766→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   504→   653→   767→   767→//
   505→   654→   768→   768→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   506→   655→   769→   769→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.

   508→   657→   771→   771→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   509→   658→   772→   772→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   510→   659→   773→   773→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   511→   660→   774→   774→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   512→   661→   775→   775→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   513→   662→   776→   776→  return false;
   514→   663→   777→   777→}

   517→   666→   780→   780→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   518→   667→   781→   781→  if (viajaHaciaLoja(ruta, tipo)) {

   534→   683→   797→   797→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   535→   684→   798→   798→    if (ruta === 'Loja - Vilcabamba') {

   561→   710→   824→   824→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"

   586→   735→   849→   849→  }

   589→   738→   852→   852→  return paradas.map(parada => {

   624→   773→   887→   887→export function matchRuta(rutaNombre: string): string {
   625→   774→   888→   888→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   626→   775→   889→   889→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   627→   776→   890→   890→  if (RUTA_PARADAS[normalized]) return normalized;
   628→   777→   891→   891→
   629→   778→   892→   892→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   630→   779→   893→   893→  for (const key of Object.keys(RUTA_PARADAS)) {
   631→   780→   894→   894→    const keyParts = key.split(' - ');
   632→   781→   895→   895→    if (parts.length >= 2 && keyParts.length >= 2) {
   633→   782→   896→   896→      const match = (
   634→   783→   897→   897→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   635→   784→   898→   898→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   636→   785→   899→   899→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   637→   786→   900→   900→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   638→   787→   901→   901→      );
   639→   788→   902→   902→      if (match) return key;
   640→   789→   903→   903→    }
   641→   790→   904→   904→  }
   642→   791→   905→   905→  return rutaNombre;
   643→   792→   906→   906→}
