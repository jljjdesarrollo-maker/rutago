    51→    51→-- 3. TARIFAS (precios por ruta, dirección y parada)

    53→    53→-- Lógica de lookup:
    54→    54→--   1. Buscar Tarifa WHERE rutaId = [ruta específica] AND parada = X AND direccion = 'vuelta'
    55→    55→--   2. Si no existe, buscar WHERE rutaId IS NULL (troncal compartido) AND parada = X AND direccion = 'vuelta'
    56→    56→--   Esto replica: preciosVueltaLaElvira -> preciosVuelta (fallback)
    57→    57→
    58→    58→CREATE TABLE "Tarifa" (
    59→    59→    "id"          TEXT NOT NULL PRIMARY KEY,
    60→    60→    "rutaId"      TEXT,
    61→    61→    "parada"      TEXT NOT NULL,
    62→    62→    "direccion"   TEXT NOT NULL,
    63→    63→    "normal"      DOUBLE PRECISION NOT NULL DEFAULT 0,
    64→    64→    "media"       DOUBLE PRECISION NOT NULL DEFAULT 0,
    65→    65→    "esTemporal"  BOOLEAN NOT NULL DEFAULT false,
    66→    66→    "prioridad"   INTEGER NOT NULL DEFAULT 1,
    67→    67→    "createdAt"   TIMESTAMP(3) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    68→    68→    "updatedAt"   TIMESTAMP(3) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    69→    69→
    70→    70→    CONSTRAINT fk_tarifa_ruta
    71→    71→        FOREIGN KEY ("rutaId") REFERENCES "Ruta"("id")
    72→    72→        ON DELETE CASCADE ON UPDATE CASCADE,
    73→    73→    CONSTRAINT chk_tarifa_direccion CHECK ("direccion" IN ('ida', 'vuelta')),
    74→    74→    CONSTRAINT uq_tarifa_ruta_parada_dir UNIQUE ("rutaId", "parada", "direccion")
    75→    75→);
    76→    76→
    77→    77→COMMENT ON TABLE "Tarifa" IS 'Precios por parada. rutaId=NULL = precio troncal compartido. rutaId=X = precio específico de ruta (prioridad)';
