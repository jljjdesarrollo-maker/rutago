    36→    36→interface GastoItem {

    40→    40→
    41→    41→const today = () => new Date().toISOString().split('T')[0];
    42→    42→
    43→    43→// Work date: use session.fecha (date when shift started, not necessarily today)
    44→    44→function workDate(session: VTSession): string {
    45→    45→  return session.fecha || today();
    46→    46→}
    47→    47→
    48→    48→const GASTOS_DEFAULT: GastoItem[] = [
    49→    49→  { description: 'Chofer', amount: '30' },
    50→    50→  { description: 'Ayudante', amount: '20' },

    52→    52→  { description: 'Plan Renova', amount: '68' },
    53→    53→];
    54→    54→
    55→    55→export function ArqueoGeneralScreen({ session, connection, onClose, onGoToSync, onSaved }: Props) {
    56→    56→  const [frecuencias, setFrecuencias] = useState<FrecuenciaResumen[]>([]);
    57→    57→  const [loading, setLoading] = useState(true);
    58→    58→  const [km, setKm] = useState('');
    59→    59→  const [gastos, setGastos] = useState<GastoItem[]>(GASTOS_DEFAULT);
    60→    60→  const [tickets, setTickets] = useState('');
    61→    61→  const [sobrante, setSobrante] = useState('');
    62→    62→  const [fotoPreview, setFotoPreview] = useState<string | null>(null);
    63→    63→  const [saving, setSaving] = useState(false);
    64→    64→  const [confirmSave, setConfirmSave] = useState(false);
    65→    65→  const [saved, setSaved] = useState(false);
    66→    66→  const [errors, setErrors] = useState<string[]>([]);
    67→    67→  const [guardadoOffline, setGuardadoOffline] = useState(false);
    68→    68→
    69→    69→  const fileInputRef = useRef<HTMLInputElement>(null);
    70→    70→  const loadData = useCallback(async () => {
    71→    71→    try {
    72→    72→      const fecha = workDate(session);
    73→    73→      // Load estados from localStorage (synchronous, reliable)
    74→    74→      const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
    75→    75→      const raw = localStorage.getItem(lsKey);
    76→    76→      if (!raw) { setLoading(false); return; }
    77→    77→      const estadosLS = JSON.parse(raw) as FrecuenciaEstado[];
    78→    78→      const cerradas = estadosLS.filter(e => e.estado === 'cerrada');
    79→    79→      // Ingresos especiales (no_realizada con monto > 0)
    80→    80→      const especiales = estadosLS.filter(e => e.estado === 'no_realizada' && (e.ingresoEspecialMonto || 0) > 0);
    81→    81→
    82→    82→      const resumenes: FrecuenciaResumen[] = await Promise.all(
    83→    83→        cerradas.map(async (e) => {
    84→    84→          const ventas = await getVentasByFrecuencia(e.estadoId);
    85→    85→          const sistemaTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
    86→    86→          // Use arqueo data if available (what helper actually counted), else system total
    87→    87→          const efectivo = (e as any).arqueoEfectivo ?? sistemaTotal;
    88→    88→          const diff = efectivo - sistemaTotal;
    89→    89→          const ccCount = (e as any).cajaComunCount || 0;
    90→    90→          const ccMonto = (e as any).cajaComunMonto || 0;
    91→    91→          return {
    92→    92→            estadoId: e.estadoId,
    93→    93→            nombre: e.nombre,
    94→    94→            ruta: e.ruta,
    95→    95→            hora: e.hora,
    96→    96→            direccion: e.direccion,
    97→    97→            ventasCount: ventas.length,
    98→    98→            totalRecaudado: sistemaTotal,
    99→    99→            efectivoContado: efectivo,
   100→   100→            diferencia: diff,
   101→   101→            boletosCaja: ccCount,
   102→   102→            cajaComunMonto: ccMonto,
   103→   103→          };
   104→   104→        })
   105→   105→      );
   106→   106→
   107→   107→      // Add ingresos especiales as virtual rows
   108→   108→      especiales.forEach(e => {
   109→   109→        resumenes.push({
   110→   110→          estadoId: e.estadoId,
   111→   111→          nombre: e.nombre,
   112→   112→          ruta: e.ruta,
   113→   113→          hora: e.hora,
   114→   114→          direccion: e.direccion,
   115→   115→          ventasCount: 0,
   116→   116→          totalRecaudado: e.ingresoEspecialMonto || 0,
   117→   117→          efectivoContado: e.ingresoEspecialMonto || 0,
   118→   118→          diferencia: 0,
   119→   119→          boletosCaja: 0,
   120→   120→          cajaComunMonto: 0,
   121→   121→          isIngresoEspecial: true,
   122→   122→          ingresoEspecialNota: e.ingresoEspecialNota,
   123→   123→        });
   124→   124→      });
   125→   125→      setFrecuencias(resumenes);
   126→   126→    } catch (err) {
   127→   127→      console.error('Error cargando resumen:', err);
   128→   128→    } finally {
   129→   129→      setLoading(false);
   130→   130→    }
   131→   131→  }, [session.vtCode]);
   132→   132→
   133→   133→  useEffect(() => { loadData(); }, [loadData]);
   134→   134→
   135→   135→  const totalIngresosAuto = useMemo(() =>
   136→   136→    frecuencias.reduce((s, f) => s + f.totalRecaudado, 0),
   137→   137→    [frecuencias]
   138→   138→  );
   139→   139→  const totalEfectivoReal = useMemo(() =>
   140→   140→    frecuencias.reduce((s, f) => s + f.efectivoContado, 0),
   141→   141→    [frecuencias]
   142→   142→  );
   143→   143→  const totalDiferencia = useMemo(() =>
   144→   144→    frecuencias.reduce((s, f) => s + f.diferencia, 0),
   145→   145→    [frecuencias]
   146→   146→  );
   147→   147→  const totalVentas = useMemo(() =>
   148→   148→    frecuencias.reduce((s, f) => s + f.ventasCount, 0),
   149→   149→    [frecuencias]
   150→   150→  );
   151→   151→  const totalCajaComunPasajeros = useMemo(() =>
   152→   152→    frecuencias.reduce((s, f) => s + f.boletosCaja, 0),
   153→   153→    [frecuencias]
   154→   154→  );
   155→   155→  const totalCajaComunMonto = useMemo(() =>
   156→   156→    frecuencias.reduce((s, f) => s + (f.cajaComunMonto || 0), 0),
   157→   157→    [frecuencias]
   158→   158→  );
   159→   159→  const totalGastos = useMemo(() =>
   160→   160→    gastos.reduce((s, g) => s + (parseFloat(g.amount) || 0), 0),
   161→   161→    [gastos]
   162→   162→  );
   163→   163→  const ticketsNum = parseFloat(tickets) || 0;
   164→   164→  const sobranteNum = parseFloat(sobrante) || 0;
   165→   165→  // PRODUCCION = efectivo real contado por el ayudante + caja comun + sobrante ajuste manual
   166→   166→  const production = totalEfectivoReal + totalCajaComunMonto + sobranteNum;
   167→   167→  // ENTREGA AYUDANTE = efectivo real + sobrante - gastos (la caja común NO pasa por el ayudante)
   168→   168→  const entregaAyudante = totalEfectivoReal + sobranteNum - totalGastos;
   169→   169→  // Lógica dual: si hay caja común se descuentan tickets, si no (pre-Jun 2026) entrega compañía = 0
   170→   170→  const entregaCompania = totalCajaComunMonto > 0 ? totalCajaComunMonto - ticketsNum : 0;
   171→   171→
   172→   172→  const addGasto = () => setGastos(prev => [...prev, { description: '', amount: '0' }]);
   173→   173→  const removeGasto = (i: number) => setGastos(prev => prev.filter((_, idx) => idx !== i));
   174→   174→  const updateGasto = (i: number, field: keyof GastoItem, value: string) => {
   175→   175→    setGastos(prev => {
   176→   176→      const next = [...prev];
   177→   177→      next[i] = { ...next[i], [field]: value };
   178→   178→      return next;
   179→   179→    });
   180→   180→  };
   181→   181→
   182→   182→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   183→   183→    const file = e.target.files?.[0];
   184→   184→    if (!file || !file.type.startsWith('image/')) return;
   185→   185→    e.target.value = '';
   186→   186→    // Comprimir imagen antes de almacenar (max ~200KB base64)
   187→   187→    const img = new Image();
   188→   188→    const url = URL.createObjectURL(file);
   189→   189→    img.onload = () => {
   190→   190→      URL.revokeObjectURL(url);
   191→   191→      const MAX_DIM = 1200;
   192→   192→      const QUALITY = 0.6;
   193→   193→      let w = img.width, h = img.height;
   194→   194→      if (w > MAX_DIM || h > MAX_DIM) {
   195→   195→        if (w > h) { h = Math.round(h * MAX_DIM / w); w = MAX_DIM; }
   196→   196→        else { w = Math.round(w * MAX_DIM / h); h = MAX_DIM; }
   197→   197→      }
   198→   198→      const canvas = document.createElement('canvas');
   199→   199→      canvas.width = w; canvas.height = h;
   200→   200→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h);
   201→   201→      setFotoPreview(canvas.toDataURL('image/jpeg', QUALITY));
   202→   202→    };
   203→   203→    img.onerror = () => URL.revokeObjectURL(url);
   204→   204→    img.src = url;
   205→   205→  };
   206→   206→
   207→   207→  const handleSaveOffline = () => {
   208→   208→    const errs: string[] = [];
   209→   209→    if (!km.trim()) errs.push('Kilometraje es obligatorio');
   210→   210→    if (!fotoPreview) errs.push('Foto del cuaderno es obligatoria');
   211→   211→    if (frecuencias.length === 0) errs.push('No hay frecuencias cerradas');
   212→   212→    // Gastos pueden ser $0 si no hubo (ej. unidad parada)
   213→   213→    if (errs.length > 0) {
   214→   214→      setErrors(errs);
   215→   215→      return;
   216→   216→    }
   217→   217→    setErrors([]);
   218→   218→    setConfirmSave(true);
   219→   219→  };
   220→   220→
   221→   221→  const handleConfirmSave = async () => {
   222→   222→    setConfirmSave(false);
   223→   223→    setSaving(true);
   224→   224→
   225→   225→    try {
   226→   226→      // Build trips from frecuencias
   227→   227→      const trips = frecuencias.map(f => ({
   228→   228→        routeFrom: f.direccion === 'ida' ? 'Loja' : f.ruta.split(' - ')[1]?.trim() || 'Loja',
   229→   229→        routeTo: f.direccion === 'ida' ? f.ruta.split(' - ')[1]?.trim() || 'Vilcabamba' : 'Loja',
   230→   230→        time: f.hora,
   231→   231→        income: f.totalRecaudado.toString(),
   232→   232→        efectivoReal: f.efectivoContado.toString(),
   233→   233→        boletos: '0',
   234→   234→        cajaComunPasajeros: f.boletosCaja.toString(),
   235→   235→        cajaComunMonto: (f.cajaComunMonto || 0).toString(),
   236→   236→      }));
   237→   237→
   238→   238→      const body = {
   239→   239→        date: workDate(session),
   240→   240→        km: km,
   241→   241→        conductor: '',
   242→   242→        ayudanteNombre: session.ayudanteNombre,
   243→   243→        vtCode: session.vtCode,
   244→   244→        trips,
   245→   245→        expenses: gastos,
   246→   246→        tickets: tickets || '0',
   247→   247→        cajaComun: totalCajaComunMonto,
   248→   248→        sobrante: sobrante || '0',
   249→   249→        photoUrl: fotoPreview,
   250→   250→      };
   251→   251→
   252→   252→      const fechaTrabajo = workDate(session);
   253→   253→      const isOnline = navigator.onLine;
   254→   254→
   255→   255→      if (isOnline) {
   256→   256→        const res = await fetch('/api/records', {
   257→   257→          method: 'POST',
   258→   258→          headers: { 'Content-Type': 'application/json' },
   259→   259→          body: JSON.stringify(body),
   260→   260→        });
   261→   261→        if (res.ok) {
   262→   262→          setSaved(true);
   263→   263→        } else {
   264→   264→          const result = await res.json().catch(() => ({ error: 'Error desconocido' }));
   265→   265→          console.error('Error guardando arqueo en servidor:', result.error);
   266→   266→          // Save offline if server fails
   267→   267→          localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   268→   268→          setGuardadoOffline(true);
   269→   269→          setSaved(true);
   270→   270→        }
   271→   271→      } else {
   272→   272→        // Save to localStorage for later sync
   273→   273→        localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   274→   274→        setGuardadoOffline(true);
   275→   275→        setSaved(true);
   276→   276→      }
   277→   277→    } catch (err) {
   278→   278→      console.error('Error guardando arqueo:', err);
   279→   279→      localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify({
   280→   280→        date: fechaTrabajo, km, vtCode: session.vtCode, ayudanteNombre: session.ayudanteNombre,
   281→   281→        trips: frecuencias.map(f => ({ routeFrom: 'Loja', routeTo: 'Vilcabamba', time: f.hora, income: f.totalRecaudado.toString(), boletos: '0' })),
   282→   282→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   283→   283→      }));
   284→   284→      setGuardadoOffline(true);
   285→   285→      setSaved(true);
   286→   286→    } finally {
   287→   287→      setSaving(false);
   288→   288→    }
   289→   289→  };
   290→   290→
   291→   291→  if (loading) {
   292→   292→    return (
   293→   293→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   294→   294→        <Loader2 className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   295→   295→        <p className="text-[#3A3A3A]">Cargando arqueo general...</p>
   296→   296→      </div>
   297→   297→    );
   298→   298→  }
   299→   299→
   300→   300→  // Saved confirmation screen
   301→   301→  if (saved) {
   302→   302→    return (
   303→   303→      <div className="flex flex-col min-h-[100dvh] bg-[#912D26] text-white">
   304→   304→        <div className="flex-1 flex flex-col items-center justify-center px-6">
   305→   305→          <CheckCircle2 className="w-20 h-20 mb-4" />
   306→   306→          <h1 className="text-2xl font-black mb-2">ARQUEO GENERAL GUARDADO</h1>
   307→   307→          <p className="text-white/80 text-sm mb-1">{session.nombre} — {workDate(session)}</p>
   308→   308→          <p className="text-white/80 text-sm mb-6">{totalVentas} ventas · ${totalIngresosAuto.toFixed(2)} ingresos</p>
   309→   309→
   310→   310→          {guardadoOffline && (
   311→   311→            <div className="bg-yellow-500/20 border border-yellow-400/30 rounded-2xl px-5 py-3 mb-6 text-center">
   312→   312→              <p className="text-yellow-200 text-sm font-bold">Guardado local (sin internet)</p>

   314→   314→            </div>
   315→   315→          )}
   316→   316→
   317→   317→          <div className="w-full max-w-xs space-y-3">
   318→   318→            {connection.pendingCount > 0 && connection.status !== 'offline' && (
   319→   319→              <button onClick={onGoToSync}
   320→   320→                className="w-full py-4 rounded-2xl bg-white text-[#912D26] font-black text-lg flex items-center justify-center gap-2 active:scale-95">
   321→   321→                <Send className="w-5 h-5" /> SINCRONIZAR {connection.pendingCount} VENTAS
   322→   322→              </button>
   323→   323→            )}
   324→   324→            <button onClick={onSaved}
   325→   325→              className="w-full py-4 rounded-2xl bg-white/20 text-white font-bold text-lg flex items-center justify-center gap-2 active:scale-95">
   326→   326→              FINALIZAR
   327→   327→            </button>
   328→   328→          </div>
   329→   329→        </div>
   330→   330→      </div>
   331→   331→    );
   332→   332→  }
   333→   333→
   334→   334→  // Confirm save dialog
   335→   335→  if (confirmSave) {
   336→   336→    return (
   337→   337→      <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   338→   338→        <div className="flex-1 flex items-center justify-center px-4">
   339→   339→          <div className="bg-white rounded-2xl border border-gray-200 p-6 max-w-md w-full shadow-lg">
   340→   340→            <h2 className="text-lg font-bold text-[#3A3A3A] mb-4">Confirmar Arqueo General</h2>
   341→   341→            <div className="bg-gray-50 rounded-xl p-4 space-y-2 mb-4">
   342→   342→              <div className="flex justify-between text-sm">
   343→   343→                <span className="text-gray-500">Producción</span>
   344→   344→                <span className="font-bold text-[#3A3A3A]">${production.toFixed(2)}</span>
   345→   345→              </div>
   346→   346→              {totalCajaComunPasajeros > 0 && (
   347→   347→                <div className="flex justify-between text-sm">
   348→   348→                  <span className="text-purple-700">Caja Comun (Ofi.Loja): {totalCajaComunPasajeros} boletos</span>
   349→   349→                  <span className="font-bold text-purple-700">${totalCajaComunMonto.toFixed(2)}</span>
   350→   350→                </div>
   351→   351→              )}
   352→   352→              <div className="flex justify-between text-sm">
   353→   353→                <span className="text-gray-500">Total Gastos</span>
   354→   354→                <span className="font-bold text-red-600">${totalGastos.toFixed(2)}</span>
   355→   355→              </div>
   356→   356→              <hr className="border-gray-200" />
   357→   357→              <div className="flex justify-between text-sm">
   358→   358→                <span className="text-gray-600">Entrega Ayudante</span>

   361→   361→                </span>
   362→   362→              </div>
   363→   363→              <div className="flex justify-between text-sm">
   364→   364→                <span className="text-gray-600">Entrega Compañía</span>
   365→   365→                <span className={`font-bold ${entregaCompania >= 0 ? 'text-green-600' : 'text-red-600'}`}>

   367→   367→                </span>
   368→   368→              </div>
   369→   369→            </div>
   370→   370→            <p className="text-sm text-[#912D26] mb-4">
   371→   371→              Entrega Ayudante: <strong>${entregaAyudante.toFixed(2)}</strong> —
   372→   372→              Entrega Compañía: <strong>${entregaCompania.toFixed(2)}</strong>

   374→   374→            <div className="flex gap-3">
   375→   375→              <button onClick={() => setConfirmSave(false)}
   376→   376→                className="flex-1 py-3 rounded-xl border-2 border-gray-200 text-gray-500 font-bold active:scale-95">
   377→   377→                Cancelar

   379→   379→              <button onClick={handleConfirmSave} disabled={saving}

   382→   382→              </button>
   383→   383→            </div>
   384→   384→          </div>
   385→   385→        </div>
   386→   386→      </div>
   387→   387→    );
   388→   388→  }
   389→   389→
   390→   390→  return (
   391→   391→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   392→   392→      {/* Barra conexion */}
   393→   393→      <div className={`${connection.bgColor} px-4 py-1.5 flex items-center justify-center gap-2 text-xs font-medium ${connection.color}`}>
   394→   394→        <span>{connection.icon}</span>
   395→   395→        <span>{connection.label}</span>
   396→   396→      </div>
