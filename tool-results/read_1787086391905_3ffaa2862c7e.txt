   478→// ─── Funciones de precio con soporte por dirección ───
   479→//
   480→// Lógica de precios (tipo semántico asignado por el API):
   481→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   482→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   483→//
   484→// El API asigna dirección según el origen:
   485→//   Si origen = Loja →  "ida" (alejándose)
   486→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   487→//
   488→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   489→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   490→
   491→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   492→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   493→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   494→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   495→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   496→  return false;
   497→}
   498→
   499→// Obtener precio considerando la dirección del viaje
   500→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   501→  if (viajaHaciaLoja(ruta, tipo)) {
   502→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   503→    if (ruta === 'Loja - El Tambo') {
   504→      const tamboVuelta = preciosVueltaElTambo[parada];
   505→      if (tamboVuelta) return tamboVuelta;
   506→    }
   507→    const vuelta = preciosVuelta[parada];
   508→    if (vuelta) return vuelta;
   509→    // Si la parada no tiene precio de vuelta, buscar en precios base
   510→  }
   511→  // Buscar primero en ida (base), luego en vuelta como fallback
   512→  const ida = preciosIda[parada];
   513→  if (ida) return ida;
   514→  const vuelta = preciosVuelta[parada];
   515→  return vuelta || null;
   516→}

   521→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   522→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   523→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   524→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   525→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   526→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   527→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   528→  let effectiveRuta = ruta;
   529→  let effectiveTipo = tipo;
   530→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   531→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   532→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   533→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   534→    const baseName = ruta.replace(' - Loja', '');
   535→    effectiveRuta = `Loja - ${baseName}`;
   536→    // NO invertir effectiveTipo — ya viene correcto del API
   537→  }
   538→
   539→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   540→  if (!rutaConfig) {
   541→    // Fallback: try original ruta
   542→    const origConfig = RUTA_PARADAS[ruta];
   543→    if (!origConfig) return [];
   544→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   545→    return paradas.map(parada => {
   546→      const precio = getPrecio(parada, ruta, tipo);
   547→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   548→    });
   549→  }
   550→
   551→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   552→  return paradas.map(parada => {
   553→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   554→    return {
   555→      parada,
   556→      normal: precio?.normal ?? 0,
   557→      media: precio?.media ?? 0,
   558→    };
   559→  });
   560→}

   562→// Normalizar ruta y tipo para rutas "X - Loja"
   563→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   564→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   565→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   566→    // El API ya asigna tipo correcto, NO invertir
   567→    const baseName = ruta.replace(' - Loja', '');
   568→    return { ruta: `Loja - ${baseName}`, tipo };
   569→  }
   570→  return { ruta, tipo };
   571→}

   573→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   574→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   575→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   576→  const precio = getPrecio(parada, nr, nt);
   577→  if (!precio) return TARIFA_MINIMA;
   578→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   579→}

   586→// Buscar la ruta más cercana dado un nombre de ruta parcial
   587→export function matchRuta(rutaNombre: string): string {
   588→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   589→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   590→  if (RUTA_PARADAS[normalized]) return normalized;
   591→
   592→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   593→  for (const key of Object.keys(RUTA_PARADAS)) {
   594→    const keyParts = key.split(' - ');
   595→    if (parts.length >= 2 && keyParts.length >= 2) {
   596→      const match = (
   597→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   598→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   599→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   600→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   601→      );
   602→      if (match) return key;
   603→    }
   604→  }
   605→  return rutaNombre;
   606→}
