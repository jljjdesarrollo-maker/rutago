     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo, useRef } from 'react';
     4→import { type VTSession, type FrecuenciaEstado } from './types-boletos';
     5→import { getVentasByFrecuencia } from '@/lib/indexeddb';
     6→import { type ConnectionInfo } from '@/hooks/use-connection';
     7→import {
     8→  ChevronLeft, DollarSign, Camera, X, Save, Loader2,
     9→  CheckCircle2, Send, AlertTriangle, Plus, Trash2, ImagePlus, Sparkles, Pencil, XCircle
    10→} from 'lucide-react';
    11→
    12→interface Props {
    13→  session: VTSession;
    14→  connection: ConnectionInfo;
    15→  onClose: () => void;
    16→  onGoToSync: () => void;
    17→  onSaved: () => void;
    18→}
    19→
    20→interface FrecuenciaResumen {
    21→  estadoId: string;
    22→  nombre: string;
    23→  ruta: string;
    24→  hora: string;
    25→  direccion: string;
    26→  ventasCount: number;
    27→  totalRecaudado: number;        // What the system says
    28→  efectivoContado: number;        // What the helper actually counted
    29→  diferencia: number;             // efectivoContado - totalRecaudado
    30→  boletosCaja: number;
    31→  cajaComunMonto?: number;
    32→  isIngresoEspecial?: boolean;
    33→  ingresoEspecialNota?: string;
    34→  isNoRealizada?: boolean;
    35→  motivoNoRealizada?: string;
    36→}
    37→
    38→interface GastoItem {
    39→  description: string;
    40→  amount: string;
    41→}
    42→
    43→const today = () => new Date().toISOString().split('T')[0];
    44→
    45→// Work date: use session.fecha (date when shift started, not necessarily today)
    46→function workDate(session: VTSession): string {
    47→  return session.fecha || today();
    48→}
    49→
    50→const GASTOS_DEFAULT: GastoItem[] = [
    51→  { description: 'Chofer', amount: '30' },
    52→  { description: 'Ayudante', amount: '20' },
    53→  { description: 'Diesel', amount: '0' },
    54→  { description: 'Plan Renova', amount: '68' },
    55→];
    56→
    57→export function ArqueoGeneralScreen({ session, connection, onClose, onGoToSync, onSaved }: Props) {
    58→  const [frecuencias, setFrecuencias] = useState<FrecuenciaResumen[]>([]);
    59→  const [loading, setLoading] = useState(true);
    60→  const [km, setKm] = useState('');
    61→  const [gastos, setGastos] = useState<GastoItem[]>(GASTOS_DEFAULT);
    62→  const [tickets, setTickets] = useState('');
    63→  const [sobrante, setSobrante] = useState('');
    64→  const [fotoPreview, setFotoPreview] = useState<string | null>(null);
    65→  const [saving, setSaving] = useState(false);
    66→  const [confirmSave, setConfirmSave] = useState(false);
    67→  const [saved, setSaved] = useState(false);
    68→  const [errors, setErrors] = useState<string[]>([]);
    69→  const [guardadoOffline, setGuardadoOffline] = useState(false);
    70→
    71→  // Inline edit caja común
    72→  const [editingCajaComun, setEditingCajaComun] = useState<string | null>(null);
    73→  const [editCajaComunValue, setEditCajaComunValue] = useState('');
    74→
    75→  const fileInputRef = useRef<HTMLInputElement>(null);
    76→  const saveCajaComunToLS = useCallback((estadoId: string, newMonto: number) => {
    77→    const fecha = workDate(session);
    78→    const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
    79→    try {
    80→      const raw = localStorage.getItem(lsKey);
    81→      if (raw) {
    82→        const all = JSON.parse(raw);
    83→        const idx = all.findIndex((e: any) => e.estadoId === estadoId);
    84→        if (idx >= 0) {
    85→          all[idx].cajaComunMonto = newMonto;
    86→          localStorage.setItem(lsKey, JSON.stringify(all));
    87→        }
    88→      }
    89→    } catch (e) { console.error('LS error:', e); }
    90→    setFrecuencias(prev => prev.map(f =>
    91→      f.estadoId === estadoId
    92→        ? { ...f, cajaComunMonto: newMonto }
    93→        : f
    94→    ));
    95→    setEditingCajaComun(null);
    96→  }, [session]);
    97→
    98→  const startEditCajaComun = (estadoId: string, currentMonto: number) => {
    99→    setEditingCajaComun(estadoId);
   100→    setEditCajaComunValue(currentMonto > 0 ? currentMonto.toFixed(2) : '');
   101→  };
   102→
   103→  const loadData = useCallback(async () => {
   104→    try {
   105→      const fecha = workDate(session);
   106→      // Load estados from localStorage (synchronous, reliable)
   107→      const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
   108→      const raw = localStorage.getItem(lsKey);
   109→      if (!raw) { setLoading(false); return; }
   110→      const estadosLS = JSON.parse(raw) as FrecuenciaEstado[];
   111→      const cerradas = estadosLS.filter(e => e.estado === 'cerrada');
   112→      // Ingresos especiales (no_realizada con monto > 0)
   113→      const especiales = estadosLS.filter(e => e.estado === 'no_realizada' && (e.ingresoEspecialMonto || 0) > 0);
   114→      // No realizadas sin ingreso especial
   115→      const noRealizadas = estadosLS.filter(e => e.estado === 'no_realizada' && (e.ingresoEspecialMonto || 0) <= 0);
   116→
   117→      const resumenes: FrecuenciaResumen[] = await Promise.all(
   118→        cerradas.map(async (e) => {
   119→          const ventas = await getVentasByFrecuencia(e.estadoId);
   120→          const sistemaTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   121→          // Use arqueo data if available (what helper actually counted), else system total
   122→          const efectivo = (e as any).arqueoEfectivo ?? sistemaTotal;
   123→          const diff = efectivo - sistemaTotal;
   124→          const ccCount = (e as any).cajaComunCount || 0;
   125→          const ccMonto = (e as any).cajaComunMonto || 0;
   126→          return {
   127→            estadoId: e.estadoId,
   128→            nombre: e.nombre,
   129→            ruta: e.ruta,
   130→            hora: e.hora,
   131→            direccion: e.direccion,
   132→            ventasCount: ventas.length,
   133→            totalRecaudado: sistemaTotal,
   134→            efectivoContado: efectivo,
   135→            diferencia: diff,
   136→            boletosCaja: ccCount,
   137→            cajaComunMonto: ccMonto,
   138→          };
   139→        })
   140→      );
   141→
   142→      // Add ingresos especiales as virtual rows
   143→      especiales.forEach(e => {
   144→        resumenes.push({
   145→          estadoId: e.estadoId,
   146→          nombre: e.nombre,
   147→          ruta: e.ruta,
   148→          hora: e.hora,
   149→          direccion: e.direccion,
   150→          ventasCount: 0,
   151→          totalRecaudado: e.ingresoEspecialMonto || 0,
   152→          efectivoContado: e.ingresoEspecialMonto || 0,
   153→          diferencia: 0,
   154→          boletosCaja: 0,
   155→          cajaComunMonto: 0,
   156→          isIngresoEspecial: true,
   157→          ingresoEspecialNota: e.ingresoEspecialNota,
   158→          isNoRealizada: true,
   159→          motivoNoRealizada: e.motivoNoRealizada,
   160→        });
   161→      });
   162→      // Add no realizadas (sin ingreso) as virtual rows
   163→      noRealizadas.forEach(e => {
   164→        resumenes.push({
   165→          estadoId: e.estadoId,
   166→          nombre: e.nombre,
   167→          ruta: e.ruta,
   168→          hora: e.hora,
   169→          direccion: e.direccion,
   170→          ventasCount: 0,
   171→          totalRecaudado: 0,
   172→          efectivoContado: 0,
   173→          diferencia: 0,
   174→          boletosCaja: 0,
   175→          cajaComunMonto: 0,
   176→          isNoRealizada: true,
   177→          motivoNoRealizada: e.motivoNoRealizada,
   178→        });
   179→      });
   180→      setFrecuencias(resumenes);
   181→    } catch (err) {
   182→      console.error('Error cargando resumen:', err);
   183→    } finally {
   184→      setLoading(false);
   185→    }
   186→  }, [session.vtCode]);
   187→
   188→  useEffect(() => { loadData(); }, [loadData]);
   189→
   190→  const totalIngresosAuto = useMemo(() =>
   191→    frecuencias.reduce((s, f) => s + f.totalRecaudado, 0),
   192→    [frecuencias]
   193→  );
   194→  const totalEfectivoReal = useMemo(() =>
   195→    frecuencias.reduce((s, f) => s + f.efectivoContado, 0),
   196→    [frecuencias]
   197→  );
   198→  const totalDiferencia = useMemo(() =>
   199→    frecuencias.reduce((s, f) => s + f.diferencia, 0),
   200→    [frecuencias]
   201→  );
   202→  const totalVentas = useMemo(() =>
   203→    frecuencias.reduce((s, f) => s + f.ventasCount, 0),
   204→    [frecuencias]
   205→  );
   206→  const totalCajaComunPasajeros = useMemo(() =>
   207→    frecuencias.reduce((s, f) => s + f.boletosCaja, 0),
   208→    [frecuencias]
   209→  );
   210→  const totalCajaComunMonto = useMemo(() =>
   211→    frecuencias.reduce((s, f) => s + (f.cajaComunMonto || 0), 0),
   212→    [frecuencias]
   213→  );
   214→  const totalGastos = useMemo(() =>
   215→    gastos.reduce((s, g) => s + (parseFloat(g.amount) || 0), 0),
   216→    [gastos]
   217→  );
   218→  const ticketsNum = parseFloat(tickets) || 0;
   219→  const sobranteNum = parseFloat(sobrante) || 0;
   220→  // PRODUCCION = efectivo real contado por el ayudante + caja comun + sobrante ajuste manual
   221→  const production = totalEfectivoReal + totalCajaComunMonto + sobranteNum;
   222→  // ENTREGA AYUDANTE = efectivo real + sobrante - gastos (la caja común NO pasa por el ayudante)
   223→  const entregaAyudante = totalEfectivoReal + sobranteNum - totalGastos;
   224→  // Lógica dual: si hay caja común se descuentan tickets, si no (pre-Jun 2026) entrega compañía = 0
   225→  const entregaCompania = totalCajaComunMonto > 0 ? totalCajaComunMonto - ticketsNum : 0;
   226→
   227→  const addGasto = () => setGastos(prev => [...prev, { description: '', amount: '0' }]);
   228→  const removeGasto = (i: number) => setGastos(prev => prev.filter((_, idx) => idx !== i));
   229→  const updateGasto = (i: number, field: keyof GastoItem, value: string) => {
   230→    setGastos(prev => {
   231→      const next = [...prev];
   232→      next[i] = { ...next[i], [field]: value };
   233→      return next;
   234→    });
   235→  };
   236→
   237→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   238→    const file = e.target.files?.[0];
   239→    if (!file || !file.type.startsWith('image/')) return;
   240→    e.target.value = '';
   241→    // Comprimir imagen antes de almacenar (max ~200KB base64)
   242→    const img = new Image();
   243→    const url = URL.createObjectURL(file);
   244→    img.onload = () => {
   245→      URL.revokeObjectURL(url);
   246→      const MAX_DIM = 1200;
   247→      const QUALITY = 0.6;
   248→      let w = img.width, h = img.height;
   249→      if (w > MAX_DIM || h > MAX_DIM) {
   250→        if (w > h) { h = Math.round(h * MAX_DIM / w); w = MAX_DIM; }
   251→        else { w = Math.round(w * MAX_DIM / h); h = MAX_DIM; }
   252→      }
   253→      const canvas = document.createElement('canvas');
   254→      canvas.width = w; canvas.height = h;
   255→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h);
   256→      setFotoPreview(canvas.toDataURL('image/jpeg', QUALITY));
   257→    };
   258→    img.onerror = () => URL.revokeObjectURL(url);
   259→    img.src = url;
   260→  };
   261→
   262→  const handleSaveOffline = () => {
   263→    const errs: string[] = [];
   264→    if (!km.trim()) errs.push('Kilometraje es obligatorio');
   265→    if (!fotoPreview) errs.push('Foto del cuaderno es obligatoria');
   266→    if (frecuencias.length === 0) errs.push('No hay frecuencias cerradas');
   267→    // Gastos pueden ser $0 si no hubo (ej. unidad parada)
   268→    if (errs.length > 0) {
   269→      setErrors(errs);
   270→      return;
   271→    }
   272→    setErrors([]);
   273→    setConfirmSave(true);
   274→  };
   275→
   276→  const handleConfirmSave = async () => {
   277→    setConfirmSave(false);
   278→    setSaving(true);
   279→
   280→    try {
   281→      const fechaTrabajo = workDate(session);
   282→
   283→      // Build trips from frecuencias (all: cerradas, ingresos especiales, no realizadas)
   284→      const trips = frecuencias.map((f, idx) => ({
   285→        routeFrom: f.isNoRealizada && !f.isIngresoEspecial
   286→          ? '-'
   287→          : (f.direccion === 'ida' ? 'Loja' : f.ruta.split(' - ')[1]?.trim() || 'Loja'),
   288→        routeTo: f.isNoRealizada && !f.isIngresoEspecial
   289→          ? '-'
   290→          : (f.direccion === 'ida' ? f.ruta.split(' - ')[1]?.trim() || 'Vilcabamba' : 'Loja'),
   291→        time: f.hora,
   292→        income: f.totalRecaudado.toString(),
   293→        efectivoReal: f.efectivoContado.toString(),
   294→        boletos: '0',
   295→        cajaComunPasajeros: f.boletosCaja.toString(),
   296→        cajaComunMonto: (f.cajaComunMonto || 0).toString(),
   297→        tipo: f.isIngresoEspecial ? 'ingreso_especial' : f.isNoRealizada ? 'no_realizada' : 'frecuencia',
   298→        motivo: f.motivoNoRealizada || undefined,
   299→        notaEspecial: f.ingresoEspecialNota || undefined,
   300→      }));
   301→
   302→      const body = {
   303→        date: workDate(session),
   304→        km: km,
   305→        conductor: '',
   306→        ayudanteNombre: session.ayudanteNombre,
   307→        vtCode: session.vtCode,
   308→        trips,
   309→        expenses: gastos,
   310→        tickets: tickets || '0',
   311→        cajaComun: totalCajaComunMonto,
   312→        sobrante: sobrante || '0',
   313→        photoUrl: fotoPreview,
   314→      };
   315→
   316→      const isOnline = navigator.onLine;
   317→
   318→      if (isOnline) {
   319→        const res = await fetch('/api/records', {
   320→          method: 'POST',
   321→          headers: { 'Content-Type': 'application/json' },
   322→          body: JSON.stringify(body),
   323→        });
   324→        if (res.ok) {
   325→          setSaved(true);
   326→        } else {
   327→          const result = await res.json().catch(() => ({ error: 'Error desconocido' }));
   328→          console.error('Error guardando arqueo en servidor:', result.error);
   329→          // Save offline if server fails
   330→          localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   331→          setGuardadoOffline(true);
   332→          setSaved(true);
   333→        }
   334→      } else {
   335→        // Save to localStorage for later sync
   336→        localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   337→        setGuardadoOffline(true);
   338→        setSaved(true);
   339→      }
   340→    } catch (err) {
   341→      console.error('Error guardando arqueo:', err);
   342→      localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify({
   343→        date: fechaTrabajo, km, vtCode: session.vtCode, ayudanteNombre: session.ayudanteNombre,
   344→        trips: frecuencias.map(f => ({
   345→          routeFrom: f.isNoRealizada ? '-' : 'Loja',
   346→          routeTo: f.isNoRealizada ? '-' : 'Vilcabamba',
   347→          time: f.hora, income: f.totalRecaudado.toString(), boletos: '0',
   348→          tipo: f.isIngresoEspecial ? 'ingreso_especial' : f.isNoRealizada ? 'no_realizada' : 'frecuencia',
   349→          motivo: f.motivoNoRealizada || undefined,
   350→          notaEspecial: f.ingresoEspecialNota || undefined,
   351→        })),
   352→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   353→      }));
   354→      setGuardadoOffline(true);
   355→      setSaved(true);
   356→    } finally {
   357→      setSaving(false);
   358→    }
   359→  };
   360→
   361→  if (loading) {
   362→    return (
   363→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   364→        <Loader2 className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   365→        <p className="text-[#3A3A3A]">Cargando arqueo general...</p>
   366→      </div>
   367→    );
   368→  }
   369→
   370→  // Saved confirmation screen
   371→  if (saved) {
   372→    return (
   373→      <div className="flex flex-col min-h-[100dvh] bg-[#912D26] text-white">
   374→        <div className="flex-1 flex flex-col items-center justify-center px-6">
   375→          <CheckCircle2 className="w-20 h-20 mb-4" />
   376→          <h1 className="text-2xl font-black mb-2">ARQUEO GENERAL GUARDADO</h1>
   377→          <p className="text-white/80 text-sm mb-1">{session.nombre} — {workDate(session)}</p>
   378→          <p className="text-white/80 text-sm mb-6">{totalVentas} ventas · ${totalIngresosAuto.toFixed(2)} ingresos</p>
   379→
   380→          {guardadoOffline && (
   381→            <div className="bg-yellow-500/20 border border-yellow-400/30 rounded-2xl px-5 py-3 mb-6 text-center">
   382→              <p className="text-yellow-200 text-sm font-bold">Guardado local (sin internet)</p>
   383→              <p className="text-yellow-200/70 text-xs mt-1">Se sincronizará cuando haya conexión</p>
   384→            </div>
   385→          )}
   386→
   387→          <div className="w-full max-w-xs space-y-3">
   388→            {connection.pendingCount > 0 && connection.status !== 'offline' && (
   389→              <button onClick={onGoToSync}
   390→                className="w-full py-4 rounded-2xl bg-white text-[#912D26] font-black text-lg flex items-center justify-center gap-2 active:scale-95">
   391→                <Send className="w-5 h-5" /> SINCRONIZAR {connection.pendingCount} VENTAS
   392→              </button>
   393→            )}
   394→            <button onClick={onSaved}
   395→              className="w-full py-4 rounded-2xl bg-white/20 text-white font-bold text-lg flex items-center justify-center gap-2 active:scale-95">
   396→              FINALIZAR
   397→            </button>
   398→          </div>
   399→        </div>
   400→      </div>
   401→    );
   402→  }
   403→
   404→  // Confirm save dialog
   405→  if (confirmSave) {
   406→    return (
   407→      <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   408→        <div className="flex-1 flex items-center justify-center px-4">
   409→          <div className="bg-white rounded-2xl border border-gray-200 p-6 max-w-md w-full shadow-lg">
   410→            <h2 className="text-lg font-bold text-[#3A3A3A] mb-4">Confirmar Arqueo General</h2>
   411→            <div className="bg-gray-50 rounded-xl p-4 space-y-2 mb-4">
   412→              <div className="flex justify-between text-sm">
   413→                <span className="text-gray-500">Producción</span>
   414→                <span className="font-bold text-[#3A3A3A]">${production.toFixed(2)}</span>
   415→              </div>
   416→              {totalCajaComunPasajeros > 0 && (
   417→                <div className="flex justify-between text-sm">
   418→                  <span className="text-purple-700">Caja Comun (Ofi.Loja): {totalCajaComunPasajeros} boletos</span>
   419→                  <span className="font-bold text-purple-700">${totalCajaComunMonto.toFixed(2)}</span>
   420→                </div>
   421→              )}
   422→              <div className="flex justify-between text-sm">
   423→                <span className="text-gray-500">Total Gastos</span>
   424→                <span className="font-bold text-red-600">${totalGastos.toFixed(2)}</span>
   425→              </div>
   426→              <hr className="border-gray-200" />
   427→              <div className="flex justify-between text-sm">
   428→                <span className="text-gray-600">Entrega Ayudante</span>
   429→                <span className={`font-bold ${entregaAyudante >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   430→                  ${entregaAyudante.toFixed(2)}
   431→                </span>
   432→              </div>
   433→              <div className="flex justify-between text-sm">
   434→                <span className="text-gray-600">Entrega Compañía</span>
   435→                <span className={`font-bold ${entregaCompania >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   436→                  ${entregaCompania.toFixed(2)}
   437→                </span>
   438→              </div>
   439→            </div>
   440→            <p className="text-sm text-[#912D26] mb-4">
   441→              Entrega Ayudante: <strong>${entregaAyudante.toFixed(2)}</strong> —
   442→              Entrega Compañía: <strong>${entregaCompania.toFixed(2)}</strong>
   443→            </p>
   444→            <div className="flex gap-3">
   445→              <button onClick={() => setConfirmSave(false)}
   446→                className="flex-1 py-3 rounded-xl border-2 border-gray-200 text-gray-500 font-bold active:scale-95">
   447→                Cancelar
   448→              </button>
   449→              <button onClick={handleConfirmSave} disabled={saving}
   450→                className={`flex-1 py-3 rounded-xl font-black active:scale-95 ${saving ? 'bg-[#D6D6D6] text-gray-400 cursor-not-allowed' : 'bg-[#912D26] text-white'}`}>
   451→                {saving ? 'Guardando...' : 'CONFIRMAR'}
   452→              </button>
   453→            </div>
   454→          </div>
   455→        </div>
   456→      </div>
   457→    );
   458→  }
   459→
   460→  return (
   461→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   462→      {/* Barra conexion */}
   463→      <div className={`${connection.bgColor} px-4 py-1.5 flex items-center justify-center gap-2 text-xs font-medium ${connection.color}`}>
   464→        <span>{connection.icon}</span>
