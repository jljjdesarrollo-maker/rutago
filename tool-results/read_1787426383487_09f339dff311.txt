   559→// ─── Funciones de precio con soporte por dirección ───
   560→//
   561→// Lógica de precios (tipo semántico asignado por el API):
   562→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   563→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   564→//
   565→// El API asigna dirección según el origen:
   566→//   Si origen = Loja →  "ida" (alejándose)
   567→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   568→//
   569→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   570→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   571→
   572→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   573→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   574→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   575→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   576→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   577→  return false;
   578→}
   579→
   580→// Obtener precio considerando la dirección del viaje
   581→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   582→  if (viajaHaciaLoja(ruta, tipo)) {
   583→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   584→    if (ruta === 'Loja - El Tambo') {
   585→      const tamboVuelta = preciosVueltaElTambo[parada];
   586→      if (tamboVuelta) return tamboVuelta;
   587→    }
   588→    const vuelta = preciosVuelta[parada];
   589→    if (vuelta) return vuelta;
   590→    // Si la parada no tiene precio de vuelta, buscar en precios base
   591→  }
   592→  // Buscar primero en ida (base), luego en vuelta como fallback
   593→  const ida = preciosIda[parada];
   594→  if (ida) return ida;
   595→  const vuelta = preciosVuelta[parada];
   596→  return vuelta || null;
   597→}

   602→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   603→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   604→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   605→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   606→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   607→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   608→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   609→  let effectiveRuta = ruta;
   610→  let effectiveTipo = tipo;
   611→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   612→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   613→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   614→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   615→    const baseName = ruta.replace(' - Loja', '');
   616→    effectiveRuta = `Loja - ${baseName}`;
   617→    // NO invertir effectiveTipo — ya viene correcto del API
   618→  }
   619→
   620→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   621→  if (!rutaConfig) {
   622→    // Fallback: try original ruta
   623→    const origConfig = RUTA_PARADAS[ruta];
   624→    if (!origConfig) return [];
   625→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   626→    return paradas.map(parada => {
   627→      const precio = getPrecio(parada, ruta, tipo);
   628→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   629→    });
   630→  }
   631→
   632→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   633→  return paradas.map(parada => {
   634→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   635→    return {
   636→      parada,
   637→      normal: precio?.normal ?? 0,
   638→      media: precio?.media ?? 0,
   639→    };
   640→  });
   641→}

   643→// Normalizar ruta y tipo para rutas "X - Loja"
   644→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   645→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   646→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   647→    // El API ya asigna tipo correcto, NO invertir
   648→    const baseName = ruta.replace(' - Loja', '');
   649→    return { ruta: `Loja - ${baseName}`, tipo };
   650→  }
   651→  return { ruta, tipo };
   652→}

   654→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   655→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   656→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);

   658→  if (!precio) return TARIFA_MINIMA;

   667→// Buscar la ruta más cercana dado un nombre de ruta parcial
   668→export function matchRuta(rutaNombre: string): string {
   669→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   670→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   671→  if (RUTA_PARADAS[normalized]) return normalized;
   672→
   673→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   674→  for (const key of Object.keys(RUTA_PARADAS)) {
   675→    const keyParts = key.split(' - ');
   676→    if (parts.length >= 2 && keyParts.length >= 2) {
   677→      const match = (
   678→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   679→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   680→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   681→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   682→      );
   683→      if (match) return key;
   684→    }
   685→  }
   686→  return rutaNombre;
   687→}
