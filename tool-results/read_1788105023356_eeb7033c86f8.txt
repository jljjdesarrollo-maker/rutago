    55→
    56→function getVTRouteType(frecuencias?: { routeFrom: string; routeTo: string }[]): VTRouteType {
    57→  if (!frecuencias || frecuencias.length === 0) return 'vilcabamba';
    58→  const allDests = [...frecuencias.map(f => f.routeTo), ...frecuencias.map(f => f.routeFrom)];
    59→  if (allDests.some(d => d === 'El Tambo')) return 'el_tambo';
    60→  if (allDests.some(d => d === 'La Elvira')) return 'la_elvira';
    61→  if (allDests.some(d => d === 'Yangana')) return 'yangana';
    62→  if (allDests.some(d => d === 'Zahuayco')) return 'zahuayco';
    63→  return 'vilcabamba';
    64→}

   477→              {printerLog.map((l, i) => (
   478→                <p key={i} className={`text-[9px] font-mono leading-relaxed ${
   479→                  l.includes('ERROR') ? 'text-red-400' :
   480→                  l.includes('OK') || l.includes('completada') ? 'text-green-400' :
   481→                  'text-gray-400'
   482→                }`}>{l}</p>
   483→              ))}
   484→            </div>
   485→          </div>
   486→        )}
   487→      </div>
   488→
   489→      <div className="flex-1 px-5 py-5 space-y-4">
   490→        {/* Ayudante activo — compacto */}
   491→        {ayudante ? (
   492→          <div className="flex items-center gap-3 bg-green-50 border border-green-200 rounded-xl px-4 py-3">
   493→            <CheckCircle className="w-5 h-5 text-green-600 flex-shrink-0" />
   494→            <div>
   495→              <p className="font-semibold text-green-800 text-sm">{ayudante.nombre}</p>
   496→              <p className="text-green-600 text-[10px]">Turno activo</p>
   497→            </div>
   498→          </div>
   499→        ) : (
   500→          <div className="bg-yellow-50 border border-yellow-200 rounded-xl px-4 py-3">
   501→            <p className="text-yellow-700 text-sm">No hay ayudante activo asignado</p>
   502→            <p className="text-yellow-600 text-xs mt-1">Configure un ayudante desde el panel de administracion</p>
   503→          </div>
   504→        )}
   505→
   506→        {/* Selector de fecha */}
   507→        <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-100">
   508→          <h2 className="text-base font-bold text-[#3A3A3A] mb-3 flex items-center gap-2">
   509→            <CalendarDays className="w-4 h-4 text-[#912D26]" /> Fecha del turno
   510→          </h2>
   511→          <input
   512→            type="date"
   513→            value={selectedDate}
   514→            max={todayStr}
   515→            onChange={(e) => {
   516→              const newDate = e.target.value;
   517→              setSelectedDate(newDate);
   518→              setDateWarning('');
   519→
   520→              // Validar si hay datos existentes para esa fecha + VT seleccionado
   521→              if (newDate && selectedVT) {
   522→                setCheckingDate(true);
   523→                const estadosKey = `rg_estados_${selectedVT}_${newDate}`;
   524→                const raw = localStorage.getItem(estadosKey);
   525→                if (raw) {
   526→                  try {
   527→                    const estados: { estado: string }[] = JSON.parse(raw);
   528→                    const hasData = estados.some(e => e.estado !== 'pendiente' || e.ventasCount > 0);
   529→                    if (hasData) {
   530→                      setDateWarning(`Ya existen datos para ${selectedVT} en esta fecha. Se eliminaran al iniciar.`);
   531→                    }
   532→                  } catch { /* ignore */ }
   533→                }
   534→                // Also check for ventas in IndexedDB
   535→                countVentasPendientesByVT(selectedVT, newDate).then(count => {
   536→                  if (count > 0) {
   537→                    setDateWarning(prev => prev
   538→                      ? `${prev} (${count} ventas pendientes)`
   539→                      : `Hay ${count} ventas sin sincronizar para esta fecha.`
   540→                    );
   541→                  }
   542→                  setCheckingDate(false);
   543→                }).catch(() => setCheckingDate(false));
   544→              }
   545→            }}
   546→            className="w-full px-4 py-3 rounded-xl border-2 border-gray-200 text-[#3A3A3A] font-semibold text-base focus:border-[#912D26] focus:outline-none transition-colors [color-scheme:light]"
   547→          />
   548→          {dateWarning && (
   549→            <div className="mt-2 flex items-start gap-2 bg-amber-50 border border-amber-200 rounded-lg px-3 py-2">
   550→              <AlertTriangle className="w-4 h-4 text-amber-500 flex-shrink-0 mt-0.5" />
   551→              <p className="text-amber-700 text-xs">{dateWarning}</p>
   552→            </div>
   553→          )}
   554→          {selectedDate !== todayStr && (
   555→            <p className="mt-2 text-xs text-blue-500 flex items-center gap-1">
   556→              <CalendarDays className="w-3 h-3" />
   557→              Fecha seleccionada: {selectedDate} (no es hoy)
   558→            </p>
   559→          )}
   560→          {checkingDate && (
   561→            <div className="mt-2 flex items-center gap-1.5 text-gray-400 text-xs">
   562→              <Loader2 className="w-3 h-3 animate-spin" /> Verificando datos...
   563→            </div>
   564→          )}
   565→        </div>
   566→
   567→        {/* Selector de VT — grid grande */}
   568→        {vts.length === 0 ? (
   569→          <div className="bg-yellow-50 border border-yellow-200 rounded-2xl p-5 text-center">
   570→            <p className="text-yellow-700 font-medium">No hay unidades activas</p>
   571→            <p className="text-yellow-600 text-sm mt-1">Ejecuta el seed: POST /api/seed-vts</p>
   572→          </div>
   573→        ) : (
   574→          <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-100">
   575→            <h2 className="text-base font-bold text-[#3A3A3A] mb-3 flex items-center gap-2">
   576→              <Bus className="w-4 h-4 text-[#912D26]" /> Unidad
   577→            </h2>
   578→            <div className={`grid gap-3 ${vts.length <= 2 ? 'grid-cols-2' : vts.length <= 4 ? 'grid-cols-2' : 'grid-cols-3'}`}>
   579→              {vts.map(vt => {
   580→                const routeType = getVTRouteType(vt.frecuencias);
   581→                const style = ROUTE_STYLES[routeType];
   582→                const isSelected = selectedVT === vt.codigo;
   583→                return (
   584→                  <button key={vt.codigo}
   585→                    onClick={() => vts.length === 1 ? startSession(vt.codigo) : setSelectedVT(vt.codigo)}
   586→                    className={`relative py-5 px-3 rounded-xl text-center transition-all active:scale-95 border ${
   587→                      isSelected
   588→                        ? `${style.selectedBg} text-white shadow-md ring-2 ${style.selectedRing}`
   589→                        : `${style.bg} ${style.border} ${style.text} hover:opacity-80 active:opacity-70`
   590→                    }`}
   591→                  >
   592→                    <div className="text-base font-bold">{vt.codigo}</div>
   593→                    <div className={`text-[10px] mt-0.5 ${isSelected ? 'text-white/70' : style.subtext}`}>{vt.nombre}</div>
   594→                    {style.label && !isSelected && (
   595→                      <span className={`absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full ${style.badgeBg} ${style.badgeText} shadow-sm`}>
   596→                        {style.label}
   597→                      </span>
   598→                    )}
   599→                    {style.label && isSelected && (
   600→                      <span className="absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full bg-white/25 text-white shadow-sm">
   601→                        {style.label}
   602→                      </span>
   603→                    )}
   604→                  </button>
   605→                );
   606→              })}
   607→            </div>
   608→          </div>
   609→        )}
   610→
   611→        {/* Boton Iniciar — grande y prominente */}
   612→        <button onClick={handleStart} disabled={!selectedVT || !ayudante || pendingVentasCount > 0}
   613→          className={`w-full py-5 rounded-2xl font-black text-xl flex items-center justify-center gap-3 transition-all ${
   614→            selectedVT && ayudante && pendingVentasCount === 0
   615→              ? 'bg-[#912D26] text-white shadow-xl shadow-red-300 active:scale-[0.97]'
   616→              : 'bg-[#D6D6D6] text-gray-400 cursor-not-allowed'
   617→          }`}
   618→        >
   619→          {pendingVentasCount > 0
   620→            ? `ESPERANDO SYNC — ${pendingVentasCount} VENTAS`
   621→            : 'Iniciar Turno'
   622→          } <ArrowRight className="w-6 h-6" />
   623→        </button>
   624→      </div>
   625→
   626→      {/* Ventas pendientes — bloquear inicio de nuevo VT */}
   627→      {pendingVentasCount > 0 && pendingCheckDone && (
   628→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-[#912D26] to-[#b33d34] p-4 shadow-2xl z-40">
   629→          <div className="max-w-lg mx-auto">
   630→            <div className="flex items-center gap-2 mb-2">
   631→              <AlertTriangle className="w-5 h-5 text-white" />
   632→              <p className="text-white font-bold text-sm">Ventas sin sincronizar</p>
   633→            </div>
   634→            <p className="text-white/90 text-xs mb-3">
   635→              Tienes <strong>{pendingVentasCount} venta{pendingVentasCount !== 1 ? 's' : ''}</strong> pendiente{pendingVentasCount !== 1 ? 's' : ''} de un turno anterior. Debes sincronizar antes de iniciar uno nuevo.
   636→            </p>
   637→            {navigator.onLine ? (
   638→              <button onClick={handleForceSync} disabled={syncingPending}
   639→                className="w-full py-3 rounded-xl bg-white text-[#912D26] font-black text-sm flex items-center justify-center gap-2 active:scale-[0.98] disabled:opacity-60">
   640→                {syncingPending ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Wifi className="w-4 h-4" />}
   641→                {syncingPending ? 'SINCRONIZANDO...' : `SINCRONIZAR ${pendingVentasCount} VENTAS`}
   642→              </button>
   643→            ) : (
   644→              <div className="w-full py-3 rounded-xl bg-white/20 text-white/80 text-sm flex items-center justify-center gap-2">
   645→                <WifiOff className="w-4 h-4" /> SIN INTERNET — CONECTA PARA SINCRONIZAR
   646→              </div>
   647→            )}
   648→          </div>
   649→        </div>
   650→      )}
   651→
   652→      {/* Existing unfinished session banner */}
   653→      {existingUnfinished && existingSession && (
   654→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-orange-500 to-red-500 p-4 shadow-2xl z-40">
   655→          <div className="max-w-lg mx-auto">
   656→            <div className="flex items-center gap-2 mb-2">
   657→              <AlertTriangle className="w-5 h-5 text-white" />
   658→              <p className="text-white font-bold text-sm">Turno sin terminar</p>
   659→            </div>
   660→            <p className="text-white/90 text-xs mb-3">
   661→              Tienes un turno pendiente en <strong>{existingSession.nombre}</strong> ({existingSession.vtCode}) con frecuencias abiertas.
   662→            </p>
   663→            <div className="flex gap-2">
   664→              <button onClick={handleResumeOld}
   665→                className="flex-1 py-2.5 rounded-xl bg-white text-orange-600 font-bold text-sm active:scale-[0.98]">
   666→                Continuar VT {existingSession.vtCode}
   667→              </button>
   668→              <button onClick={() => setConfirmNewSession(true)}
   669→                className="flex-1 py-2.5 rounded-xl bg-white/20 text-white font-semibold text-sm active:scale-[0.98]">
   670→                Iniciar nuevo
   671→              </button>
   672→            </div>
   673→          </div>
   674→        </div>
   675→      )}
   676→
   677→      {/* Confirm new session modal */}
   678→      {confirmNewSession && existingSession && (
   679→        <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
   680→          <div className="bg-white rounded-2xl w-full max-w-sm p-6 text-center">
   681→            <div className="w-16 h-16 bg-red-100 rounded-full flex items-center justify-center mx-auto mb-4">
   682→              <AlertTriangle className="w-8 h-8 text-[#912D26]" />
   683→            </div>
   684→            <h3 className="text-lg font-bold text-[#3A3A3A] mb-2">Abandonar turno anterior?</h3>
   685→            <div className="bg-red-50 rounded-xl p-3 mb-3">
   686→              <p className="text-sm text-[#912D26] font-semibold">
   687→                {existingSession.nombre} ({existingSession.vtCode}) — frecuencias sin completar.
   688→              </p>
   689→            </div>
   690→            <div className="bg-yellow-50 border border-yellow-200 rounded-xl p-3 mb-4">
   691→              <p className="text-xs text-yellow-800 font-bold">
   692→                Se ELIMINARAN todos los datos del turno anterior. Solo puede haber un turno activo por dia.
   693→              </p>
   694→            </div>
   695→            <div className="flex gap-2">
   696→              <button onClick={() => setConfirmNewSession(false)}
   697→                className="flex-1 py-3 rounded-xl bg-[#912D26] text-white font-bold text-sm active:scale-[0.98] shadow-lg shadow-red-200">
   698→                Cancelar
   699→              </button>
   700→              <button onClick={handleForceNew}
   701→                className="flex-1 py-3 rounded-xl border-2 border-red-200 text-[#912D26] font-bold text-sm active:scale-[0.98]">
   702→                Nuevo Turno
   703→              </button>
   704→            </div>
   705→          </div>
   706→        </div>
   707→      )}
   708→    </div>
   709→  );
