   601→// ─── Funciones de precio con soporte por dirección ───
   602→//
   603→// Lógica de precios (tipo semántico asignado por el API):
   604→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   605→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   606→//
   607→// El API asigna dirección según el origen:
   608→//   Si origen = Loja →  "ida" (alejándose)
   609→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   610→//
   611→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   612→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   613→
   614→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   615→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   616→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   617→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   618→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   619→  return false;
   620→}
   621→
   622→// Obtener precio considerando la dirección del viaje
   623→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   624→  if (viajaHaciaLoja(ruta, tipo)) {
   625→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   626→    if (ruta === 'Loja - El Tambo') {
   627→      const tamboVuelta = preciosVueltaElTambo[parada];
   628→      if (tamboVuelta) return tamboVuelta;
   629→    }
   630→    const vuelta = preciosVuelta[parada];
   631→    if (vuelta) return vuelta;
   632→    // Si la parada no tiene precio de vuelta, buscar en precios base
   633→  }
   634→  // Buscar primero en ida (base), luego en vuelta como fallback
   635→  const ida = preciosIda[parada];
   636→  if (ida) return ida;
   637→  const vuelta = preciosVuelta[parada];
   638→  return vuelta || null;
   639→}

   644→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   645→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   646→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   647→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   648→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   649→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   650→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   651→  let effectiveRuta = ruta;
   652→  let effectiveTipo = tipo;
   653→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   654→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   655→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   656→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   657→    const baseName = ruta.replace(' - Loja', '');
   658→    effectiveRuta = `Loja - ${baseName}`;
   659→    // NO invertir effectiveTipo — ya viene correcto del API
   660→  }
   661→
   662→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   663→  if (!rutaConfig) {
   664→    // Fallback: try original ruta
   665→    const origConfig = RUTA_PARADAS[ruta];
   666→    if (!origConfig) return [];
   667→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   668→    return paradas.map(parada => {
   669→      const precio = getPrecio(parada, ruta, tipo);
   670→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   671→    });
   672→  }
   673→
   674→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   675→  return paradas.map(parada => {
   676→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   677→    return {
   678→      parada,
   679→      normal: precio?.normal ?? 0,
   680→      media: precio?.media ?? 0,
   681→    };
   682→  });
   683→}

   685→// Normalizar ruta y tipo para rutas "X - Loja"
   686→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   687→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   688→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   689→    // El API ya asigna tipo correcto, NO invertir
   690→    const baseName = ruta.replace(' - Loja', '');
   691→    return { ruta: `Loja - ${baseName}`, tipo };
   692→  }
   693→  return { ruta, tipo };
   694→}

   696→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   697→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   698→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   699→  const precio = getPrecio(parada, nr, nt);
   700→  if (!precio) return TARIFA_MINIMA;
   701→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   702→}

   709→// Buscar la ruta más cercana dado un nombre de ruta parcial
   710→export function matchRuta(rutaNombre: string): string {
   711→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   712→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   713→  if (RUTA_PARADAS[normalized]) return normalized;
   714→
   715→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   716→  for (const key of Object.keys(RUTA_PARADAS)) {
   717→    const keyParts = key.split(' - ');
   718→    if (parts.length >= 2 && keyParts.length >= 2) {
   719→      const match = (
   720→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   721→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   722→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   723→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   724→      );
   725→      if (match) return key;
   726→    }
   727→  }
   728→  return rutaNombre;
   729→}
