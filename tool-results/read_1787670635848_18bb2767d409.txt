   122→   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];

   124→   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→   127→  return 'yellow';
   128→   128→}

   758→   758→// ─── Funciones de precio con soporte por dirección ───
   759→   759→//
   760→   760→// Lógica de precios (tipo semántico asignado por el API):
   761→   761→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   762→   762→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   763→   763→//
   764→   764→// El API asigna dirección según el origen:
   765→   765→//   Si origen = Loja →  "ida" (alejándose)
   766→   766→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   767→   767→//
   768→   768→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   769→   769→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   770→   770→
   771→   771→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   772→   772→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   773→   773→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   774→   774→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   775→   775→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   776→   776→  return false;
   777→   777→}

   779→   779→// Obtener precio considerando la dirección del viaje
   780→   780→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   781→   781→  if (viajaHaciaLoja(ruta, tipo)) {
   782→   782→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   783→   783→    if (ruta === 'Loja - El Tambo') {
   784→   784→      const tamboVuelta = preciosVueltaElTambo[parada];
   785→   785→      if (tamboVuelta) return tamboVuelta;
   786→   786→    }
   787→   787→    // Ruta Zahuayco tiene precios de vuelta propios (directos desde Zahuayco)
   788→   788→    if (ruta === 'Loja - Zahuayco') {
   789→   789→      const zahuaycoVuelta = preciosVueltaZahuayco[parada];
   790→   790→      if (zahuaycoVuelta) return zahuaycoVuelta;
   791→   791→    }
   792→   792→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   793→   793→    if (ruta === 'Loja - Yangana') {
   794→   794→      const yangVuelta = preciosVueltaYangana[parada];
   795→   795→      if (yangVuelta) return yangVuelta;
   796→   796→    }
   797→   797→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   798→   798→    if (ruta === 'Loja - Vilcabamba') {
   799→   799→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   800→   800→      if (vilcVuelta) return vilcVuelta;
   801→   801→    }
   802→   802→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   803→   803→    if (ruta === 'Loja - La Elvira') {
   804→   804→      const elviraVuelta = preciosVueltaLaElvira[parada];
   805→   805→      if (elviraVuelta) return elviraVuelta;
   806→   806→    }
   807→   807→    const vuelta = preciosVuelta[parada];
   808→   808→    if (vuelta) return vuelta;
   809→   809→    // Si la parada no tiene precio de vuelta, buscar en precios base
   810→   810→  }
   811→   811→  // Buscar primero en ida (base), luego en vuelta como fallback
   812→   812→  const ida = preciosIda[parada];
   813→   813→  if (ida) return ida;
   814→   814→  const vuelta = preciosVuelta[parada];
   815→   815→  return vuelta || null;
   816→   816→}

   821→   821→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   822→   822→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   823→   823→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   824→   824→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   825→   825→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   826→   826→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   827→   827→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   828→   828→  let effectiveRuta = ruta;
   829→   829→  let effectiveTipo = tipo;
   830→   830→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   831→   831→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   832→   832→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   833→   833→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   834→   834→    const baseName = ruta.replace(' - Loja', '');
   835→   835→    effectiveRuta = `Loja - ${baseName}`;
   836→   836→    // NO invertir effectiveTipo — ya viene correcto del API
   837→   837→  }
   838→   838→
   839→   839→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   840→   840→  if (!rutaConfig) {
   841→   841→    // Fallback: try original ruta
   842→   842→    const origConfig = RUTA_PARADAS[ruta];
   843→   843→    if (!origConfig) return [];
   844→   844→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   845→   845→    return paradas.map(parada => {
   846→   846→      const precio = getPrecio(parada, ruta, tipo);
   847→   847→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   848→   848→    });
   849→   849→  }
   850→   850→
   851→   851→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   852→   852→  return paradas.map(parada => {
   853→   853→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   854→   854→    return {
   855→   855→      parada,
   856→   856→      normal: precio?.normal ?? 0,
   857→   857→      media: precio?.media ?? 0,
   858→   858→    };
   859→   859→  });
   860→   860→}

   863→   863→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   864→   864→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   865→   865→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   866→   866→    // El API ya asigna tipo correcto, NO invertir
   867→   867→    const baseName = ruta.replace(' - Loja', '');
   868→   868→    return { ruta: `Loja - ${baseName}`, tipo };
   869→   869→  }
   870→   870→  return { ruta, tipo };

   874→   874→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {

   886→   886→// Buscar la ruta más cercana dado un nombre de ruta parcial
   887→   887→export function matchRuta(rutaNombre: string): string {
   888→   888→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   889→   889→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   890→   890→  if (RUTA_PARADAS[normalized]) return normalized;
   891→   891→
   892→   892→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   893→   893→  for (const key of Object.keys(RUTA_PARADAS)) {
   894→   894→    const keyParts = key.split(' - ');
   895→   895→    if (parts.length >= 2 && keyParts.length >= 2) {
   896→   896→      const match = (
   897→   897→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   898→   898→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   899→   899→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   900→   900→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   901→   901→      );
   902→   902→      if (match) return key;
   903→   903→    }
   904→   904→  }
   905→   905→  return rutaNombre;
   906→   906→}
