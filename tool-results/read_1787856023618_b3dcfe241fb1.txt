     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo, useRef } from 'react';
     4→import { type VTSession, type FrecuenciaEstado } from './types-boletos';
     5→import { getVentasByFrecuencia } from '@/lib/indexeddb';
     6→import { type ConnectionInfo } from '@/hooks/use-connection';
     7→import {
     8→  ChevronLeft, DollarSign, Camera, X, Save, Loader2,
     9→  CheckCircle2, Send, AlertTriangle, Plus, Trash2, ImagePlus
    10→} from 'lucide-react';
    11→
    12→interface Props {
    13→  session: VTSession;
    14→  connection: ConnectionInfo;
    15→  onClose: () => void;
    16→  onGoToSync: () => void;
    17→  onSaved: () => void;
    18→}
    19→
    20→interface FrecuenciaResumen {
    21→  estadoId: string;
    22→  nombre: string;
    23→  ruta: string;
    24→  hora: string;
    25→  direccion: string;
    26→  ventasCount: number;
    27→  totalRecaudado: number;        // What the system says
    28→  efectivoContado: number;        // What the helper actually counted
    29→  diferencia: number;             // efectivoContado - totalRecaudado
    30→  boletosCaja: number;
    31→  cajaComunMonto?: number;
    32→}
    33→
    34→interface GastoItem {
    35→  description: string;
    36→  amount: string;
    37→}
    38→
    39→const today = () => new Date().toISOString().split('T')[0];
    40→
    41→// Work date: use session.fecha (date when shift started, not necessarily today)
    42→function workDate(session: VTSession): string {
    43→  return session.fecha || today();
    44→}
    45→
    46→const GASTOS_DEFAULT: GastoItem[] = [
    47→  { description: 'Chofer', amount: '30' },
    48→  { description: 'Ayudante', amount: '20' },
    49→  { description: 'Diesel', amount: '0' },
    50→  { description: 'Plan Renova', amount: '68' },
    51→];
    52→
    53→export function ArqueoGeneralScreen({ session, connection, onClose, onGoToSync, onSaved }: Props) {
    54→  const [frecuencias, setFrecuencias] = useState<FrecuenciaResumen[]>([]);
    55→  const [loading, setLoading] = useState(true);
    56→  const [km, setKm] = useState('');
    57→  const [gastos, setGastos] = useState<GastoItem[]>(GASTOS_DEFAULT);
    58→  const [tickets, setTickets] = useState('');
    59→  const [boletosCajaTotal, setBoletosCajaTotal] = useState('');
    60→  const [sobrante, setSobrante] = useState('');
    61→  const [fotoPreview, setFotoPreview] = useState<string | null>(null);
    62→  const [saving, setSaving] = useState(false);
    63→  const [confirmSave, setConfirmSave] = useState(false);
    64→  const [saved, setSaved] = useState(false);
    65→  const [errors, setErrors] = useState<string[]>([]);
    66→  const [guardadoOffline, setGuardadoOffline] = useState(false);
    67→
    68→  const fileInputRef = useRef<HTMLInputElement>(null);
    69→  const loadData = useCallback(async () => {
    70→    try {
    71→      const fecha = workDate(session);
    72→      // Load estados from localStorage (synchronous, reliable)
    73→      const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
    74→      const raw = localStorage.getItem(lsKey);
    75→      if (!raw) { setLoading(false); return; }
    76→      const estadosLS = JSON.parse(raw) as FrecuenciaEstado[];
    77→      const cerradas = estadosLS.filter(e => e.estado === 'cerrada' || e.estado === 'no_realizada');
    78→
    79→      const resumenes: FrecuenciaResumen[] = await Promise.all(
    80→        cerradas.map(async (e) => {
    81→          const ventas = await getVentasByFrecuencia(e.estadoId);
    82→          const sistemaTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
    83→          // Use arqueo data if available (what helper actually counted), else system total
    84→          const efectivo = (e as any).arqueoEfectivo ?? sistemaTotal;
    85→          const diff = efectivo - sistemaTotal;
    86→          const ccCount = (e as any).cajaComunCount || 0;
    87→          const ccMonto = (e as any).cajaComunMonto || 0;
    88→          return {
    89→            estadoId: e.estadoId,
    90→            nombre: e.nombre,
    91→            ruta: e.ruta,
    92→            hora: e.hora,
    93→            direccion: e.direccion,
    94→            ventasCount: ventas.length,
    95→            totalRecaudado: sistemaTotal,
    96→            efectivoContado: efectivo,
    97→            diferencia: diff,
    98→            boletosCaja: ccCount,
    99→            cajaComunMonto: ccMonto,
   100→          };
   101→        })
   102→      );
   103→      setFrecuencias(resumenes);
   104→    } catch (err) {
   105→      console.error('Error cargando resumen:', err);
   106→    } finally {
   107→      setLoading(false);
   108→    }
   109→  }, [session.vtCode]);
   110→
   111→  useEffect(() => { loadData(); }, [loadData]);
   112→
   113→  const totalIngresosAuto = useMemo(() =>
   114→    frecuencias.reduce((s, f) => s + f.totalRecaudado, 0),
   115→    [frecuencias]
   116→  );
   117→  const totalEfectivoReal = useMemo(() =>
   118→    frecuencias.reduce((s, f) => s + f.efectivoContado, 0),
   119→    [frecuencias]
   120→  );
   121→  const totalDiferencia = useMemo(() =>
   122→    frecuencias.reduce((s, f) => s + f.diferencia, 0),
   123→    [frecuencias]
   124→  );
   125→  const totalVentas = useMemo(() =>
   126→    frecuencias.reduce((s, f) => s + f.ventasCount, 0),
   127→    [frecuencias]
   128→  );
   129→  const totalCajaComunPasajeros = useMemo(() =>
   130→    frecuencias.reduce((s, f) => s + f.boletosCaja, 0),
   131→    [frecuencias]
   132→  );
   133→  const totalCajaComunMonto = useMemo(() =>
   134→    frecuencias.reduce((s, f) => s + (f.cajaComunMonto || 0), 0),
   135→    [frecuencias]
   136→  );
   137→  const totalGastos = useMemo(() =>
   138→    gastos.reduce((s, g) => s + (parseFloat(g.amount) || 0), 0),
   139→    [gastos]
   140→  );
   141→  const ticketsNum = parseFloat(tickets) || 0;
   142→  const boletosCajaNum = parseFloat(boletosCajaTotal) || 0;
   143→  const sobranteNum = parseFloat(sobrante) || 0;
   144→  // PRODUCCION = efectivo real contado por el ayudante + caja comun + sobrante ajuste manual
   145→  const production = totalEfectivoReal + totalCajaComunMonto + sobranteNum;
   146→  const entregaAyudante = production - totalGastos;
   147→  const entregaCompania = boletosCajaNum - ticketsNum;
   148→
   149→  const addGasto = () => setGastos(prev => [...prev, { description: '', amount: '0' }]);
   150→  const removeGasto = (i: number) => setGastos(prev => prev.filter((_, idx) => idx !== i));
   151→  const updateGasto = (i: number, field: keyof GastoItem, value: string) => {
   152→    setGastos(prev => {
   153→      const next = [...prev];
   154→      next[i] = { ...next[i], [field]: value };
   155→      return next;
   156→    });
   157→  };
   158→
   159→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   160→    const file = e.target.files?.[0];
   161→    if (!file || !file.type.startsWith('image/')) return;
   162→    const reader = new FileReader();
   163→    reader.onload = (ev) => setFotoPreview(ev.target?.result as string);
   164→    reader.readAsDataURL(file);
   165→    e.target.value = '';
   166→  };
   167→
   168→  const handleSaveOffline = () => {
   169→    const errs: string[] = [];
   170→    if (!km.trim()) errs.push('Kilometraje es obligatorio');
   171→    if (!fotoPreview) errs.push('Foto del cuaderno es obligatoria');
   172→    if (frecuencias.length === 0) errs.push('No hay frecuencias cerradas');
   173→    if (totalGastos <= 0) errs.push('Total de gastos debe ser mayor a 0');
   174→    if (errs.length > 0) {
   175→      setErrors(errs);
   176→      return;
   177→    }
   178→    setErrors([]);
   179→    setConfirmSave(true);
   180→  };
   181→
   182→  const handleConfirmSave = async () => {
   183→    setConfirmSave(false);
   184→    setSaving(true);
   185→
   186→    try {
   187→      // Build trips from frecuencias
   188→      const trips = frecuencias.map(f => ({
   189→        routeFrom: f.direccion === 'ida' ? 'Loja' : f.ruta.split(' - ')[1]?.trim() || 'Loja',
   190→        routeTo: f.direccion === 'ida' ? f.ruta.split(' - ')[1]?.trim() || 'Vilcabamba' : 'Loja',
   191→        time: f.hora,
   192→        income: f.totalRecaudado.toString(),
   193→        efectivoReal: f.efectivoContado.toString(),
   194→        boletos: '0',
   195→        cajaComunPasajeros: f.boletosCaja.toString(),
   196→        cajaComunMonto: (f.cajaComunMonto || 0).toString(),
   197→      }));
   198→
   199→      const body = {
   200→        date: workDate(session),
   201→        km: km,
   202→        conductor: '',
   203→        ayudanteNombre: session.ayudanteNombre,
   204→        vtCode: session.vtCode,
   205→        trips,
   206→        expenses: gastos,
   207→        tickets: tickets || '0',
   208→        sobrante: sobrante || '0',
   209→        photoUrl: fotoPreview,
   210→      };
   211→
   212→      const fechaTrabajo = workDate(session);
   213→      const isOnline = navigator.onLine;
   214→
   215→      if (isOnline) {
   216→        const res = await fetch('/api/records', {
   217→          method: 'POST',
   218→          headers: { 'Content-Type': 'application/json' },
   219→          body: JSON.stringify(body),
   220→        });
   221→        if (res.ok) {
   222→          setSaved(true);
   223→        } else {
   224→          const result = await res.json().catch(() => ({ error: 'Error desconocido' }));
   225→          console.error('Error guardando arqueo en servidor:', result.error);
   226→          // Save offline if server fails
   227→          localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   228→          setGuardadoOffline(true);
   229→          setSaved(true);
   230→        }
   231→      } else {
   232→        // Save to localStorage for later sync
   233→        localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   234→        setGuardadoOffline(true);
   235→        setSaved(true);
   236→      }
   237→    } catch (err) {
   238→      console.error('Error guardando arqueo:', err);
   239→      localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify({
   240→        date: fechaTrabajo, km, vtCode: session.vtCode, ayudanteNombre: session.ayudanteNombre,
   241→        trips: frecuencias.map(f => ({ routeFrom: 'Loja', routeTo: 'Vilcabamba', time: f.hora, income: f.totalRecaudado.toString(), boletos: '0' })),
   242→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   243→      }));
   244→      setGuardadoOffline(true);
   245→      setSaved(true);
   246→    } finally {
   247→      setSaving(false);
   248→    }
   249→  };
   250→
   251→  if (loading) {
   252→    return (
   253→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   254→        <Loader2 className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   255→        <p className="text-[#3A3A3A]">Cargando arqueo general...</p>
   256→      </div>
   257→    );
   258→  }
   259→
   260→  // Saved confirmation screen
   261→  if (saved) {
   262→    return (
   263→      <div className="flex flex-col min-h-[100dvh] bg-[#912D26] text-white">
   264→        <div className="flex-1 flex flex-col items-center justify-center px-6">
   265→          <CheckCircle2 className="w-20 h-20 mb-4" />
   266→          <h1 className="text-2xl font-black mb-2">ARQUEO GENERAL GUARDADO</h1>
   267→          <p className="text-white/80 text-sm mb-1">{session.nombre} — {workDate(session)}</p>
   268→          <p className="text-white/80 text-sm mb-6">{totalVentas} ventas · ${totalIngresosAuto.toFixed(2)} ingresos</p>
   269→
   270→          {guardadoOffline && (
   271→            <div className="bg-yellow-500/20 border border-yellow-400/30 rounded-2xl px-5 py-3 mb-6 text-center">
   272→              <p className="text-yellow-200 text-sm font-bold">Guardado local (sin internet)</p>
   273→              <p className="text-yellow-200/70 text-xs mt-1">Se sincronizará cuando haya conexión</p>
   274→            </div>
   275→          )}
   276→
   277→          <div className="w-full max-w-xs space-y-3">
   278→            {connection.pendingCount > 0 && connection.status !== 'offline' && (
   279→              <button onClick={onGoToSync}
   280→                className="w-full py-4 rounded-2xl bg-white text-[#912D26] font-black text-lg flex items-center justify-center gap-2 active:scale-95">
   281→                <Send className="w-5 h-5" /> SINCRONIZAR {connection.pendingCount} VENTAS
   282→              </button>
   283→            )}
   284→            <button onClick={onSaved}
   285→              className="w-full py-4 rounded-2xl bg-white/20 text-white font-bold text-lg flex items-center justify-center gap-2 active:scale-95">
   286→              FINALIZAR
   287→            </button>
   288→          </div>
   289→        </div>
   290→      </div>
   291→    );
   292→  }
   293→
   294→  // Confirm save dialog
   295→  if (confirmSave) {
   296→    return (
   297→      <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   298→        <div className="flex-1 flex items-center justify-center px-4">
   299→          <div className="bg-white rounded-2xl border border-gray-200 p-6 max-w-md w-full shadow-lg">
   300→            <h2 className="text-lg font-bold text-[#3A3A3A] mb-4">Confirmar Arqueo General</h2>
   301→            <div className="bg-gray-50 rounded-xl p-4 space-y-2 mb-4">
   302→              <div className="flex justify-between text-sm">
   303→                <span className="text-gray-500">Producción</span>
   304→                <span className="font-bold text-[#3A3A3A]">${production.toFixed(2)}</span>
   305→              </div>
   306→              {totalCajaComunPasajeros > 0 && (
   307→                <div className="flex justify-between text-sm">
   308→                  <span className="text-purple-700">Caja Comun (Ofi.Loja): {totalCajaComunPasajeros} boletos</span>
   309→                  <span className="font-bold text-purple-700">${totalCajaComunMonto.toFixed(2)}</span>
   310→                </div>
   311→              )}
   312→              <div className="flex justify-between text-sm">
   313→                <span className="text-gray-500">Total Gastos</span>
   314→                <span className="font-bold text-red-600">${totalGastos.toFixed(2)}</span>
   315→              </div>
   316→              <div className="flex justify-between text-sm">
   317→                <span className="text-gray-500">Boletos (Caja Común)</span>
   318→                <span className="font-bold text-[#3A3A3A]">${boletosCajaNum.toFixed(2)}</span>
   319→              </div>
   320→              <hr className="border-gray-200" />
   321→              <div className="flex justify-between text-sm">
   322→                <span className="text-gray-600">Entrega Ayudante</span>
   323→                <span className={`font-bold ${entregaAyudante >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   324→                  ${entregaAyudante.toFixed(2)}
   325→                </span>
   326→              </div>
   327→              <div className="flex justify-between text-sm">
   328→                <span className="text-gray-600">Entrega Compañía</span>
   329→                <span className={`font-bold ${entregaCompania >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   330→                  ${entregaCompania.toFixed(2)}
   331→                </span>
   332→              </div>
   333→            </div>
   334→            <p className="text-sm text-[#912D26] mb-4">
   335→              Entrega Ayudante: <strong>${entregaAyudante.toFixed(2)}</strong> —
   336→              Entrega Compañía: <strong>${entregaCompania.toFixed(2)}</strong>
   337→            </p>
   338→            <div className="flex gap-3">
   339→              <button onClick={() => setConfirmSave(false)}
   340→                className="flex-1 py-3 rounded-xl border-2 border-gray-200 text-gray-500 font-bold active:scale-95">
   341→                Cancelar
   342→              </button>
   343→              <button onClick={handleConfirmSave}
   344→                className="flex-1 py-3 rounded-xl bg-[#912D26] text-white font-black active:scale-95">
   345→                CONFIRMAR
   346→              </button>
   347→            </div>
   348→          </div>
   349→        </div>
   350→      </div>
   351→    );
   352→  }
   353→
   354→  return (
   355→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   356→      {/* Barra conexion */}
   357→      <div className={`${connection.bgColor} px-4 py-1.5 flex items-center justify-center gap-2 text-xs font-medium ${connection.color}`}>
   358→        <span>{connection.icon}</span>
   359→        <span>{connection.label}</span>
   360→      </div>
   361→
   362→      {/* Header */}
   363→      <div className="bg-[#912D26] text-white px-4 py-3">
   364→        <div className="flex items-center justify-between">
   365→          <button onClick={onClose} className="text-red-100"><ChevronLeft className="w-6 h-6" /></button>
   366→          <div className="text-center">
   367→            <h2 className="text-lg font-bold">Arqueo General</h2>
   368→            <p className="text-red-100 text-xs">{session.nombre} — Turno completo</p>
   369→          </div>
   370→          <div className="w-6" />
   371→        </div>
   372→      </div>
   373→
   374→      <div className="flex-1 px-4 py-4 space-y-4 overflow-y-auto pb-8">
   375→        {/* Errores */}
   376→        {errors.length > 0 && (
   377→          <div className="p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm space-y-1">
   378→            <p className="font-semibold flex items-center gap-1.5"><AlertTriangle className="w-4 h-4" /> Campos obligatorios:</p>
   379→            {errors.map((e, i) => <p key={i}>- {e}</p>)}
   380→          </div>
   381→        )}
   382→
   383→        {/* 1. INGRESOS POR FRECUENCIA — Sistema vs Efectivo Real */}
   384→        <div className="bg-white rounded-2xl border border-gray-100 p-4 shadow-sm">
   385→          <h3 className="font-bold text-[#3A3A3A] mb-3 text-sm uppercase flex items-center gap-2">
   386→            <DollarSign className="w-4 h-4 text-green-600" /> Ingresos por Frecuencia
   387→          </h3>
   388→          {frecuencias.length === 0 ? (
   389→            <p className="text-gray-400 text-xs text-center py-4">No hay frecuencias cerradas</p>
   390→          ) : (
   391→            <div className="space-y-2">
   392→              {/* Header row */}
   393→              <div className="flex items-center justify-between px-3 py-1 text-[10px] font-bold text-gray-400 uppercase">
   394→                <div className="flex-1">Frecuencia</div>
   395→                <div className="w-20 text-right">Sistema</div>
   396→                <div className="w-20 text-right">Real</div>
   397→                <div className="w-16 text-right">Dif.</div>
   398→              </div>
   399→              {frecuencias.map((f, i) => {
   400→                const hasDiff = Math.abs(f.diferencia) >= 0.01;
   401→                return (
   402→                  <div key={f.estadoId} className={`flex items-center justify-between py-2 px-3 rounded-xl ${i % 2 === 0 ? 'bg-[#912D26]/5' : 'bg-gray-50'}`}>
   403→                    <div className="flex items-center gap-2 flex-1 min-w-0">
   404→                      <span className={`text-[10px] font-bold px-1.5 py-0.5 rounded flex-shrink-0 ${i % 2 === 0 ? 'bg-[#912D26] text-white' : 'bg-[#3A3A3A] text-white'}`}>
   405→                        {i + 1}
   406→                      </span>
   407→                      <div className="min-w-0">
   408→                        <div className="text-sm font-semibold text-[#3A3A3A] truncate">{f.hora} — {f.nombre}</div>
   409→                        <div className="text-[10px] text-gray-400">{f.ventasCount} ventas{f.boletosCaja > 0 ? ` + ${f.boletosCaja} c.comun` : ''}</div>
   410→                      </div>
   411→                    </div>
   412→                    <div className="w-20 text-right">
   413→                      <div className="font-bold text-[#3A3A3A]">${f.totalRecaudado.toFixed(2)}</div>
   414→                    </div>
   415→                    <div className="w-20 text-right">
   416→                      <div className="font-black text-green-700">${f.efectivoContado.toFixed(2)}</div>
   417→                    </div>
   418→                    <div className="w-16 text-right">
   419→                      {hasDiff ? (
   420→                        <span className={`text-xs font-bold ${f.diferencia > 0 ? 'text-blue-600' : 'text-red-600'}`}>
   421→                          {f.diferencia > 0 ? '+' : ''}{f.diferencia.toFixed(2)}
   422→                        </span>
   423→                      ) : (
   424→                        <span className="text-xs text-gray-400">—</span>
   425→                      )}
   426→                    </div>
   427→                  </div>
   428→                );
   429→              })}
   430→              {/* Totals row */}
   431→              <div className="border-t-2 border-green-200 pt-2 mt-2 space-y-1">
   432→                <div className="flex justify-between items-center">
   433→                  <span className="font-bold text-[#3A3A3A] text-xs">Total Sistema</span>
   434→                  <span className="font-bold text-sm text-[#3A3A3A]">${totalIngresosAuto.toFixed(2)}</span>
   435→                </div>
   436→                <div className="flex justify-between items-center">
   437→                  <span className="font-bold text-green-700 text-xs">Total Efectivo Real</span>
   438→                  <span className="font-black text-lg text-green-700">${totalEfectivoReal.toFixed(2)}</span>
   439→                </div>
   440→                {Math.abs(totalDiferencia) >= 0.01 && (
   441→                  <div className="flex justify-between items-center">
   442→                    <span className="text-xs text-gray-500">Diferencia global</span>
   443→                    <span className={`text-xs font-bold ${totalDiferencia > 0 ? 'text-blue-600' : 'text-red-600'}`}>
   444→                      {totalDiferencia > 0 ? '+' : ''}{totalDiferencia.toFixed(2)}
   445→                    </span>
   446→                  </div>
   447→                )}
   448→                {totalCajaComunPasajeros > 0 && (
