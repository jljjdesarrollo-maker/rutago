   478→   478→// ─── Funciones de precio con soporte por dirección ───
   479→   479→//
   480→   480→// Lógica de precios (tipo semántico asignado por el API):
   481→   481→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   482→   482→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   483→   483→//
   484→   484→// El API asigna dirección según el origen:
   485→   485→//   Si origen = Loja →  "ida" (alejándose)
   486→   486→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   487→   487→//
   488→   488→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   489→   489→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   490→   490→
   491→   491→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   492→   492→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   493→   493→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   494→   494→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   495→   495→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   496→   496→  return false;
   497→   497→}

   499→   499→// Obtener precio considerando la dirección del viaje
   500→   500→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   501→   501→  if (viajaHaciaLoja(ruta, tipo)) {
   502→   502→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   503→   503→    if (ruta === 'Loja - El Tambo') {
   504→   504→      const tamboVuelta = preciosVueltaElTambo[parada];
   505→   505→      if (tamboVuelta) return tamboVuelta;
   506→   506→    }
   507→   507→    const vuelta = preciosVuelta[parada];
   508→   508→    if (vuelta) return vuelta;
   509→   509→    // Si la parada no tiene precio de vuelta, buscar en precios base
   510→   510→  }
   511→   511→  // Buscar primero en ida (base), luego en vuelta como fallback
   512→   512→  const ida = preciosIda[parada];
   513→   513→  if (ida) return ida;
   514→   514→  const vuelta = preciosVuelta[parada];
   515→   515→  return vuelta || null;
   516→   516→}

   521→   521→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   522→   522→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   523→   523→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   524→   524→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   525→   525→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   526→   526→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   527→   527→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   528→   528→  let effectiveRuta = ruta;
   529→   529→  let effectiveTipo = tipo;
   530→   530→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   531→   531→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   532→   532→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   533→   533→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   534→   534→    const baseName = ruta.replace(' - Loja', '');
   535→   535→    effectiveRuta = `Loja - ${baseName}`;
   536→   536→    // NO invertir effectiveTipo — ya viene correcto del API
   537→   537→  }
   538→   538→
   539→   539→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   540→   540→  if (!rutaConfig) {
   541→   541→    // Fallback: try original ruta
   542→   542→    const origConfig = RUTA_PARADAS[ruta];
   543→   543→    if (!origConfig) return [];
   544→   544→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   545→   545→    return paradas.map(parada => {
   546→   546→      const precio = getPrecio(parada, ruta, tipo);
   547→   547→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   548→   548→    });
   549→   549→  }
   550→   550→
   551→   551→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   552→   552→  return paradas.map(parada => {
   553→   553→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   554→   554→    return {
   555→   555→      parada,
   556→   556→      normal: precio?.normal ?? 0,
   557→   557→      media: precio?.media ?? 0,
   558→   558→    };
   559→   559→  });
   560→   560→}

   563→   563→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   564→   564→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   565→   565→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   566→   566→    // El API ya asigna tipo correcto, NO invertir
   567→   567→    const baseName = ruta.replace(' - Loja', '');
   568→   568→    return { ruta: `Loja - ${baseName}`, tipo };
   569→   569→  }
   570→   570→  return { ruta, tipo };
   571→   571→}

   574→   574→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {

   586→   586→// Buscar la ruta más cercana dado un nombre de ruta parcial
   587→   587→export function matchRuta(rutaNombre: string): string {
   588→   588→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   589→   589→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   590→   590→  if (RUTA_PARADAS[normalized]) return normalized;
   591→   591→
   592→   592→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   593→   593→  for (const key of Object.keys(RUTA_PARADAS)) {
   594→   594→    const keyParts = key.split(' - ');
   595→   595→    if (parts.length >= 2 && keyParts.length >= 2) {
   596→   596→      const match = (
   597→   597→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   598→   598→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   599→   599→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   600→   600→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   601→   601→      );
   602→   602→      if (match) return key;
   603→   603→    }
   604→   604→  }
   605→   605→  return rutaNombre;
   606→   606→}
