   590→// ─── Funciones de precio con soporte por dirección ───
   591→//
   592→// Lógica de precios (tipo semántico asignado por el API):
   593→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   594→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   595→//
   596→// El API asigna dirección según el origen:
   597→//   Si origen = Loja →  "ida" (alejándose)
   598→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   599→//
   600→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   601→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   602→
   603→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   604→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   605→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   606→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   607→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   608→  return false;
   609→}
   610→
   611→// Obtener precio considerando la dirección del viaje
   612→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   613→  if (viajaHaciaLoja(ruta, tipo)) {
   614→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   615→    if (ruta === 'Loja - El Tambo') {
   616→      const tamboVuelta = preciosVueltaElTambo[parada];
   617→      if (tamboVuelta) return tamboVuelta;
   618→    }
   619→    const vuelta = preciosVuelta[parada];
   620→    if (vuelta) return vuelta;
   621→    // Si la parada no tiene precio de vuelta, buscar en precios base
   622→  }
   623→  // Buscar primero en ida (base), luego en vuelta como fallback
   624→  const ida = preciosIda[parada];
   625→  if (ida) return ida;
   626→  const vuelta = preciosVuelta[parada];
   627→  return vuelta || null;
   628→}

   633→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   634→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   635→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   636→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   637→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   638→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   639→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   640→  let effectiveRuta = ruta;
   641→  let effectiveTipo = tipo;
   642→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   643→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   644→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   645→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   646→    const baseName = ruta.replace(' - Loja', '');
   647→    effectiveRuta = `Loja - ${baseName}`;
   648→    // NO invertir effectiveTipo — ya viene correcto del API
   649→  }
   650→
   651→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   652→  if (!rutaConfig) {
   653→    // Fallback: try original ruta
   654→    const origConfig = RUTA_PARADAS[ruta];
   655→    if (!origConfig) return [];
   656→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   657→    return paradas.map(parada => {
   658→      const precio = getPrecio(parada, ruta, tipo);
   659→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   660→    });
   661→  }
   662→
   663→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   664→  return paradas.map(parada => {
   665→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   666→    return {
   667→      parada,
   668→      normal: precio?.normal ?? 0,
   669→      media: precio?.media ?? 0,
   670→    };
   671→  });
   672→}

   674→// Normalizar ruta y tipo para rutas "X - Loja"
   675→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   676→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   677→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   678→    // El API ya asigna tipo correcto, NO invertir
   679→    const baseName = ruta.replace(' - Loja', '');
   680→    return { ruta: `Loja - ${baseName}`, tipo };
   681→  }
   682→  return { ruta, tipo };
   683→}

   686→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   687→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);

   689→  if (!precio) return TARIFA_MINIMA;

   698→// Buscar la ruta más cercana dado un nombre de ruta parcial
   699→export function matchRuta(rutaNombre: string): string {
   700→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   701→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   702→  if (RUTA_PARADAS[normalized]) return normalized;
   703→
   704→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   705→  for (const key of Object.keys(RUTA_PARADAS)) {
   706→    const keyParts = key.split(' - ');
   707→    if (parts.length >= 2 && keyParts.length >= 2) {
   708→      const match = (
   709→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   710→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   711→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   712→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   713→      );
   714→      if (match) return key;
   715→    }
   716→  }
   717→  return rutaNombre;
   718→}
