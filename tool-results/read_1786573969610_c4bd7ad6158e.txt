    17→const today = () => new Date().toISOString().split('T')[0];
    18→
    19→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    20→
    21→function estadosKey(vtCode: string, fecha: string) {
    22→  return `rg_estados_${vtCode}_${fecha}`;
    23→}
    24→
    25→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    26→  try {
    27→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    28→  } catch (e) { console.error('Error saving estados to LS:', e); }
    29→}
    30→
    31→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    32→  try {
    33→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    34→    return raw ? JSON.parse(raw) : null;
    35→  } catch { return null; }
    36→}
    37→
    38→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    39→  const all = loadEstadosFromLS(vtCode, fecha);
    40→  if (!all) return;
    41→  const idx = all.findIndex(e => e.estadoId === estadoId);
    42→  if (idx >= 0) {
    43→    all[idx] = { ...all[idx], ...updates };
    44→    saveEstadosToLS(vtCode, fecha, all);
    45→  }
    46→}

    48→export function FrecuenciaSelector({ session, onOpenFrequency, onGoToArqueo, onGoToArqueoGeneral, onBack, onGoToSync }: Props) {
    49→  const [estados, setEstados] = useState<FrecuenciaEstado[]>([]);
    50→  const [frecuencias, setFrecuencias] = useState<FrecuenciaData[]>([]);
    51→  const [loading, setLoading] = useState(true);
    52→  const [reassigning, setReassigning] = useState<string | null>(null);
    53→  const [pendingCount, setPendingCount] = useState(0);
    54→  const [isOnline, setIsOnline] = useState(navigator.onLine);
    55→  const [noRealizadaModal, setNoRealizadaModal] = useState<FrecuenciaEstado | null>(null);
    56→  const [motivoSeleccionado, setMotivoSeleccionado] = useState('');

    59→  const [allFrecuencias, setAllFrecuencias] = useState<FrecuenciaData[]>([]);

    63→    const handle = () => setIsOnline(navigator.onLine);

    65→    window.addEventListener('offline', handle);

    72→      if (res.ok) {

    74→        setFrecuencias(data);

    76→        // Try loading saved estados from localStorage first
    77→        const saved = loadEstadosFromLS(session.vtCode, fecha);

    79→          setEstados(saved);
    80→        } else {
    81→          // Create fresh pendiente estados
    82→          const newEstados: FrecuenciaEstado[] = data.map((f: FrecuenciaData) => ({
    83→            id: f.id,
    84→            estadoId: `${fecha}_${f.id}`,
    85→            frecuenciaId: f.id,
    86→            nombre: f.nombre,
    87→            ruta: f.ruta,
    88→            hora: f.hora,
    89→            direccion: f.direccion,
    90→            estado: 'pendiente' as const,

    92→            totalRecaudado: 0,
    93→          }));
    94→          setEstados(newEstados);
    95→          saveEstadosToLS(session.vtCode, fecha, newEstados);
    96→        }
    97→      }
    98→    } catch (error) {
    99→      console.error('Error cargando frecuencias:', error);
   100→      // Fallback: try to restore from localStorage only
   101→      const saved = loadEstadosFromLS(session.vtCode, fecha);
   102→      if (saved) setEstados(saved);
   103→    } finally {
   104→      setLoading(false);
   105→    }
   106→  }, [session.vtCode, fecha]);
   107→

   109→    try {
   110→      const count = await countVentasPendientes();
   111→      setPendingCount(count);
   112→    } catch { /* ignore */ }
   113→  }, []);
   114→
   115→  // Reload ventas counts from IndexedDB periodically
   116→  const refreshVentasCounts = useCallback(async () => {
   117→    for (const estado of estados) {
   118→      try {
   119→        const ventas = await getVentasByFrecuencia(estado.estadoId);
   120→        // Count ALL ventas (including synced) for display in arqueo
   121→        const newCount = ventas.length;
   122→        const newTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   123→        if (newCount !== estado.ventasCount || Math.abs(newTotal - estado.totalRecaudado) > 0.01) {
   124→          updateEstadoInLS(session.vtCode, fecha, estado.estadoId, { ventasCount: newCount, totalRecaudado: newTotal });
   125→          setEstados(prev => prev.map(e =>
   126→            e.estadoId === estado.estadoId ? { ...e, ventasCount: newCount, totalRecaudado: newTotal } : e
   127→          ));
   128→        }
   129→      } catch { /* ignore */ }
   130→    }
   131→  }, [estados, session.vtCode, fecha]);
   132→
   133→  useEffect(() => {
   134→    loadFrecuencias();
   135→    loadPendingCount();
   136→    const interval = setInterval(() => { loadPendingCount(); }, 5000);
   137→    return () => clearInterval(interval);
   138→  }, [loadFrecuencias, loadPendingCount]);
   139→
   140→  // Refresh counts after initial load and periodically
   141→  useEffect(() => {
   142→    if (!loading && estados.length > 0) {
   143→      refreshVentasCounts();
   144→      const interval = setInterval(refreshVentasCounts, 5000);
   145→      return () => clearInterval(interval);
   146→    }

   148→
   149→  const updateEstado = (estadoId: string, updates: Partial<FrecuenciaEstado>) => {
   150→    // Update in-memory state
   151→    setEstados(prev => prev.map(e => e.estadoId === estadoId ? { ...e, ...updates } : e));
   152→    // Persist to localStorage (synchronous, always reliable)
   153→    updateEstadoInLS(session.vtCode, fecha, estadoId, updates);
   154→  };
   155→
   156→  // Sequential: frequency available if previous are all cerrada/no_realizada
   157→  const isFrecuenciaDisponible = (index: number): boolean => {
   158→    const estado = estados[index];

   161→    for (let i = 0; i < index; i++) {

   163→      if (prev && prev.estado !== 'cerrada' && prev.estado !== 'no_realizada') {
   164→        return false;
   165→      }

   168→  };

   172→    const { getGPSPosition } = await import('@/lib/gps');
   173→    const gps = await getGPSPosition();

   175→      estado: 'abierta',
   176→      ...(gps ? { gpsLatStart: gps.lat, gpsLngStart: gps.lng } : {}),
   177→    });
   178→    onOpenFrequency({ ...estado, estado: 'abierta' });
   179→  };
   180→

   183→    setMotivoSeleccionado('');
   184→    setMotivoPersonalizado('');
   185→    setConfirmNoRealizada(false);

   189→    if (!noRealizadaModal) return;

   191→    updateEstado(noRealizadaModal.estadoId, {

   194→      totalRecaudado: 0,
   195→      motivoNoRealizada: motivo || 'Sin especificar',
   196→    });

   198→  };
   199→

   212→        const fOrigen = fPartes[0]?.trim().toLowerCase();
   213→        return fOrigen === origen;

   226→        }

   235→      id: targetFrec.id, frecuenciaId: targetFrec.id,
   236→      nombre: targetFrec.nombre, ruta: targetFrec.ruta,
   237→      hora: targetFrec.hora, direccion: targetFrec.direccion,

   241→  };
   242→

   247→      case 'cerrada': return <CheckCircle2 className="w-4 h-4 text-blue-500" />;
   248→      case 'no_realizada': return <XCircle className="w-4 h-4 text-orange-500" />;

   250→    }
   251→  };
   252→
   253→  const getStateColor = (estado: string) => {
   254→    switch (estado) {
   255→      case 'pendiente': return 'bg-gray-100 border-gray-200';
   256→      case 'abierta': return 'bg-green-50 border-green-200';
   257→      case 'cerrada': return 'bg-blue-50 border-blue-200';
   258→      case 'no_realizada': return 'bg-orange-50 border-orange-200';
   259→      default: return 'bg-gray-100 border-gray-200';
   260→    }
   261→  };
   262→
   263→  const MOTIVOS_NO_REALIZADA = [
   264→    { id: 'daño_unidad', label: 'Daño en la unidad', icon: <Truck className="w-4 h-4" />, color: 'text-red-500' },
   265→    { id: 'mantenimiento', label: 'Mantenimiento', icon: <Wrench className="w-4 h-4" />, color: 'text-blue-500' },
   266→    { id: 'clima', label: 'Clima / Lluvia', icon: <Droplets className="w-4 h-4" />, color: 'text-cyan-500' },
   267→    { id: 'sin_pasajeros', label: 'Sin pasajeros', icon: <UserX className="w-4 h-4" />, color: 'text-purple-500' },
   268→    { id: 'problema_ruta', label: 'Problema en la ruta', icon: <AlertTriangle className="w-4 h-4" />, color: 'text-yellow-500' },
   269→    { id: 'orden_superior', label: 'Orden superior', icon: <Ban className="w-4 h-4" />, color: 'text-gray-600' },
   270→    { id: 'otro', label: 'Otro motivo', icon: <FileText className="w-4 h-4" />, color: 'text-gray-500' },
   271→  ];
   272→
   273→  const getEstadoBadge = (estado: string) => {
   274→    switch (estado) {
   275→      case 'pendiente': return <span className="px-2 py-0.5 rounded-full text-xs bg-gray-200 text-gray-600">Pendiente</span>;
   276→      case 'abierta': return <span className="px-2 py-0.5 rounded-full text-xs bg-green-200 text-green-700">Abierta</span>;
   277→      case 'cerrada': return <span className="px-2 py-0.5 rounded-full text-xs bg-blue-200 text-blue-700">Cerrada</span>;
   278→      case 'no_realizada': return <span className="px-2 py-0.5 rounded-full text-xs bg-orange-200 text-orange-700">No Realizada</span>;
   279→      default: return null;
   280→    }
   281→  };
   282→
   283→  if (loading) {
   284→    return (
   285→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   286→        <RefreshCw className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   287→        <p className="text-[#3A3A3A]">Cargando frecuencias...</p>
   288→      </div>
   289→    );
   290→  }
   291→
   292→  // Check if ALL frequencies are done (cerrada or no_realizada)
   293→  const allFrecuenciasDone = estados.length > 0 && estados.every(e => e.estado === 'cerrada' || e.estado === 'no_realizada');
   294→
   295→  // Totals across all frequencies
   296→  const totalVentasAll = estados.reduce((s, e) => s + e.ventasCount, 0);
   297→  const totalRecaudadoAll = estados.reduce((s, e) => s + e.totalRecaudado, 0);
   298→  const cerradasCount = estados.filter(e => e.estado === 'cerrada').length;
   299→
   300→  // SYNC control: if online with pending ventas, block arqueo
   301→  const mustSyncBeforeArqueo = isOnline && pendingCount > 0;
   302→
   303→  return (
   304→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   305→      {/* Header */}
   306→      <div className="bg-[#912D26] text-white px-4 py-3">
   307→        <div className="flex items-center justify-between mb-2">
   308→          <div className="flex items-center gap-2">
   309→            <button onClick={onBack} className="text-red-100"><ArrowLeft className="w-5 h-5" /></button>
   310→            <div>
   311→              <h1 className="text-lg font-bold">Frecuencias</h1>
   312→              <p className="text-red-100 text-xs">{session.nombre} - {session.ayudanteNombre}</p>
   313→            </div>
   314→          </div>

   317→          </div>
   318→        </div>
   319→        <div className="flex items-center gap-2">
   320→          <div className="flex-1 bg-white/15 rounded-xl px-3 py-1.5 text-sm flex items-center gap-2">

   322→              <>
   323→                <span className="bg-yellow-400 text-yellow-900 text-xs font-bold px-1.5 py-0.5 rounded-full">{pendingCount}</span>
   324→                <span className="text-red-100 text-xs">pendientes</span>
   325→              </>
   326→            ) : (

   332→              <Send className="w-3 h-3" /> SYNC
   333→            </button>
   334→          )}

   336→            <div className="bg-white/10 text-red-200 px-3 py-1.5 rounded-xl text-xs font-bold flex items-center gap-1">
   337→              <WifiOff className="w-3 h-3" /> Sin red
   338→            </div>
   339→          )}
   340→        </div>
   341→      </div>

   343→      {/* ─── SYNC OBLIGATORIO banner when online + pending ─── */}
   344→      {mustSyncBeforeArqueo && (
   345→        <div className="mx-4 mt-3 bg-gradient-to-r from-orange-500 to-yellow-500 rounded-2xl p-3 flex items-center gap-3 shadow-md">
   346→          <div className="w-10 h-10 bg-white/20 rounded-xl flex items-center justify-center flex-shrink-0">
   347→            <Wifi className="w-5 h-5 text-white" />
   348→          </div>
   349→          <div className="flex-1">
   350→            <p className="text-white font-black text-sm">SYNC OBLIGATORIO</p>
   351→            <p className="text-white/80 text-xs">Tienes {pendingCount} venta{pendingCount !== 1 ? 's' : ''} pendiente{pendingCount !== 1 ? 's' : ''}. Debes sincronizar antes de hacer el arqueo.</p>
   352→          </div>
   353→          <button onClick={onGoToSync} className="bg-white text-orange-600 font-black text-xs px-4 py-2.5 rounded-xl flex items-center gap-1.5 active:scale-95 shadow-sm flex-shrink-0">
   354→            <Send className="w-4 h-4" /> SYNC
   355→          </button>
   356→        </div>
   357→      )}
   358→
   359→      <div className="flex-1 px-4 py-3 space-y-2 overflow-y-auto pb-6">
   360→        <div className="text-center text-xs text-gray-500 mb-1">

   362→        </div>

   367→            <div className="bg-white rounded-t-3xl sm:rounded-2xl w-full max-w-md p-6 max-h-[80vh] overflow-y-auto">

   370→                Rutas desde <strong className="text-[#912D26]">{estados.find(e => e.estadoId === reassigning)?.ruta.split(' - ')[0]?.trim()}</strong>

   372→              <div className="space-y-2">

   459→                  Motivo: <strong>{motivoSeleccionado === 'otro' ? motivoPersonalizado.trim() : MOTIVOS_NO_REALIZADA.find(m => m.id === motivoSeleccionado)?.label}</strong>

   475→        {estados.length === 0 ? (

   543→                          <CheckCircle2 className="w-4 h-4" /> Arqueo
