    17→const today = () => new Date().toISOString().split('T')[0];
    18→
    19→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    20→
    21→function estadosKey(vtCode: string, fecha: string) {
    22→  return `rg_estados_${vtCode}_${fecha}`;
    23→}
    24→
    25→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    26→  try {
    27→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    28→  } catch (e) { console.error('Error saving estados to LS:', e); }
    29→}
    30→
    31→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    32→  try {
    33→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    34→    return raw ? JSON.parse(raw) : null;
    35→  } catch { return null; }
    36→}
    37→
    38→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    39→  const all = loadEstadosFromLS(vtCode, fecha);
    40→  if (!all) return;
    41→  const idx = all.findIndex(e => e.estadoId === estadoId);
    42→  if (idx >= 0) {
    43→    all[idx] = { ...all[idx], ...updates };
    44→    saveEstadosToLS(vtCode, fecha, all);
    45→  }
    46→}

   113→    } catch { /* ignore */ }
   114→  }, []);

   128→          ));
   129→        }
   130→      } catch { /* ignore */ }
   131→    }
