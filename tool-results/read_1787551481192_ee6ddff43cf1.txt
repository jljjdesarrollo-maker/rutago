   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→  return 'yellow';
   128→}
   129→
   130→export function isParadaPrincipal(parada: string): boolean {
   131→  // Paradas principales: no contienen → en el nombre
   132→  return !parada.includes('→');
   133→}

   683→// ─── Funciones de precio con soporte por dirección ───
   684→//
   685→// Lógica de precios (tipo semántico asignado por el API):
   686→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   687→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   688→//
   689→// El API asigna dirección según el origen:
   690→//   Si origen = Loja →  "ida" (alejándose)
   691→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   692→//
   693→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   694→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   695→
   696→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   697→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   698→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   699→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   700→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   701→  return false;
   702→}
   703→
   704→// Obtener precio considerando la dirección del viaje
   705→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   706→  if (viajaHaciaLoja(ruta, tipo)) {
   707→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   708→    if (ruta === 'Loja - El Tambo') {
   709→      const tamboVuelta = preciosVueltaElTambo[parada];
   710→      if (tamboVuelta) return tamboVuelta;
   711→    }
   712→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   713→    if (ruta === 'Loja - Yangana') {
   714→      const yangVuelta = preciosVueltaYangana[parada];
   715→      if (yangVuelta) return yangVuelta;
   716→    }
   717→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   718→    if (ruta === 'Loja - Vilcabamba') {
   719→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   720→      if (vilcVuelta) return vilcVuelta;
   721→    }
   722→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   723→    if (ruta === 'Loja - La Elvira') {
   724→      const elviraVuelta = preciosVueltaLaElvira[parada];
   725→      if (elviraVuelta) return elviraVuelta;
   726→    }
   727→    const vuelta = preciosVuelta[parada];
   728→    if (vuelta) return vuelta;
   729→    // Si la parada no tiene precio de vuelta, buscar en precios base
   730→  }
   731→  // Buscar primero en ida (base), luego en vuelta como fallback
   732→  const ida = preciosIda[parada];
   733→  if (ida) return ida;
   734→  const vuelta = preciosVuelta[parada];
   735→  return vuelta || null;
   736→}

   741→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   742→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   743→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   744→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   745→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   746→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   747→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   748→  let effectiveRuta = ruta;
   749→  let effectiveTipo = tipo;
   750→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   751→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   752→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   753→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   754→    const baseName = ruta.replace(' - Loja', '');
   755→    effectiveRuta = `Loja - ${baseName}`;
   756→    // NO invertir effectiveTipo — ya viene correcto del API
   757→  }
   758→
   759→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   760→  if (!rutaConfig) {
   761→    // Fallback: try original ruta
   762→    const origConfig = RUTA_PARADAS[ruta];
   763→    if (!origConfig) return [];
   764→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   765→    return paradas.map(parada => {
   766→      const precio = getPrecio(parada, ruta, tipo);
   767→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   768→    });
   769→  }
   770→
   771→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   772→  return paradas.map(parada => {
   773→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   774→    return {
   775→      parada,
   776→      normal: precio?.normal ?? 0,
   777→      media: precio?.media ?? 0,
   778→    };
   779→  });
   780→}

   782→// Normalizar ruta y tipo para rutas "X - Loja"
   783→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   784→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   785→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   786→    // El API ya asigna tipo correcto, NO invertir
   787→    const baseName = ruta.replace(' - Loja', '');
   788→    return { ruta: `Loja - ${baseName}`, tipo };
   789→  }
   790→  return { ruta, tipo };
   791→}

   793→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   794→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   795→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   796→  const precio = getPrecio(parada, nr, nt);
   797→  if (!precio) return TARIFA_MINIMA;
   798→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   799→}

   806→// Buscar la ruta más cercana dado un nombre de ruta parcial
   807→export function matchRuta(rutaNombre: string): string {
   808→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   809→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   810→  if (RUTA_PARADAS[normalized]) return normalized;
   811→
   812→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   813→  for (const key of Object.keys(RUTA_PARADAS)) {
   814→    const keyParts = key.split(' - ');
   815→    if (parts.length >= 2 && keyParts.length >= 2) {
   816→      const match = (
   817→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   818→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   819→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   820→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   821→      );
   822→      if (match) return key;
   823→    }
   824→  }
   825→  return rutaNombre;
   826→}
