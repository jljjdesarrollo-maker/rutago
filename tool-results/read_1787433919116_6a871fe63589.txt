   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   126→  return 'yellow';
   127→}

   129→export function isParadaPrincipal(parada: string): boolean {
   130→  // Paradas principales: no contienen → en el nombre
   131→  return !parada.includes('→');
   132→}

   619→// ─── Funciones de precio con soporte por dirección ───
   620→//
   621→// Lógica de precios (tipo semántico asignado por el API):
   622→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   623→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   624→//
   625→// El API asigna dirección según el origen:
   626→//   Si origen = Loja →  "ida" (alejándose)
   627→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   628→//
   629→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   630→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   631→
   632→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   633→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   634→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   635→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   636→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   637→  return false;
   638→}
   639→
   640→// Obtener precio considerando la dirección del viaje
   641→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   642→  if (viajaHaciaLoja(ruta, tipo)) {
   643→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   644→    if (ruta === 'Loja - El Tambo') {
   645→      const tamboVuelta = preciosVueltaElTambo[parada];
   646→      if (tamboVuelta) return tamboVuelta;
   647→    }
   648→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   649→    if (ruta === 'Loja - Yangana') {
   650→      const yangVuelta = preciosVueltaYangana[parada];
   651→      if (yangVuelta) return yangVuelta;
   652→    }
   653→    const vuelta = preciosVuelta[parada];
   654→    if (vuelta) return vuelta;
   655→    // Si la parada no tiene precio de vuelta, buscar en precios base
   656→  }
   657→  // Buscar primero en ida (base), luego en vuelta como fallback
   658→  const ida = preciosIda[parada];
   659→  if (ida) return ida;
   660→  const vuelta = preciosVuelta[parada];
   661→  return vuelta || null;
   662→}

   667→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   668→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   669→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   670→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   671→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   672→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   673→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   674→  let effectiveRuta = ruta;
   675→  let effectiveTipo = tipo;
   676→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   677→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   678→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   679→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   680→    const baseName = ruta.replace(' - Loja', '');
   681→    effectiveRuta = `Loja - ${baseName}`;
   682→    // NO invertir effectiveTipo — ya viene correcto del API
   683→  }
   684→
   685→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   686→  if (!rutaConfig) {
   687→    // Fallback: try original ruta
   688→    const origConfig = RUTA_PARADAS[ruta];
   689→    if (!origConfig) return [];
   690→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   691→    return paradas.map(parada => {
   692→      const precio = getPrecio(parada, ruta, tipo);
   693→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   694→    });
   695→  }
   696→
   697→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   698→  return paradas.map(parada => {
   699→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   700→    return {
   701→      parada,
   702→      normal: precio?.normal ?? 0,
   703→      media: precio?.media ?? 0,
   704→    };
   705→  });
   706→}

   708→// Normalizar ruta y tipo para rutas "X - Loja"
   709→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   710→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   711→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   712→    // El API ya asigna tipo correcto, NO invertir
   713→    const baseName = ruta.replace(' - Loja', '');
   714→    return { ruta: `Loja - ${baseName}`, tipo };
   715→  }
   716→  return { ruta, tipo };
   717→}

   719→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   720→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   721→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   722→  const precio = getPrecio(parada, nr, nt);
   723→  if (!precio) return TARIFA_MINIMA;
   724→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   725→}

   732→// Buscar la ruta más cercana dado un nombre de ruta parcial
   733→export function matchRuta(rutaNombre: string): string {
   734→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   735→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   736→  if (RUTA_PARADAS[normalized]) return normalized;
   737→
   738→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   739→  for (const key of Object.keys(RUTA_PARADAS)) {
   740→    const keyParts = key.split(' - ');
   741→    if (parts.length >= 2 && keyParts.length >= 2) {
   742→      const match = (
   743→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   744→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   745→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   746→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   747→      );
   748→      if (match) return key;
   749→    }
   750→  }
   751→  return rutaNombre;
   752→}
