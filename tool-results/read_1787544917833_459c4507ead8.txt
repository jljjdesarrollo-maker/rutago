   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→  return 'yellow';
   128→}
   129→
   130→export function isParadaPrincipal(parada: string): boolean {
   131→  // Paradas principales: no contienen → en el nombre
   132→  return !parada.includes('→');
   133→}

   701→// ─── Funciones de precio con soporte por dirección ───
   702→//
   703→// Lógica de precios (tipo semántico asignado por el API):
   704→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   705→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   706→//
   707→// El API asigna dirección según el origen:
   708→//   Si origen = Loja →  "ida" (alejándose)
   709→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   710→//
   711→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   712→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   713→
   714→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   715→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   716→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   717→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   718→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   719→  return false;
   720→}

   722→// Obtener precio considerando la dirección del viaje
   723→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   724→  if (viajaHaciaLoja(ruta, tipo)) {
   725→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   726→    if (ruta === 'Loja - El Tambo') {
   727→      const tamboVuelta = preciosVueltaElTambo[parada];
   728→      if (tamboVuelta) return tamboVuelta;
   729→    }
   730→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   731→    if (ruta === 'Loja - Yangana') {
   732→      const yangVuelta = preciosVueltaYangana[parada];
   733→      if (yangVuelta) return yangVuelta;
   734→    }
   735→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   736→    if (ruta === 'Loja - Vilcabamba') {
   737→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   738→      if (vilcVuelta) return vilcVuelta;
   739→    }
   740→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   741→    if (ruta === 'Loja - La Elvira') {
   742→      const elviraVuelta = preciosVueltaLaElvira[parada];
   743→      if (elviraVuelta) return elviraVuelta;
   744→    }
   745→    const vuelta = preciosVuelta[parada];
   746→    if (vuelta) return vuelta;
   747→    // Si la parada no tiene precio de vuelta, buscar en precios base
   748→  }
   749→  // Buscar primero en ida (base), luego en vuelta como fallback
   750→  const ida = preciosIda[parada];
   751→  if (ida) return ida;
   752→  const vuelta = preciosVuelta[parada];
   753→  return vuelta || null;
   754→}

   759→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   760→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   761→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   762→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   763→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   764→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   765→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   766→  let effectiveRuta = ruta;
   767→  let effectiveTipo = tipo;
   768→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   769→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   770→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   771→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   772→    const baseName = ruta.replace(' - Loja', '');
   773→    effectiveRuta = `Loja - ${baseName}`;
   774→    // NO invertir effectiveTipo — ya viene correcto del API
   775→  }
   776→
   777→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   778→  if (!rutaConfig) {
   779→    // Fallback: try original ruta
   780→    const origConfig = RUTA_PARADAS[ruta];
   781→    if (!origConfig) return [];
   782→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   783→    return paradas.map(parada => {
   784→      const precio = getPrecio(parada, ruta, tipo);
   785→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   786→    });
   787→  }
   788→
   789→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   790→  return paradas.map(parada => {
   791→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   792→    return {
   793→      parada,
   794→      normal: precio?.normal ?? 0,
   795→      media: precio?.media ?? 0,
   796→    };
   797→  });
   798→}

   800→// Normalizar ruta y tipo para rutas "X - Loja"
   801→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   802→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   803→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   804→    // El API ya asigna tipo correcto, NO invertir
   805→    const baseName = ruta.replace(' - Loja', '');
   806→    return { ruta: `Loja - ${baseName}`, tipo };
   807→  }
   808→  return { ruta, tipo };
   809→}

   811→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   812→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   813→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   814→  const precio = getPrecio(parada, nr, nt);
   815→  if (!precio) return TARIFA_MINIMA;
   816→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   817→}

   824→// Buscar la ruta más cercana dado un nombre de ruta parcial
   825→export function matchRuta(rutaNombre: string): string {
   826→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   827→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   828→  if (RUTA_PARADAS[normalized]) return normalized;
   829→
   830→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   831→  for (const key of Object.keys(RUTA_PARADAS)) {
   832→    const keyParts = key.split(' - ');
   833→    if (parts.length >= 2 && keyParts.length >= 2) {
   834→      const match = (
   835→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   836→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   837→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   838→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   839→      );
   840→      if (match) return key;
   841→    }
   842→  }
   843→  return rutaNombre;
   844→}
