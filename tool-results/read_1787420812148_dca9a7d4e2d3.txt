   552→   552→// ─── Funciones de precio con soporte por dirección ───
   553→   553→//
   554→   554→// Lógica de precios (tipo semántico asignado por el API):
   555→   555→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   556→   556→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   557→   557→//
   558→   558→// El API asigna dirección según el origen:
   559→   559→//   Si origen = Loja →  "ida" (alejándose)
   560→   560→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   561→   561→//
   562→   562→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   563→   563→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   564→   564→
   565→   565→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   566→   566→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   567→   567→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   568→   568→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   569→   569→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   570→   570→  return false;
   571→   571→}

   573→   573→// Obtener precio considerando la dirección del viaje
   574→   574→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   575→   575→  if (viajaHaciaLoja(ruta, tipo)) {
   576→   576→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   577→   577→    if (ruta === 'Loja - El Tambo') {
   578→   578→      const tamboVuelta = preciosVueltaElTambo[parada];
   579→   579→      if (tamboVuelta) return tamboVuelta;
   580→   580→    }
   581→   581→    const vuelta = preciosVuelta[parada];
   582→   582→    if (vuelta) return vuelta;
   583→   583→    // Si la parada no tiene precio de vuelta, buscar en precios base
   584→   584→  }
   585→   585→  // Buscar primero en ida (base), luego en vuelta como fallback
   586→   586→  const ida = preciosIda[parada];
   587→   587→  if (ida) return ida;
   588→   588→  const vuelta = preciosVuelta[parada];
   589→   589→  return vuelta || null;
   590→   590→}

   595→   595→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   596→   596→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   597→   597→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   598→   598→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   599→   599→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   600→   600→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   601→   601→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   602→   602→  let effectiveRuta = ruta;
   603→   603→  let effectiveTipo = tipo;
   604→   604→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   605→   605→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   606→   606→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   607→   607→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   608→   608→    const baseName = ruta.replace(' - Loja', '');
   609→   609→    effectiveRuta = `Loja - ${baseName}`;
   610→   610→    // NO invertir effectiveTipo — ya viene correcto del API
   611→   611→  }
   612→   612→
   613→   613→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   614→   614→  if (!rutaConfig) {
   615→   615→    // Fallback: try original ruta
   616→   616→    const origConfig = RUTA_PARADAS[ruta];
   617→   617→    if (!origConfig) return [];
   618→   618→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   619→   619→    return paradas.map(parada => {
   620→   620→      const precio = getPrecio(parada, ruta, tipo);
   621→   621→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   622→   622→    });
   623→   623→  }
   624→   624→
   625→   625→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   626→   626→  return paradas.map(parada => {
   627→   627→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   628→   628→    return {
   629→   629→      parada,
   630→   630→      normal: precio?.normal ?? 0,
   631→   631→      media: precio?.media ?? 0,
   632→   632→    };
   633→   633→  });
   634→   634→}

   636→   636→// Normalizar ruta y tipo para rutas "X - Loja"
   637→   637→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   638→   638→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   639→   639→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   640→   640→    // El API ya asigna tipo correcto, NO invertir
   641→   641→    const baseName = ruta.replace(' - Loja', '');
   642→   642→    return { ruta: `Loja - ${baseName}`, tipo };
   643→   643→  }
   644→   644→  return { ruta, tipo };
   645→   645→}

   647→   647→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   648→   648→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   649→   649→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   650→   650→  const precio = getPrecio(parada, nr, nt);
   651→   651→  if (!precio) return TARIFA_MINIMA;
   652→   652→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   653→   653→}

   660→   660→// Buscar la ruta más cercana dado un nombre de ruta parcial
   661→   661→export function matchRuta(rutaNombre: string): string {
   662→   662→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   663→   663→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   664→   664→  if (RUTA_PARADAS[normalized]) return normalized;
   665→   665→
   666→   666→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   667→   667→  for (const key of Object.keys(RUTA_PARADAS)) {
   668→   668→    const keyParts = key.split(' - ');
   669→   669→    if (parts.length >= 2 && keyParts.length >= 2) {
   670→   670→      const match = (
   671→   671→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   672→   672→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   673→   673→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   674→   674→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   675→   675→      );
   676→   676→      if (match) return key;
   677→   677→    }
   678→   678→  }
   679→   679→  return rutaNombre;
   680→   680→}
