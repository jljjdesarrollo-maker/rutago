    66→function getRutaColorKey(rutaNombre: string): RutaColorKey {
    67→  const ruta = matchRuta(rutaNombre).toLowerCase();
    68→  if (ruta.includes('el tambo')) return 'el_tambo';
    69→  if (ruta.includes('la elvira')) return 'la_elvira';
    70→  if (ruta.includes('yangana')) return 'yangana';
    71→  if (ruta.includes('zahuayco')) return 'zahuayco';
    72→  return 'vilcabamba';
    73→}

    75→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    76→
    77→function estadosKey(vtCode: string, fecha: string) {
    78→  return `rg_estados_${vtCode}_${fecha}`;
    79→}
    80→
    81→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    82→  try {
    83→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    84→  } catch (e) { console.error('Error saving estados to LS:', e); }
    85→}
    86→
    87→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    88→  try {
    89→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    90→    return raw ? JSON.parse(raw) : null;
    91→  } catch { return null; }
    92→}
    93→
    94→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    95→  const all = loadEstadosFromLS(vtCode, fecha);
    96→  if (!all) return;
    97→  const idx = all.findIndex(e => e.estadoId === estadoId);
    98→  if (idx >= 0) {
    99→    all[idx] = { ...all[idx], ...updates };
   100→    saveEstadosToLS(vtCode, fecha, all);
   101→  }
   102→}

   104→export function FrecuenciaSelector({ session, onOpenFrequency, onGoToArqueo, onGoToArqueoGeneral, onBack, onGoToSync }: Props) {

   138→        const data = await res.json();

   140→
   141→        // Try loading saved estados from localStorage first
   142→        const saved = loadEstadosFromLS(session.vtCode, fecha);
   143→        if (saved && saved.length === data.length) {
   144→          setEstados(saved);
   145→        } else {
   146→          // Create fresh pendiente estados
   147→          const newEstados: FrecuenciaEstado[] = data.map((f: FrecuenciaData) => ({
   148→            id: f.id,
   149→            estadoId: `${fecha}_${f.id}`,
   150→            frecuenciaId: f.id,
   151→            nombre: f.nombre,
   152→            ruta: f.ruta,
   153→            hora: f.hora,
   154→            direccion: f.direccion,
   155→            estado: 'pendiente' as const,
   156→            ventasCount: 0,
   157→            totalRecaudado: 0,
   158→          }));
   159→          setEstados(newEstados);
   160→          saveEstadosToLS(session.vtCode, fecha, newEstados);
   161→        }
   162→      }
   163→    } catch (error) {
   164→      console.error('Error cargando frecuencias:', error);
   165→      // Fallback: try to restore from localStorage only
   166→      const saved = loadEstadosFromLS(session.vtCode, fecha);
   167→      if (saved) setEstados(saved);
   168→    } finally {
   169→      setLoading(false);
   170→    }
   171→  }, [session.vtCode, fecha]);
   172→
   173→  const loadPendingCount = useCallback(async () => {
   174→    try {
   175→      const count = await countVentasPendientes();
   176→      setPendingCount(count);
   177→    } catch { /* ignore */ }
   178→  }, []);
   179→
   180→  // Reload ventas counts from IndexedDB periodically
   181→  const refreshVentasCounts = useCallback(async () => {
   182→    for (const estado of estados) {
   183→      try {
   184→        const ventas = await getVentasByFrecuencia(estado.estadoId);
   185→        // Count ALL ventas (including synced) for display in arqueo
   186→        const newCount = ventas.length;
   187→        const newTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   188→        if (newCount !== estado.ventasCount || Math.abs(newTotal - estado.totalRecaudado) > 0.01) {
   189→          updateEstadoInLS(session.vtCode, fecha, estado.estadoId, { ventasCount: newCount, totalRecaudado: newTotal });
   190→          setEstados(prev => prev.map(e =>
   191→            e.estadoId === estado.estadoId ? { ...e, ventasCount: newCount, totalRecaudado: newTotal } : e

   193→        }
   194→      } catch { /* ignore */ }
   195→    }
   196→  }, [estados, session.vtCode, fecha]);
   197→
   198→  useEffect(() => {
   199→    loadFrecuencias();

   202→    return () => clearInterval(interval);
   203→  }, [loadFrecuencias, loadPendingCount]);
   204→
   205→  // Refresh counts after initial load and periodically
   206→  useEffect(() => {
   207→    if (!loading && estados.length > 0) {
   208→      refreshVentasCounts();
   209→      const interval = setInterval(refreshVentasCounts, 5000);
   210→      return () => clearInterval(interval);
   211→    }
   212→  }, [loading, estados.length, refreshVentasCounts]);
   213→
   214→  const updateEstado = (estadoId: string, updates: Partial<FrecuenciaEstado>) => {
   215→    // Update in-memory state
   216→    setEstados(prev => prev.map(e => e.estadoId === estadoId ? { ...e, ...updates } : e));
   217→    // Persist to localStorage (synchronous, always reliable)
   218→    updateEstadoInLS(session.vtCode, fecha, estadoId, updates);
   219→  };
   220→
   221→  // Sequential: frequency available if previous are all cerrada/no_realizada
   222→  const isFrecuenciaDisponible = (index: number): boolean => {
   223→    const estado = estados[index];
   224→    if (!estado) return false;
   225→    if (estado.estado === 'cerrada' || estado.estado === 'no_realizada') return false;
   226→    for (let i = 0; i < index; i++) {
   227→      const prev = estados[i];
   228→      if (prev && prev.estado !== 'cerrada' && prev.estado !== 'no_realizada') {
   229→        return false;
   230→      }
   231→    }
   232→    return true;
   233→  };

   244→      const min = promoConfig.rangoMin || 3;
   245→      const max = promoConfig.rangoMax || 30;
   246→      ganadorPosicion = Math.floor(Math.random() * (max - min + 1)) + min;
   247→    }
   248→
   249→    const updatedEstado = { ...estado, estado: 'abierta' as const, ganadorPosicion };
   250→    updateEstado(estado.estadoId, {
   251→      estado: 'abierta',
   252→      ganadorPosicion,
   253→      ...(gps ? { gpsLatStart: gps.lat, gpsLngStart: gps.lng } : {}),

   255→    onOpenFrequency(updatedEstado);
   256→  };

   258→  const handleNoRealizada = (estado: FrecuenciaEstado) => {

   260→    setMotivoSeleccionado('');

   275→      estado: 'no_realizada',

   303→    setCajaComunModal(null);

   305→    const updatedEstado = { ...cajaComunModal, cajaComunCount: cajaComunCount, cajaComunMonto: montoNum };

   308→
   309→  // Filtrar frecuencias por origen para reasignación (usa TODAS las frecuencias del sistema)

   315→    const origen = partes[0]?.trim().toLowerCase();
   316→    if (!origen) return allFrecuencias;
   317→    // Filtrar frecuencias que tengan el mismo origen y ordenar por hora
   318→    return allFrecuencias
   319→      .filter(f => {
   320→        const fPartes = f.ruta.split(' - ');
   321→        const fOrigen = fPartes[0]?.trim().toLowerCase();

   332→        if (res.ok) {

   334→          setAllFrecuencias(data);
   335→        }

   339→  };

   341→  const handleSelectReassignment = (targetFrec: FrecuenciaData) => {

   344→      id: targetFrec.id, frecuenciaId: targetFrec.id,

   346→      hora: targetFrec.hora, direccion: targetFrec.direccion,

   360→  };
   361→

   365→      case 'abierta': return 'bg-green-50 border-green-200';

   368→      default: return 'bg-gray-100 border-gray-200';

   372→  const MOTIVOS_NO_REALIZADA = [
   373→    { id: 'daño_unidad', label: 'Daño en la unidad', icon: <Truck className="w-4 h-4" />, color: 'text-red-500' },
   374→    { id: 'mantenimiento', label: 'Mantenimiento', icon: <Wrench className="w-4 h-4" />, color: 'text-blue-500' },

   378→    { id: 'orden_superior', label: 'Orden superior', icon: <Ban className="w-4 h-4" />, color: 'text-gray-600' },
   379→    { id: 'ingreso_especial', label: 'Ingreso especial', icon: <Sparkles className="w-4 h-4" />, color: 'text-amber-500' },

   382→

   414→  const allFrecuenciasDone = estados.length > 0 && estados.every(e => e.estado === 'cerrada' || e.estado === 'no_realizada');
   415→
   416→  // Totals across all frequencies

   418→  const totalRecaudadoAll = estados.reduce((s, e) => s + e.totalRecaudado, 0);
   419→  const cerradasCount = estados.filter(e => e.estado === 'cerrada').length;
   420→
   421→  // SYNC control: ALWAYS block arqueo general if there are pending ventas

   424→  const handleSyncForArqueo = async () => {

   436→    setSyncingForArqueo(false);
   437→  };
   438→
   439→  return (
   440→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   441→      {/* Header */}
   442→      <div className="bg-[#912D26] text-white px-4 py-3">
   443→        <div className="flex items-center justify-between mb-2">
   444→          <div className="flex items-center gap-2">
