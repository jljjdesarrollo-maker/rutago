    56→function getVTRouteType(frecuencias?: { routeFrom: string; routeTo: string }[]): VTRouteType {
    57→  if (!frecuencias || frecuencias.length === 0) return 'vilcabamba';
    58→  const allDests = [...frecuencias.map(f => f.routeTo), ...frecuencias.map(f => f.routeFrom)];
    59→  if (allDests.some(d => d === 'El Tambo')) return 'el_tambo';
    60→  if (allDests.some(d => d === 'La Elvira')) return 'la_elvira';
    61→  if (allDests.some(d => d === 'Yangana')) return 'yangana';
    62→  if (allDests.some(d => d === 'Zahuayco')) return 'zahuayco';
    63→  return 'vilcabamba';
    64→}

   473→            <p className="text-yellow-600 text-sm mt-1">Ejecuta el seed: POST /api/seed-vts</p>

   476→          <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-100">
   477→            <h2 className="text-base font-bold text-[#3A3A3A] mb-3 flex items-center gap-2">
   478→              <Bus className="w-4 h-4 text-[#912D26]" /> Unidad
   479→            </h2>

   497→                      <span className={`absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full ${style.badgeBg} ${style.badgeText} shadow-sm`}>
   498→                        {style.label}
   499→                      </span>
   500→                    )}

   502→                      <span className="absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full bg-white/25 text-white shadow-sm">
   503→                        {style.label}
   504→                      </span>
   505→                    )}
   506→                  </button>

   508→              })}
   509→            </div>
   510→          </div>
   511→        )}
   512→
   513→        {/* Boton Iniciar — grande y prominente */}
   514→        <button onClick={handleStart} disabled={!selectedVT || !ayudante || pendingVentasCount > 0}
   515→          className={`w-full py-5 rounded-2xl font-black text-xl flex items-center justify-center gap-3 transition-all ${
   516→            selectedVT && ayudante && pendingVentasCount === 0
   517→              ? 'bg-[#912D26] text-white shadow-xl shadow-red-300 active:scale-[0.97]'

   522→            ? `ESPERANDO SYNC — ${pendingVentasCount} VENTAS`
   523→            : 'Iniciar Turno'
   524→          } <ArrowRight className="w-6 h-6" />
   525→        </button>
   526→      </div>
   527→
   528→      {/* Ventas pendientes — bloquear inicio de nuevo VT */}
   529→      {pendingVentasCount > 0 && pendingCheckDone && (
   530→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-[#912D26] to-[#b33d34] p-4 shadow-2xl z-40">
   531→          <div className="max-w-lg mx-auto">
   532→            <div className="flex items-center gap-2 mb-2">
   533→              <AlertTriangle className="w-5 h-5 text-white" />
   534→              <p className="text-white font-bold text-sm">Ventas sin sincronizar</p>
   535→            </div>
   536→            <p className="text-white/90 text-xs mb-3">
   537→              Tienes <strong>{pendingVentasCount} venta{pendingVentasCount !== 1 ? 's' : ''}</strong> pendiente{pendingVentasCount !== 1 ? 's' : ''} de un turno anterior. Debes sincronizar antes de iniciar uno nuevo.
   538→            </p>
   539→            {navigator.onLine ? (
   540→              <button onClick={handleForceSync} disabled={syncingPending}
   541→                className="w-full py-3 rounded-xl bg-white text-[#912D26] font-black text-sm flex items-center justify-center gap-2 active:scale-[0.98] disabled:opacity-60">

   544→              </button>
   545→            ) : (
   546→              <div className="w-full py-3 rounded-xl bg-white/20 text-white/80 text-sm flex items-center justify-center gap-2">
   547→                <WifiOff className="w-4 h-4" /> SIN INTERNET — CONECTA PARA SINCRONIZAR
   548→              </div>
   549→            )}
   550→          </div>
   551→        </div>
   552→      )}
   553→
   554→      {/* Existing unfinished session banner */}
   555→      {existingUnfinished && existingSession && (
   556→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-orange-500 to-red-500 p-4 shadow-2xl z-40">
   557→          <div className="max-w-lg mx-auto">
   558→            <div className="flex items-center gap-2 mb-2">
   559→              <AlertTriangle className="w-5 h-5 text-white" />
   560→              <p className="text-white font-bold text-sm">Turno sin terminar</p>

   562→            <p className="text-white/90 text-xs mb-3">

   564→            </p>
   565→            <div className="flex gap-2">
   566→              <button onClick={handleResumeOld}
   567→                className="flex-1 py-2.5 rounded-xl bg-white text-orange-600 font-bold text-sm active:scale-[0.98]">
   568→                Continuar VT {existingSession.vtCode}
   569→              </button>
   570→              <button onClick={() => setConfirmNewSession(true)}
   571→                className="flex-1 py-2.5 rounded-xl bg-white/20 text-white font-semibold text-sm active:scale-[0.98]">
   572→                Iniciar nuevo
   573→              </button>
   574→            </div>
   575→          </div>
   576→        </div>
   577→      )}
   578→
   579→      {/* Confirm new session modal */}
   580→      {confirmNewSession && existingSession && (
   581→        <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
   582→          <div className="bg-white rounded-2xl w-full max-w-sm p-6 text-center">
   583→            <div className="w-16 h-16 bg-red-100 rounded-full flex items-center justify-center mx-auto mb-4">
   584→              <AlertTriangle className="w-8 h-8 text-[#912D26]" />
   585→            </div>
   586→            <h3 className="text-lg font-bold text-[#3A3A3A] mb-2">Abandonar turno anterior?</h3>
   587→            <div className="bg-red-50 rounded-xl p-3 mb-4">
   588→              <p className="text-sm text-[#912D26] font-semibold">
   589→                El turno de {existingSession.nombre} ({existingSession.vtCode}) tiene frecuencias pendientes/abiertas que no se han completado.
   590→              </p>
   591→            </div>
   592→            <p className="text-sm text-gray-500 mb-5">
   593→              Los datos del turno anterior se conservaran en el dispositivo. Podras retomarlo mas tarde desde la pantalla de inicio.
   594→            </p>
   595→            <div className="flex gap-2">
   596→              <button onClick={() => setConfirmNewSession(false)}
   597→                className="flex-1 py-3 rounded-xl bg-[#912D26] text-white font-bold text-sm active:scale-[0.98] shadow-lg shadow-red-200">
   598→                Cancelar
   599→              </button>
   600→              <button onClick={handleForceNew}
   601→                className="flex-1 py-3 rounded-xl border-2 border-red-200 text-[#912D26] font-bold text-sm active:scale-[0.98]">
   602→                Nuevo Turno
   603→              </button>
   604→            </div>
   605→          </div>
   606→        </div>
   607→      )}
   608→    </div>
   609→  );
