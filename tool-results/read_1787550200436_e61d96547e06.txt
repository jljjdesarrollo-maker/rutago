   128→   128→}

   682→   682→// ─── Funciones de precio con soporte por dirección ───
   683→   683→//
   684→   684→// Lógica de precios (tipo semántico asignado por el API):
   685→   685→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   686→   686→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   687→   687→//
   688→   688→// El API asigna dirección según el origen:
   689→   689→//   Si origen = Loja →  "ida" (alejándose)
   690→   690→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   691→   691→//
   692→   692→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   693→   693→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   694→   694→
   695→   695→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   696→   696→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   697→   697→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   698→   698→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   699→   699→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   700→   700→  return false;
   701→   701→}

   703→   703→// Obtener precio considerando la dirección del viaje
   704→   704→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   705→   705→  if (viajaHaciaLoja(ruta, tipo)) {
   706→   706→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   707→   707→    if (ruta === 'Loja - El Tambo') {
   708→   708→      const tamboVuelta = preciosVueltaElTambo[parada];
   709→   709→      if (tamboVuelta) return tamboVuelta;
   710→   710→    }
   711→   711→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   712→   712→    if (ruta === 'Loja - Yangana') {
   713→   713→      const yangVuelta = preciosVueltaYangana[parada];
   714→   714→      if (yangVuelta) return yangVuelta;
   715→   715→    }
   716→   716→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   717→   717→    if (ruta === 'Loja - Vilcabamba') {
   718→   718→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   719→   719→      if (vilcVuelta) return vilcVuelta;
   720→   720→    }
   721→   721→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   722→   722→    if (ruta === 'Loja - La Elvira') {
   723→   723→      const elviraVuelta = preciosVueltaLaElvira[parada];
   724→   724→      if (elviraVuelta) return elviraVuelta;
   725→   725→    }
   726→   726→    const vuelta = preciosVuelta[parada];
   727→   727→    if (vuelta) return vuelta;
   728→   728→    // Si la parada no tiene precio de vuelta, buscar en precios base
   729→   729→  }
   730→   730→  // Buscar primero en ida (base), luego en vuelta como fallback
   731→   731→  const ida = preciosIda[parada];
   732→   732→  if (ida) return ida;
   733→   733→  const vuelta = preciosVuelta[parada];
   734→   734→  return vuelta || null;
   735→   735→}

   740→   740→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   741→   741→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   742→   742→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   743→   743→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   744→   744→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   745→   745→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   746→   746→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   747→   747→  let effectiveRuta = ruta;
   748→   748→  let effectiveTipo = tipo;
   749→   749→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   750→   750→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   751→   751→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   752→   752→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   753→   753→    const baseName = ruta.replace(' - Loja', '');
   754→   754→    effectiveRuta = `Loja - ${baseName}`;
   755→   755→    // NO invertir effectiveTipo — ya viene correcto del API
   756→   756→  }
   757→   757→
   758→   758→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   759→   759→  if (!rutaConfig) {
   760→   760→    // Fallback: try original ruta
   761→   761→    const origConfig = RUTA_PARADAS[ruta];
   762→   762→    if (!origConfig) return [];
   763→   763→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   764→   764→    return paradas.map(parada => {
   765→   765→      const precio = getPrecio(parada, ruta, tipo);
   766→   766→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   767→   767→    });
   768→   768→  }
   769→   769→
   770→   770→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   771→   771→  return paradas.map(parada => {
   772→   772→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   773→   773→    return {
   774→   774→      parada,
   775→   775→      normal: precio?.normal ?? 0,
   776→   776→      media: precio?.media ?? 0,
   777→   777→    };
   778→   778→  });
   779→   779→}

   781→   781→// Normalizar ruta y tipo para rutas "X - Loja"
   782→   782→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   783→   783→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   784→   784→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   785→   785→    // El API ya asigna tipo correcto, NO invertir
   786→   786→    const baseName = ruta.replace(' - Loja', '');
   787→   787→    return { ruta: `Loja - ${baseName}`, tipo };
   788→   788→  }
   789→   789→  return { ruta, tipo };
   790→   790→}

   793→   793→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   794→   794→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);

   796→   796→  if (!precio) return TARIFA_MINIMA;
   797→   797→  return pasajeroTipo === 'media' ? precio.media : precio.normal;

   805→   805→// Buscar la ruta más cercana dado un nombre de ruta parcial
   806→   806→export function matchRuta(rutaNombre: string): string {
   807→   807→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   808→   808→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   809→   809→  if (RUTA_PARADAS[normalized]) return normalized;
   810→   810→
   811→   811→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   812→   812→  for (const key of Object.keys(RUTA_PARADAS)) {
   813→   813→    const keyParts = key.split(' - ');
   814→   814→    if (parts.length >= 2 && keyParts.length >= 2) {
   815→   815→      const match = (
   816→   816→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   817→   817→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   818→   818→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   819→   819→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   820→   820→      );
   821→   821→      if (match) return key;
   822→   822→    }
   823→   823→  }
   824→   824→  return rutaNombre;
   825→   825→}
