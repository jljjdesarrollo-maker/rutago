    56→    56→function getVTRouteType(frecuencias?: { routeFrom: string; routeTo: string }[]): VTRouteType {
    57→    57→  if (!frecuencias || frecuencias.length === 0) return 'vilcabamba';
    58→    58→  const allDests = [...frecuencias.map(f => f.routeTo), ...frecuencias.map(f => f.routeFrom)];
    59→    59→  if (allDests.some(d => d === 'El Tambo')) return 'el_tambo';
    60→    60→  if (allDests.some(d => d === 'La Elvira')) return 'la_elvira';
    61→    61→  if (allDests.some(d => d === 'Yangana')) return 'yangana';
    62→    62→  if (allDests.some(d => d === 'Zahuayco')) return 'zahuayco';
    63→    63→  return 'vilcabamba';
    64→    64→}
