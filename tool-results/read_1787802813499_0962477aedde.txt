    66→function getRutaColorKey(rutaNombre: string): RutaColorKey {
    67→  const ruta = matchRuta(rutaNombre).toLowerCase();
    68→  if (ruta.includes('el tambo')) return 'el_tambo';
    69→  if (ruta.includes('la elvira')) return 'la_elvira';
    70→  if (ruta.includes('yangana')) return 'yangana';
    71→  if (ruta.includes('zahuayco')) return 'zahuayco';
    72→  return 'vilcabamba';
    73→}
    74→
    75→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    76→
    77→function estadosKey(vtCode: string, fecha: string) {
    78→  return `rg_estados_${vtCode}_${fecha}`;
    79→}
    80→
    81→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    82→  try {
    83→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    84→  } catch (e) { console.error('Error saving estados to LS:', e); }
    85→}
    86→
    87→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    88→  try {
    89→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    90→    return raw ? JSON.parse(raw) : null;
    91→  } catch { return null; }
    92→}
    93→
    94→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    95→  const all = loadEstadosFromLS(vtCode, fecha);
    96→  if (!all) return;
    97→  const idx = all.findIndex(e => e.estadoId === estadoId);
    98→  if (idx >= 0) {
    99→    all[idx] = { ...all[idx], ...updates };
   100→    saveEstadosToLS(vtCode, fecha, all);
   101→  }
   102→}

   104→export function FrecuenciaSelector({ session, onOpenFrequency, onGoToArqueo, onGoToArqueoGeneral, onBack, onGoToSync }: Props) {

   106→  const [frecuencias, setFrecuencias] = useState<FrecuenciaData[]>([]);
   107→  const [loading, setLoading] = useState(true);
   108→  const [reassigning, setReassigning] = useState<string | null>(null);
   109→  const [pendingCount, setPendingCount] = useState(0);
   110→  const [isOnline, setIsOnline] = useState(navigator.onLine);
   111→  const [noRealizadaModal, setNoRealizadaModal] = useState<FrecuenciaEstado | null>(null);
   112→  const [motivoSeleccionado, setMotivoSeleccionado] = useState('');

   115→  const [allFrecuencias, setAllFrecuencias] = useState<FrecuenciaData[]>([]);
   116→  const [exitModal, setExitModal] = useState(false);
   117→  const [cajaComunModal, setCajaComunModal] = useState<FrecuenciaEstado | null>(null);
   118→  const [cajaComunCount, setCajaComunCount] = useState(0);
   119→  const [cajaComunMonto, setCajaComunMonto] = useState('');

   121→  const fecha = today();
   122→
   123→  useEffect(() => {
   124→    const handle = () => setIsOnline(navigator.onLine);
   125→    window.addEventListener('online', handle);
   126→    window.addEventListener('offline', handle);
   127→    return () => { window.removeEventListener('online', handle); window.removeEventListener('offline', handle); };
   128→  }, []);
   129→
   130→  const loadFrecuencias = useCallback(async () => {
   131→    try {

   133→      if (res.ok) {
   134→        const data = await res.json();
   135→        setFrecuencias(data);
   136→
   137→        // Try loading saved estados from localStorage first
   138→        const saved = loadEstadosFromLS(session.vtCode, fecha);
   139→        if (saved && saved.length === data.length) {
   140→          setEstados(saved);
   141→        } else {
   142→          // Create fresh pendiente estados
   143→          const newEstados: FrecuenciaEstado[] = data.map((f: FrecuenciaData) => ({
   144→            id: f.id,
   145→            estadoId: `${fecha}_${f.id}`,
   146→            frecuenciaId: f.id,
   147→            nombre: f.nombre,
   148→            ruta: f.ruta,
   149→            hora: f.hora,
   150→            direccion: f.direccion,
   151→            estado: 'pendiente' as const,
   152→            ventasCount: 0,
   153→            totalRecaudado: 0,
   154→          }));
   155→          setEstados(newEstados);
   156→          saveEstadosToLS(session.vtCode, fecha, newEstados);
   157→        }
   158→      }
   159→    } catch (error) {
   160→      console.error('Error cargando frecuencias:', error);
   161→      // Fallback: try to restore from localStorage only
   162→      const saved = loadEstadosFromLS(session.vtCode, fecha);
   163→      if (saved) setEstados(saved);
   164→    } finally {
   165→      setLoading(false);
   166→    }
   167→  }, [session.vtCode, fecha]);
   168→
   169→  const loadPendingCount = useCallback(async () => {
   170→    try {
   171→      const count = await countVentasPendientes();
   172→      setPendingCount(count);
   173→    } catch { /* ignore */ }
   174→  }, []);
   175→
   176→  // Reload ventas counts from IndexedDB periodically
   177→  const refreshVentasCounts = useCallback(async () => {
   178→    for (const estado of estados) {
   179→      try {
   180→        const ventas = await getVentasByFrecuencia(estado.estadoId);
   181→        // Count ALL ventas (including synced) for display in arqueo
   182→        const newCount = ventas.length;
   183→        const newTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   184→        if (newCount !== estado.ventasCount || Math.abs(newTotal - estado.totalRecaudado) > 0.01) {
   185→          updateEstadoInLS(session.vtCode, fecha, estado.estadoId, { ventasCount: newCount, totalRecaudado: newTotal });
   186→          setEstados(prev => prev.map(e =>
   187→            e.estadoId === estado.estadoId ? { ...e, ventasCount: newCount, totalRecaudado: newTotal } : e
   188→          ));
   189→        }
   190→      } catch { /* ignore */ }
   191→    }
   192→  }, [estados, session.vtCode, fecha]);
   193→
   194→  useEffect(() => {
   195→    loadFrecuencias();
   196→    loadPendingCount();
   197→    const interval = setInterval(() => { loadPendingCount(); }, 5000);

   199→  }, [loadFrecuencias, loadPendingCount]);
   200→
   201→  // Refresh counts after initial load and periodically
   202→  useEffect(() => {
   203→    if (!loading && estados.length > 0) {
   204→      refreshVentasCounts();
   205→      const interval = setInterval(refreshVentasCounts, 5000);
   206→      return () => clearInterval(interval);
   207→    }
   208→  }, [loading, estados.length, refreshVentasCounts]);
   209→
   210→  const updateEstado = (estadoId: string, updates: Partial<FrecuenciaEstado>) => {
   211→    // Update in-memory state
   212→    setEstados(prev => prev.map(e => e.estadoId === estadoId ? { ...e, ...updates } : e));
   213→    // Persist to localStorage (synchronous, always reliable)
   214→    updateEstadoInLS(session.vtCode, fecha, estadoId, updates);
   215→  };
   216→
   217→  // Sequential: frequency available if previous are all cerrada/no_realizada
   218→  const isFrecuenciaDisponible = (index: number): boolean => {
   219→    const estado = estados[index];
   220→    if (!estado) return false;
   221→    if (estado.estado === 'cerrada' || estado.estado === 'no_realizada') return false;
   222→    for (let i = 0; i < index; i++) {
   223→      const prev = estados[i];
   224→      if (prev && prev.estado !== 'cerrada' && prev.estado !== 'no_realizada') {
   225→        return false;
   226→      }
   227→    }
   228→    return true;
   229→  };

   233→    const { getGPSPosition } = await import('@/lib/gps');

   235→
   236→    // Generate random winner position for Viaje Gratis
   237→    const promoConfig = loadPromoConfig();

   239→    if (promoConfig.activa) {
   240→      const min = promoConfig.rangoMin || 3;
   241→      const max = promoConfig.rangoMax || 30;
   242→      ganadorPosicion = Math.floor(Math.random() * (max - min + 1)) + min;
   243→    }
   244→
   245→    const updatedEstado = { ...estado, estado: 'abierta' as const, ganadorPosicion };
   246→    updateEstado(estado.estadoId, {
   247→      estado: 'abierta',
   248→      ganadorPosicion,
   249→      ...(gps ? { gpsLatStart: gps.lat, gpsLngStart: gps.lng } : {}),
   250→    });
   251→    onOpenFrequency(updatedEstado);
   252→  };
   253→
   254→  const handleNoRealizada = (estado: FrecuenciaEstado) => {
   255→    setNoRealizadaModal(estado);
   256→    setMotivoSeleccionado('');
   257→    setMotivoPersonalizado('');

   259→  };
   260→
   261→  const confirmNoRealizadaAction = () => {
   262→    if (!noRealizadaModal) return;
   263→    const motivo = motivoSeleccionado === 'otro' ? motivoPersonalizado.trim() : motivoSeleccionado;

   265→      estado: 'no_realizada',
   266→      ventasCount: 0,
   267→      totalRecaudado: 0,
   268→      motivoNoRealizada: motivo || 'Sin especificar',
   269→    });
   270→    setNoRealizadaModal(null);
   271→  };

   273→  // ─── Caja Común (Oficina Loja) ───
   274→  const handleGoToArqueo = (estado: FrecuenciaEstado, esUltima: boolean) => {
   275→    // Open caja común modal first, then proceed to arqueo

   279→    setCajaComunCount(estado.cajaComunCount || 0);
   280→    setCajaComunMonto(estado.cajaComunMonto ? estado.cajaComunMonto.toString() : '');
   281→  };
   282→
   283→  const confirmCajaComun = () => {

   285→    const montoNum = parseFloat(cajaComunMonto) || 0;
   286→    // Save caja común data to the estado
   287→    updateEstado(cajaComunModal.estadoId, {
   288→      cajaComunCount: cajaComunCount,
   289→      cajaComunMonto: montoNum,
   290→    });
   291→    setCajaComunModal(null);
   292→    // Proceed to arqueo with the updated estado
   293→    const updatedEstado = { ...cajaComunModal, cajaComunCount: cajaComunCount, cajaComunMonto: montoNum };
   294→    onGoToArqueo(updatedEstado, cajaComunEsUltima);
   295→  };
   296→
   297→  // Filtrar frecuencias por origen para reasignación (usa TODAS las frecuencias del sistema)
   298→  const getFrecuenciasParaReasignar = (estadoId: string): FrecuenciaData[] => {
   299→    const estadoActual = estados.find(e => e.estadoId === estadoId);
   300→    if (!estadoActual) return allFrecuencias;
   301→    // Extraer origen de la ruta: "Vilcabamba - Loja" → "Vilcabamba"
   302→    const partes = estadoActual.ruta.split(' - ');
   303→    const origen = partes[0]?.trim().toLowerCase();
   304→    if (!origen) return allFrecuencias;
   305→    // Filtrar frecuencias que tengan el mismo origen y ordenar por hora
   306→    return allFrecuencias
   307→      .filter(f => {
   308→        const fPartes = f.ruta.split(' - ');
   309→        const fOrigen = fPartes[0]?.trim().toLowerCase();
   310→        return fOrigen === origen;
   311→      })
   312→      .sort((a, b) => a.hora.localeCompare(b.hora));
   313→  };
   314→
   315→  const handleReassign = async (estado: FrecuenciaEstado) => {
   316→    // Cargar todas las frecuencias del sistema si no están cargadas

   318→      try {
   319→        const res = await fetch('/api/frecuencias?all=true');
   320→        if (res.ok) {
   321→          const data = await res.json();
   322→          setAllFrecuencias(data);
   323→        }
   324→      } catch { /* usar las que tengamos */ }
   325→    }
   326→    setReassigning(estado.estadoId);
   327→  };
   328→
   329→  const handleSelectReassignment = (targetFrec: FrecuenciaData) => {
   330→    if (!reassigning) return;
   331→    updateEstado(reassigning, {
   332→      id: targetFrec.id, frecuenciaId: targetFrec.id,
   333→      nombre: targetFrec.nombre, ruta: targetFrec.ruta,
   334→      hora: targetFrec.hora, direccion: targetFrec.direccion,
   335→      estado: 'pendiente', ventasCount: 0, totalRecaudado: 0,
   336→    });
   337→    setReassigning(null);
   338→  };
   339→
   340→  const getStateIcon = (estado: string) => {
   341→    switch (estado) {
   342→      case 'pendiente': return <Clock className="w-4 h-4 text-gray-400" />;
   343→      case 'abierta': return <Play className="w-4 h-4 text-green-500" />;
   344→      case 'cerrada': return <CheckCircle2 className="w-4 h-4 text-blue-500" />;
   345→      case 'no_realizada': return <XCircle className="w-4 h-4 text-orange-500" />;
   346→      default: return null;
   347→    }
   348→  };
   349→
   350→  const getStateColor = (estado: string) => {
   351→    switch (estado) {
   352→      case 'pendiente': return 'bg-gray-100 border-gray-200';
   353→      case 'abierta': return 'bg-green-50 border-green-200';
   354→      case 'cerrada': return 'bg-blue-50 border-blue-200';
   355→      case 'no_realizada': return 'bg-orange-50 border-orange-200';
   356→      default: return 'bg-gray-100 border-gray-200';
   357→    }
   358→  };
   359→
   360→  const MOTIVOS_NO_REALIZADA = [
   361→    { id: 'daño_unidad', label: 'Daño en la unidad', icon: <Truck className="w-4 h-4" />, color: 'text-red-500' },
   362→    { id: 'mantenimiento', label: 'Mantenimiento', icon: <Wrench className="w-4 h-4" />, color: 'text-blue-500' },
   363→    { id: 'clima', label: 'Clima / Lluvia', icon: <Droplets className="w-4 h-4" />, color: 'text-cyan-500' },
   364→    { id: 'sin_pasajeros', label: 'Sin pasajeros', icon: <UserX className="w-4 h-4" />, color: 'text-purple-500' },
   365→    { id: 'problema_ruta', label: 'Problema en la ruta', icon: <AlertTriangle className="w-4 h-4" />, color: 'text-yellow-500' },
   366→    { id: 'orden_superior', label: 'Orden superior', icon: <Ban className="w-4 h-4" />, color: 'text-gray-600' },
   367→    { id: 'otro', label: 'Otro motivo', icon: <FileText className="w-4 h-4" />, color: 'text-gray-500' },
   368→  ];
   369→
   370→  const getEstadoBadge = (estado: string) => {
   371→    switch (estado) {
   372→      case 'pendiente': return <span className="px-2 py-0.5 rounded-full text-xs bg-gray-200 text-gray-600">Pendiente</span>;
   373→      case 'abierta': return <span className="px-2 py-0.5 rounded-full text-xs bg-green-200 text-green-700">Abierta</span>;
   374→      case 'cerrada': return <span className="px-2 py-0.5 rounded-full text-xs bg-blue-200 text-blue-700">Cerrada</span>;
   375→      case 'no_realizada': return <span className="px-2 py-0.5 rounded-full text-xs bg-orange-200 text-orange-700">No Realizada</span>;
   376→      default: return null;
   377→    }
   378→  };
   379→
   380→  if (loading) {
   381→    return (
   382→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   383→        <RefreshCw className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   384→        <p className="text-[#3A3A3A]">Cargando frecuencias...</p>

   386→    );

   388→

   390→  const hasUnfinishedFrecuencias = estados.some(e => e.estado === 'pendiente' || e.estado === 'abierta');
   391→
   392→  const handleBack = () => {
   393→    if (hasUnfinishedFrecuencias) {
   394→      setExitModal(true);
   395→    } else {
   396→      onBack();
   397→    }
   398→  };
   399→
   400→  // Check if ALL frequencies are done (cerrada or no_realizada)

   402→
   403→  // Totals across all frequencies
   404→  const totalVentasAll = estados.reduce((s, e) => s + e.ventasCount, 0);
   405→  const totalRecaudadoAll = estados.reduce((s, e) => s + e.totalRecaudado, 0);
   406→  const cerradasCount = estados.filter(e => e.estado === 'cerrada').length;
   407→
   408→  // SYNC control: if online with pending ventas, block arqueo

   411→  return (

   413→      {/* Header */}

   415→        <div className="flex items-center justify-between mb-2">
   416→          <div className="flex items-center gap-2">
   417→            <button onClick={handleBack} className="text-red-100"><ArrowLeft className="w-5 h-5" /></button>
   418→            <div>
   419→              <h1 className="text-lg font-bold">Frecuencias</h1>
   420→              <p className="text-red-100 text-xs">{session.nombre} - {session.ayudanteNombre}</p>
   421→            </div>
   422→          </div>
   423→          <div className="flex items-center gap-2">
   424→            {isOnline ? <Wifi className="w-4 h-4 text-green-300" /> : <WifiOff className="w-4 h-4 text-red-300" />}
   425→          </div>
   426→        </div>
   427→        <div className="flex items-center gap-2">
   428→          <div className="flex-1 bg-white/15 rounded-xl px-3 py-1.5 text-sm flex items-center gap-2">
   429→            {pendingCount > 0 ? (
   430→              <>
   431→                <span className="bg-yellow-400 text-yellow-900 text-xs font-bold px-1.5 py-0.5 rounded-full">{pendingCount}</span>

   433→              </>
   434→            ) : (
   435→              <span className="text-green-200 text-xs">Sin pendientes</span>
   436→            )}
   437→          </div>
   438→          {pendingCount > 0 && isOnline && (
   439→            <button onClick={onGoToSync} className="bg-green-500 text-white px-3 py-1.5 rounded-xl text-xs font-bold flex items-center gap-1 active:scale-95">
   440→              <Send className="w-3 h-3" /> SYNC
   441→            </button>
   442→          )}
   443→          {pendingCount > 0 && !isOnline && (
   444→            <div className="bg-white/10 text-red-200 px-3 py-1.5 rounded-xl text-xs font-bold flex items-center gap-1">
   445→              <WifiOff className="w-3 h-3" /> Sin red
   446→            </div>
   447→          )}
   448→        </div>
