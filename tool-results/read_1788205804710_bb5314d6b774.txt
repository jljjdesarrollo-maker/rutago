    42→    42→
    43→    43→// Work date: use session.fecha (date when shift started, not necessarily today)

    57→    57→  const [loading, setLoading] = useState(true);

    60→    60→  const [tickets, setTickets] = useState('');

    62→    62→  const [fotoPreview, setFotoPreview] = useState<string | null>(null);
    63→    63→  const [saving, setSaving] = useState(false);

    66→    66→  const [errors, setErrors] = useState<string[]>([]);
    67→    67→  const [guardadoOffline, setGuardadoOffline] = useState(false);
    68→    68→
    69→    69→  // Inline edit caja común
    70→    70→  const [editingCajaComun, setEditingCajaComun] = useState<string | null>(null);
    71→    71→  const [editCajaComunValue, setEditCajaComunValue] = useState('');
    72→    72→
    73→    73→  const fileInputRef = useRef<HTMLInputElement>(null);
    74→    74→  const saveCajaComunToLS = useCallback((estadoId: string, newMonto: number) => {

    76→    76→    const lsKey = `rg_estados_${session.vtCode}_${fecha}`;

    78→    78→      const raw = localStorage.getItem(lsKey);

    81→    81→        const idx = all.findIndex((e: any) => e.estadoId === estadoId);
    82→    82→        if (idx >= 0) {
    83→    83→          all[idx].cajaComunMonto = newMonto;
    84→    84→          localStorage.setItem(lsKey, JSON.stringify(all));
    85→    85→        }

    87→    87→    } catch (e) { console.error('LS error:', e); }
    88→    88→    setFrecuencias(prev => prev.map(f =>
    89→    89→      f.estadoId === estadoId
    90→    90→        ? { ...f, cajaComunMonto: newMonto }

   105→   105→      const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
   106→   106→      const raw = localStorage.getItem(lsKey);
   107→   107→      if (!raw) { setLoading(false); return; }
   108→   108→      const estadosLS = JSON.parse(raw) as FrecuenciaEstado[];
   109→   109→      const cerradas = estadosLS.filter(e => e.estado === 'cerrada');
   110→   110→      // Ingresos especiales (no_realizada con monto > 0)
   111→   111→      const especiales = estadosLS.filter(e => e.estado === 'no_realizada' && (e.ingresoEspecialMonto || 0) > 0);
   112→   112→
   113→   113→      const resumenes: FrecuenciaResumen[] = await Promise.all(
   114→   114→        cerradas.map(async (e) => {
   115→   115→          const ventas = await getVentasByFrecuencia(e.estadoId);
   116→   116→          const sistemaTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   117→   117→          // Use arqueo data if available (what helper actually counted), else system total
   118→   118→          const efectivo = (e as any).arqueoEfectivo ?? sistemaTotal;
   119→   119→          const diff = efectivo - sistemaTotal;
   120→   120→          const ccCount = (e as any).cajaComunCount || 0;
   121→   121→          const ccMonto = (e as any).cajaComunMonto || 0;

   123→   123→            estadoId: e.estadoId,
   124→   124→            nombre: e.nombre,
   125→   125→            ruta: e.ruta,
   126→   126→            hora: e.hora,
   127→   127→            direccion: e.direccion,
   128→   128→            ventasCount: ventas.length,
   129→   129→            totalRecaudado: sistemaTotal,
   130→   130→            efectivoContado: efectivo,
   131→   131→            diferencia: diff,
   132→   132→            boletosCaja: ccCount,
   133→   133→            cajaComunMonto: ccMonto,
   134→   134→          };
   135→   135→        })
   136→   136→      );
   137→   137→
   138→   138→      // Add ingresos especiales as virtual rows
   139→   139→      especiales.forEach(e => {
   140→   140→        resumenes.push({
   141→   141→          estadoId: e.estadoId,
   142→   142→          nombre: e.nombre,
   143→   143→          ruta: e.ruta,
   144→   144→          hora: e.hora,
   145→   145→          direccion: e.direccion,
   146→   146→          ventasCount: 0,
   147→   147→          totalRecaudado: e.ingresoEspecialMonto || 0,
   148→   148→          efectivoContado: e.ingresoEspecialMonto || 0,
   149→   149→          diferencia: 0,
   150→   150→          boletosCaja: 0,
   151→   151→          cajaComunMonto: 0,
   152→   152→          isIngresoEspecial: true,
   153→   153→          ingresoEspecialNota: e.ingresoEspecialNota,
   154→   154→        });
   155→   155→      });
   156→   156→      setFrecuencias(resumenes);
   157→   157→    } catch (err) {
   158→   158→      console.error('Error cargando resumen:', err);
   159→   159→    } finally {
   160→   160→      setLoading(false);
   161→   161→    }
   162→   162→  }, [session.vtCode]);
   163→   163→
   164→   164→  useEffect(() => { loadData(); }, [loadData]);
   165→   165→
   166→   166→  const totalIngresosAuto = useMemo(() =>
   167→   167→    frecuencias.reduce((s, f) => s + f.totalRecaudado, 0),
   168→   168→    [frecuencias]
   169→   169→  );
   170→   170→  const totalEfectivoReal = useMemo(() =>
   171→   171→    frecuencias.reduce((s, f) => s + f.efectivoContado, 0),
   172→   172→    [frecuencias]
   173→   173→  );
   174→   174→  const totalDiferencia = useMemo(() =>
   175→   175→    frecuencias.reduce((s, f) => s + f.diferencia, 0),
   176→   176→    [frecuencias]
   177→   177→  );
   178→   178→  const totalVentas = useMemo(() =>
   179→   179→    frecuencias.reduce((s, f) => s + f.ventasCount, 0),
   180→   180→    [frecuencias]
   181→   181→  );
   182→   182→  const totalCajaComunPasajeros = useMemo(() =>
   183→   183→    frecuencias.reduce((s, f) => s + f.boletosCaja, 0),
   184→   184→    [frecuencias]
   185→   185→  );
   186→   186→  const totalCajaComunMonto = useMemo(() =>
   187→   187→    frecuencias.reduce((s, f) => s + (f.cajaComunMonto || 0), 0),
   188→   188→    [frecuencias]
   189→   189→  );
   190→   190→  const totalGastos = useMemo(() =>
   191→   191→    gastos.reduce((s, g) => s + (parseFloat(g.amount) || 0), 0),
   192→   192→    [gastos]
   193→   193→  );
   194→   194→  const ticketsNum = parseFloat(tickets) || 0;
   195→   195→  const sobranteNum = parseFloat(sobrante) || 0;
   196→   196→  // PRODUCCION = efectivo real contado por el ayudante + caja comun + sobrante ajuste manual
   197→   197→  const production = totalEfectivoReal + totalCajaComunMonto + sobranteNum;
   198→   198→  // ENTREGA AYUDANTE = efectivo real + sobrante - gastos (la caja común NO pasa por el ayudante)
   199→   199→  const entregaAyudante = totalEfectivoReal + sobranteNum - totalGastos;
   200→   200→  // Lógica dual: si hay caja común se descuentan tickets, si no (pre-Jun 2026) entrega compañía = 0
   201→   201→  const entregaCompania = totalCajaComunMonto > 0 ? totalCajaComunMonto - ticketsNum : 0;
   202→   202→
   203→   203→  const addGasto = () => setGastos(prev => [...prev, { description: '', amount: '0' }]);
   204→   204→  const removeGasto = (i: number) => setGastos(prev => prev.filter((_, idx) => idx !== i));
   205→   205→  const updateGasto = (i: number, field: keyof GastoItem, value: string) => {

   207→   207→      const next = [...prev];
   208→   208→      next[i] = { ...next[i], [field]: value };
   209→   209→      return next;
   210→   210→    });
   211→   211→  };
   212→   212→
   213→   213→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   214→   214→    const file = e.target.files?.[0];
   215→   215→    if (!file || !file.type.startsWith('image/')) return;
   216→   216→    e.target.value = '';
   217→   217→    // Comprimir imagen antes de almacenar (max ~200KB base64)
   218→   218→    const img = new Image();
   219→   219→    const url = URL.createObjectURL(file);
   220→   220→    img.onload = () => {
   221→   221→      URL.revokeObjectURL(url);
   222→   222→      const MAX_DIM = 1200;
   223→   223→      const QUALITY = 0.6;
   224→   224→      let w = img.width, h = img.height;
   225→   225→      if (w > MAX_DIM || h > MAX_DIM) {
   226→   226→        if (w > h) { h = Math.round(h * MAX_DIM / w); w = MAX_DIM; }
   227→   227→        else { w = Math.round(w * MAX_DIM / h); h = MAX_DIM; }
   228→   228→      }
   229→   229→      const canvas = document.createElement('canvas');
   230→   230→      canvas.width = w; canvas.height = h;
   231→   231→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h);
   232→   232→      setFotoPreview(canvas.toDataURL('image/jpeg', QUALITY));
   233→   233→    };
   234→   234→    img.onerror = () => URL.revokeObjectURL(url);
   235→   235→    img.src = url;
   236→   236→  };
   237→   237→
   238→   238→  const handleSaveOffline = () => {
   239→   239→    const errs: string[] = [];
   240→   240→    if (!km.trim()) errs.push('Kilometraje es obligatorio');
   241→   241→    if (!fotoPreview) errs.push('Foto del cuaderno es obligatoria');
   242→   242→    if (frecuencias.length === 0) errs.push('No hay frecuencias cerradas');
   243→   243→    // Gastos pueden ser $0 si no hubo (ej. unidad parada)
   244→   244→    if (errs.length > 0) {
   245→   245→      setErrors(errs);
   246→   246→      return;
   247→   247→    }
   248→   248→    setErrors([]);
   249→   249→    setConfirmSave(true);
   250→   250→  };
   251→   251→
   252→   252→  const handleConfirmSave = async () => {
   253→   253→    setConfirmSave(false);
   254→   254→    setSaving(true);
   255→   255→
   256→   256→    try {
   257→   257→      // Build trips from frecuencias
   258→   258→      const trips = frecuencias.map(f => ({
   259→   259→        routeFrom: f.direccion === 'ida' ? 'Loja' : f.ruta.split(' - ')[1]?.trim() || 'Loja',
   260→   260→        routeTo: f.direccion === 'ida' ? f.ruta.split(' - ')[1]?.trim() || 'Vilcabamba' : 'Loja',
   261→   261→        time: f.hora,
   262→   262→        income: f.totalRecaudado.toString(),
   263→   263→        efectivoReal: f.efectivoContado.toString(),
   264→   264→        boletos: '0',
   265→   265→        cajaComunPasajeros: f.boletosCaja.toString(),
   266→   266→        cajaComunMonto: (f.cajaComunMonto || 0).toString(),
   267→   267→      }));
   268→   268→
   269→   269→      const body = {
   270→   270→        date: workDate(session),
   271→   271→        km: km,
   272→   272→        conductor: '',
   273→   273→        ayudanteNombre: session.ayudanteNombre,
   274→   274→        vtCode: session.vtCode,
   275→   275→        trips,
   276→   276→        expenses: gastos,
   277→   277→        tickets: tickets || '0',
   278→   278→        cajaComun: totalCajaComunMonto,
   279→   279→        sobrante: sobrante || '0',
   280→   280→        photoUrl: fotoPreview,
   281→   281→      };
   282→   282→
   283→   283→      const fechaTrabajo = workDate(session);
   284→   284→      const isOnline = navigator.onLine;
   285→   285→

   287→   287→        const res = await fetch('/api/records', {
   288→   288→          method: 'POST',
   289→   289→          headers: { 'Content-Type': 'application/json' },
   290→   290→          body: JSON.stringify(body),
   291→   291→        });
   292→   292→        if (res.ok) {
   293→   293→          setSaved(true);
   294→   294→        } else {
   295→   295→          const result = await res.json().catch(() => ({ error: 'Error desconocido' }));
   296→   296→          console.error('Error guardando arqueo en servidor:', result.error);
   297→   297→          // Save offline if server fails
   298→   298→          localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   299→   299→          setGuardadoOffline(true);
   300→   300→          setSaved(true);
   301→   301→        }
   302→   302→      } else {
   303→   303→        // Save to localStorage for later sync
   304→   304→        localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   305→   305→        setGuardadoOffline(true);
   306→   306→        setSaved(true);
   307→   307→      }
   308→   308→    } catch (err) {

   312→   312→        trips: frecuencias.map(f => ({ routeFrom: 'Loja', routeTo: 'Vilcabamba', time: f.hora, income: f.totalRecaudado.toString(), boletos: '0' })),
   313→   313→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   314→   314→      }));
   315→   315→      setGuardadoOffline(true);
   316→   316→      setSaved(true);
   317→   317→    } finally {
   318→   318→      setSaving(false);
   319→   319→    }
   320→   320→  };
   321→   321→
   322→   322→  if (loading) {
   323→   323→    return (
   324→   324→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   325→   325→        <Loader2 className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   326→   326→        <p className="text-[#3A3A3A]">Cargando arqueo general...</p>
   327→   327→      </div>
   328→   328→    );
   329→   329→  }
   330→   330→
   331→   331→  // Saved confirmation screen
   332→   332→  if (saved) {
   333→   333→    return (
   334→   334→      <div className="flex flex-col min-h-[100dvh] bg-[#912D26] text-white">
   335→   335→        <div className="flex-1 flex flex-col items-center justify-center px-6">
   336→   336→          <CheckCircle2 className="w-20 h-20 mb-4" />
   337→   337→          <h1 className="text-2xl font-black mb-2">ARQUEO GENERAL GUARDADO</h1>
   338→   338→          <p className="text-white/80 text-sm mb-1">{session.nombre} — {workDate(session)}</p>
   339→   339→          <p className="text-white/80 text-sm mb-6">{totalVentas} ventas · ${totalIngresosAuto.toFixed(2)} ingresos</p>
   340→   340→
   341→   341→          {guardadoOffline && (
   342→   342→            <div className="bg-yellow-500/20 border border-yellow-400/30 rounded-2xl px-5 py-3 mb-6 text-center">
   343→   343→              <p className="text-yellow-200 text-sm font-bold">Guardado local (sin internet)</p>
   344→   344→              <p className="text-yellow-200/70 text-xs mt-1">Se sincronizará cuando haya conexión</p>
   345→   345→            </div>
   346→   346→          )}
   347→   347→
   348→   348→          <div className="w-full max-w-xs space-y-3">
   349→   349→            {connection.pendingCount > 0 && connection.status !== 'offline' && (
   350→   350→              <button onClick={onGoToSync}
   351→   351→                className="w-full py-4 rounded-2xl bg-white text-[#912D26] font-black text-lg flex items-center justify-center gap-2 active:scale-95">
   352→   352→                <Send className="w-5 h-5" /> SINCRONIZAR {connection.pendingCount} VENTAS
   353→   353→              </button>
   354→   354→            )}
   355→   355→            <button onClick={onSaved}
   356→   356→              className="w-full py-4 rounded-2xl bg-white/20 text-white font-bold text-lg flex items-center justify-center gap-2 active:scale-95">
   357→   357→              FINALIZAR
   358→   358→            </button>
   359→   359→          </div>
   360→   360→        </div>
   361→   361→      </div>
   362→   362→    );
   363→   363→  }
   364→   364→
   365→   365→  // Confirm save dialog
   366→   366→  if (confirmSave) {
   367→   367→    return (
   368→   368→      <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   369→   369→        <div className="flex-1 flex items-center justify-center px-4">
   370→   370→          <div className="bg-white rounded-2xl border border-gray-200 p-6 max-w-md w-full shadow-lg">
   371→   371→            <h2 className="text-lg font-bold text-[#3A3A3A] mb-4">Confirmar Arqueo General</h2>
   372→   372→            <div className="bg-gray-50 rounded-xl p-4 space-y-2 mb-4">
   373→   373→              <div className="flex justify-between text-sm">
   374→   374→                <span className="text-gray-500">Producción</span>
   375→   375→                <span className="font-bold text-[#3A3A3A]">${production.toFixed(2)}</span>
   376→   376→              </div>
   377→   377→              {totalCajaComunPasajeros > 0 && (

   379→   379→                  <span className="text-purple-700">Caja Comun (Ofi.Loja): {totalCajaComunPasajeros} boletos</span>

   382→   382→              )}
   383→   383→              <div className="flex justify-between text-sm">
   384→   384→                <span className="text-gray-500">Total Gastos</span>
   385→   385→                <span className="font-bold text-red-600">${totalGastos.toFixed(2)}</span>
   386→   386→              </div>
   387→   387→              <hr className="border-gray-200" />
   388→   388→              <div className="flex justify-between text-sm">
   389→   389→                <span className="text-gray-600">Entrega Ayudante</span>
   390→   390→                <span className={`font-bold ${entregaAyudante >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   391→   391→                  ${entregaAyudante.toFixed(2)}
   392→   392→                </span>
   393→   393→              </div>
   394→   394→              <div className="flex justify-between text-sm">
   395→   395→                <span className="text-gray-600">Entrega Compañía</span>
   396→   396→                <span className={`font-bold ${entregaCompania >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   397→   397→                  ${entregaCompania.toFixed(2)}
   398→   398→                </span>
