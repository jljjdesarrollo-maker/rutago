   754→// ─── Funciones de precio con soporte por dirección ───
   755→//
   756→// Lógica de precios (tipo semántico asignado por el API):
   757→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   758→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   759→//
   760→// El API asigna dirección según el origen:
   761→//   Si origen = Loja →  "ida" (alejándose)
   762→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   763→//
   764→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   765→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   766→
   767→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   768→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   769→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   770→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   771→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   772→  return false;
   773→}
   774→
   775→// Obtener precio considerando la dirección del viaje
   776→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   777→  if (viajaHaciaLoja(ruta, tipo)) {
   778→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   779→    if (ruta === 'Loja - El Tambo') {
   780→      const tamboVuelta = preciosVueltaElTambo[parada];
   781→      if (tamboVuelta) return tamboVuelta;
   782→    }
   783→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   784→    if (ruta === 'Loja - Yangana') {
   785→      const yangVuelta = preciosVueltaYangana[parada];
   786→      if (yangVuelta) return yangVuelta;
   787→    }
   788→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   789→    if (ruta === 'Loja - Vilcabamba') {
   790→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   791→      if (vilcVuelta) return vilcVuelta;
   792→    }
   793→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   794→    if (ruta === 'Loja - La Elvira') {
   795→      const elviraVuelta = preciosVueltaLaElvira[parada];
   796→      if (elviraVuelta) return elviraVuelta;
   797→    }
   798→    const vuelta = preciosVuelta[parada];
   799→    if (vuelta) return vuelta;
   800→    // Si la parada no tiene precio de vuelta, buscar en precios base
   801→  }
   802→  // Buscar primero en ida (base), luego en vuelta como fallback
   803→  const ida = preciosIda[parada];
   804→  if (ida) return ida;
   805→  const vuelta = preciosVuelta[parada];
   806→  return vuelta || null;
   807→}

   812→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   813→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   814→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   815→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   816→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   817→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   818→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   819→  let effectiveRuta = ruta;
   820→  let effectiveTipo = tipo;
   821→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   822→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   823→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   824→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   825→    const baseName = ruta.replace(' - Loja', '');
   826→    effectiveRuta = `Loja - ${baseName}`;
   827→    // NO invertir effectiveTipo — ya viene correcto del API
   828→  }
   829→
   830→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   831→  if (!rutaConfig) {
   832→    // Fallback: try original ruta
   833→    const origConfig = RUTA_PARADAS[ruta];
   834→    if (!origConfig) return [];
   835→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   836→    return paradas.map(parada => {
   837→      const precio = getPrecio(parada, ruta, tipo);
   838→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   839→    });
   840→  }
   841→
   842→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   843→  return paradas.map(parada => {
   844→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   845→    return {
   846→      parada,
   847→      normal: precio?.normal ?? 0,
   848→      media: precio?.media ?? 0,
   849→    };
   850→  });
   851→}

   853→// Normalizar ruta y tipo para rutas "X - Loja"
   854→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   855→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   856→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   857→    // El API ya asigna tipo correcto, NO invertir
   858→    const baseName = ruta.replace(' - Loja', '');
   859→    return { ruta: `Loja - ${baseName}`, tipo };
   860→  }
   861→  return { ruta, tipo };
   862→}

   864→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   865→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   866→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   867→  const precio = getPrecio(parada, nr, nt);
   868→  if (!precio) return TARIFA_MINIMA;
   869→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   870→}

   877→// Buscar la ruta más cercana dado un nombre de ruta parcial
   878→export function matchRuta(rutaNombre: string): string {
   879→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   880→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   881→  if (RUTA_PARADAS[normalized]) return normalized;
   882→
   883→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   884→  for (const key of Object.keys(RUTA_PARADAS)) {
   885→    const keyParts = key.split(' - ');
   886→    if (parts.length >= 2 && keyParts.length >= 2) {
   887→      const match = (
   888→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   889→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   890→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   891→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   892→      );
   893→      if (match) return key;
   894→    }
   895→  }
   896→  return rutaNombre;
   897→}
