   541→   541→// ─── Funciones de precio con soporte por dirección ───
   542→   542→//
   543→   543→// Lógica de precios (tipo semántico asignado por el API):
   544→   544→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   545→   545→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   546→   546→//
   547→   547→// El API asigna dirección según el origen:
   548→   548→//   Si origen = Loja →  "ida" (alejándose)
   549→   549→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   550→   550→//
   551→   551→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   552→   552→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   553→   553→
   554→   554→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   555→   555→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   556→   556→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   557→   557→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   558→   558→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   559→   559→  return false;
   560→   560→}

   562→   562→// Obtener precio considerando la dirección del viaje
   563→   563→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   564→   564→  if (viajaHaciaLoja(ruta, tipo)) {
   565→   565→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   566→   566→    if (ruta === 'Loja - El Tambo') {
   567→   567→      const tamboVuelta = preciosVueltaElTambo[parada];
   568→   568→      if (tamboVuelta) return tamboVuelta;
   569→   569→    }
   570→   570→    const vuelta = preciosVuelta[parada];
   571→   571→    if (vuelta) return vuelta;
   572→   572→    // Si la parada no tiene precio de vuelta, buscar en precios base
   573→   573→  }
   574→   574→  // Buscar primero en ida (base), luego en vuelta como fallback
   575→   575→  const ida = preciosIda[parada];
   576→   576→  if (ida) return ida;
   577→   577→  const vuelta = preciosVuelta[parada];
   578→   578→  return vuelta || null;
   579→   579→}

   584→   584→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   585→   585→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   586→   586→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   587→   587→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   588→   588→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   589→   589→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   590→   590→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   591→   591→  let effectiveRuta = ruta;
   592→   592→  let effectiveTipo = tipo;
   593→   593→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   594→   594→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   595→   595→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   596→   596→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   597→   597→    const baseName = ruta.replace(' - Loja', '');
   598→   598→    effectiveRuta = `Loja - ${baseName}`;
   599→   599→    // NO invertir effectiveTipo — ya viene correcto del API
   600→   600→  }
   601→   601→
   602→   602→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   603→   603→  if (!rutaConfig) {
   604→   604→    // Fallback: try original ruta
   605→   605→    const origConfig = RUTA_PARADAS[ruta];
   606→   606→    if (!origConfig) return [];
   607→   607→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   608→   608→    return paradas.map(parada => {
   609→   609→      const precio = getPrecio(parada, ruta, tipo);
   610→   610→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   611→   611→    });
   612→   612→  }
   613→   613→
   614→   614→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   615→   615→  return paradas.map(parada => {
   616→   616→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   617→   617→    return {
   618→   618→      parada,
   619→   619→      normal: precio?.normal ?? 0,
   620→   620→      media: precio?.media ?? 0,
   621→   621→    };
   622→   622→  });
   623→   623→}

   625→   625→// Normalizar ruta y tipo para rutas "X - Loja"
   626→   626→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   627→   627→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   628→   628→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   629→   629→    // El API ya asigna tipo correcto, NO invertir
   630→   630→    const baseName = ruta.replace(' - Loja', '');
   631→   631→    return { ruta: `Loja - ${baseName}`, tipo };
   632→   632→  }
   633→   633→  return { ruta, tipo };
   634→   634→}

   636→   636→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   637→   637→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   638→   638→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   639→   639→  const precio = getPrecio(parada, nr, nt);
   640→   640→  if (!precio) return TARIFA_MINIMA;
   641→   641→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   642→   642→}

   649→   649→// Buscar la ruta más cercana dado un nombre de ruta parcial
   650→   650→export function matchRuta(rutaNombre: string): string {
   651→   651→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   652→   652→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   653→   653→  if (RUTA_PARADAS[normalized]) return normalized;
   654→   654→
   655→   655→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   656→   656→  for (const key of Object.keys(RUTA_PARADAS)) {
   657→   657→    const keyParts = key.split(' - ');
   658→   658→    if (parts.length >= 2 && keyParts.length >= 2) {
   659→   659→      const match = (
   660→   660→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   661→   661→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   662→   662→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   663→   663→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   664→   664→      );
   665→   665→      if (match) return key;
   666→   666→    }
   667→   667→  }
   668→   668→  return rutaNombre;
   669→   669→}
