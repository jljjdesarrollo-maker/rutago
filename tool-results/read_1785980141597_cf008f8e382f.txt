     1→     1→'use client';
     2→     2→
     3→     3→import { useState, useMemo, useRef, useEffect } from 'react';
     4→     4→import { ArrowLeft, Save, Plus, Trash2, Camera, X, Loader2, AlertTriangle } from 'lucide-react';
     5→     5→import { Button } from '@/components/ui/button';
     6→     6→import { Input } from '@/components/ui/input';
     7→     7→import { Label } from '@/components/ui/label';
     8→     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     9→     9→import { Separator } from '@/components/ui/separator';
    10→    10→import {
    11→    11→  AlertDialog,
    12→    12→  AlertDialogContent,
    13→    13→  AlertDialogHeader,
    14→    14→  AlertDialogTitle,
    15→    15→  AlertDialogDescription,
    16→    16→  AlertDialogFooter,
    17→    17→  AlertDialogCancel,
    18→    18→  AlertDialogAction,
    19→    19→} from '@/components/ui/alert-dialog';
    20→    20→import { type RecordFormData, type TripData, type ExpenseData, createEmptyFormData, num, getCurrentConductor, getCurrentAyudante } from './types';
    21→    21→import { routes, getRouteLabel, getTimesForRoute } from '@/lib/routes';
    22→    22→
    23→    23→interface RecordFormProps {
    24→    24→  saving: boolean;
    25→    25→  error: string | null;
    26→    26→  onBack: () => void;
    27→    27→  onSave: (data: RecordFormData, photoBase64: string | null) => void;
    28→    28→}
    29→    29→
    30→    30→export function RecordForm({ saving, error, onBack, onSave }: RecordFormProps) {
    31→    31→  const [form, setForm] = useState<RecordFormData>(createEmptyFormData());
    32→    32→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
    33→    33→  const [validationErrors, setValidationErrors] = useState<string[]>([]);
    34→    34→  const [showConfirm, setShowConfirm] = useState(false);
    35→    35→  const [loadedDefaults, setLoadedDefaults] = useState(false);
    36→    36→
    37→    37→  // Load current conductor and ayudante from DB
    38→    38→  useEffect(() => {
    39→    39→    if (loadedDefaults) return;
    40→    40→    Promise.all([getCurrentConductor(), getCurrentAyudante()]).then(([conductor, ayudante]) => {
    41→    41→      if (conductor || ayudante) {
    42→    42→        setForm(prev => ({
    43→    43→          ...prev,
    44→    44→          ...(conductor ? { conductor } : {}),
    45→    45→          ...(ayudante ? { ayudanteNombre: ayudante } : {}),
    46→    46→        }));
    47→    47→      }
    48→    48→      setLoadedDefaults(true);
    49→    49→    });
    50→    50→  }, [loadedDefaults]);
    51→    51→  const fileInputRef = useRef<HTMLInputElement>(null);
    52→    52→
    53→    53→  const updateField = (field: keyof RecordFormData, value: string) => {
    54→    54→    setForm((prev) => ({ ...prev, [field]: value }));
    55→    55→  };
    56→    56→
    57→    57→  const addTrip = () => {
    58→    58→    setForm((prev) => ({
    59→    59→      ...prev,
    60→    60→      trips: [...prev.trips, { routeFrom: 'Loja', routeTo: 'Vilcabamba', time: '', income: '0', boletos: '0' }],
    61→    61→    }));
    62→    62→  };
    63→    63→
    64→    64→  const removeTrip = (index: number) => {
    65→    65→    setForm((prev) => ({
    66→    66→      ...prev,
    67→    67→      trips: prev.trips.filter((_, i) => i !== index),
    68→    68→    }));
    69→    69→  };
    70→    70→
    71→    71→  const updateTrip = (index: number, field: keyof TripData, value: string) => {
    72→    72→    setForm((prev) => {
    73→    73→      const newTrips = [...prev.trips];
    74→    74→      newTrips[index] = { ...newTrips[index], [field]: value };
    75→    75→      return { ...prev, trips: newTrips };
    76→    76→    });
    77→    77→  };
    78→    78→
    79→    79→  const handleRouteSelect = (index: number, routeLabel: string) => {
    80→    80→    const route = routes.find((r) => getRouteLabel(r.from, r.to) === routeLabel);
    81→    81→    if (route) {
    82→    82→      setForm((prev) => {
    83→    83→        const newTrips = [...prev.trips];
    84→    84→        newTrips[index] = { ...newTrips[index], routeFrom: route.from, routeTo: route.to, time: '' };
    85→    85→        return { ...prev, trips: newTrips };
    86→    86→      });
    87→    87→    }
    88→    88→  };
    89→    89→
    90→    90→  const addExpense = () => {
    91→    91→    setForm((prev) => ({
    92→    92→      ...prev,
    93→    93→      expenses: [...prev.expenses, { description: '', amount: '0' }],
    94→    94→    }));
    95→    95→  };
    96→    96→
    97→    97→  const removeExpense = (index: number) => {
    98→    98→    setForm((prev) => ({
    99→    99→      ...prev,
   100→   100→      expenses: prev.expenses.filter((_, i) => i !== index),
   101→   101→    }));
   102→   102→  };
   103→   103→
   104→   104→  const updateExpense = (index: number, field: keyof ExpenseData, value: string) => {
   105→   105→    setForm((prev) => {
   106→   106→      const newExp = [...prev.expenses];
   107→   107→      newExp[index] = { ...newExp[index], [field]: value };
   108→   108→      return { ...prev, expenses: newExp };
   109→   109→    });
   110→   110→  };
   111→   111→
   112→   112→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   113→   113→    const file = e.target.files?.[0];
   114→   114→    if (!file || !file.type.startsWith('image/')) return;
   115→   115→    const reader = new FileReader();
   116→   116→    reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   117→   117→    reader.readAsDataURL(file);
   118→   118→    e.target.value = '';
   119→   119→  };
   120→   120→
   121→   121→  const totals = useMemo(() => {
   122→   122→    const tripIncome = form.trips.reduce((s, t) => s + num(t.income), 0);
   123→   123→    const cajaComun = form.trips.reduce((s, t) => s + num(t.boletos), 0);
   124→   124→    const sobrante = num(form.sobrante);
   125→   125→    const production = tripIncome + sobrante;
   126→   126→    const totalGastos = form.expenses.reduce((s, e) => s + num(e.amount), 0);
   127→   127→    const tickets = num(form.tickets);
   128→   128→    const entregaAyudante = production - totalGastos;
   129→   129→    const entregaCompania = cajaComun - tickets;
   130→   130→    return { production, cajaComun, sobrante, totalGastos, tickets, entregaAyudante, entregaCompania };
   131→   131→  }, [form]);
   132→   132→
   133→   133→  const handleSave = () => {
   134→   134→    // Validaciones obligatorias
   135→   135→    const errors: string[] = [];
   136→   136→    if (!form.km.trim()) errors.push('Kilometraje es obligatorio');
   137→   137→    if (!photoPreview) errors.push('Foto del cuaderno es obligatoria');
   138→   138→    if (totals.totalGastos <= 0) errors.push('Total de gastos debe ser mayor a 0');
   139→   139→    if (totals.production <= 0) errors.push('Produccion total debe ser mayor a 0');
   140→   140→    if (errors.length > 0) {
   141→   141→      setValidationErrors(errors);
   142→   142→      return;
   143→   143→    }
   144→   144→    setValidationErrors([]);
   145→   145→    setShowConfirm(true);
   146→   146→  };
   147→   147→
   148→   148→  const handleConfirmSave = () => {
   149→   149→    setShowConfirm(false);
   150→   150→    onSave(form, photoPreview);
   151→   151→  };
   152→   152→
   153→   153→  const allRouteOptions = routes.map((r) => getRouteLabel(r.from, r.to));
   154→   154→
   155→   155→  const isOddTrip = (index: number) => (index + 1) % 2 !== 0;
   156→   156→
   157→   157→  return (
   158→   158→    <div className="flex flex-col min-h-[100dvh] bg-[#FAFAFA]">
   159→   159→      <header className="sticky top-0 z-10 bg-white border-b border-[#D6D6D6] px-4 py-3 flex items-center justify-between">
   160→   160→        <div className="flex items-center gap-3">
   161→   161→          <Button variant="ghost" size="icon" onClick={onBack} className="rounded-xl text-[#3A3A3A]">
   162→   162→            <ArrowLeft className="w-5 h-5" />
   163→   163→          </Button>
   164→   164→          <div>
   165→   165→            <h1 className="text-lg font-semibold text-[#3A3A3A]">Registro del Dia</h1>
   166→   166→          </div>
   167→   167→        </div>
   168→   168→        <Button
   169→   169→          onClick={handleSave}
   170→   170→          disabled={saving}
   171→   171→          className="h-10 px-4 rounded-xl bg-[#912D26] hover:bg-[#7A2520] text-white font-semibold text-sm"
   172→   172→        >
   173→   173→          {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Save className="w-4 h-4 mr-1" />}
   174→   174→          {saving ? 'Guardando...' : 'Guardar'}
   175→   175→        </Button>
   176→   176→      </header>
   177→   177→
   178→   178→      <main className="flex-1 overflow-y-auto pb-8">
   179→   179→        <div className="px-4 py-4 space-y-4">
   180→   180→          {error && (
   181→   181→            <div className="p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm">
   182→   182→              {error}
   183→   183→            </div>
   184→   184→          )}
   185→   185→          {validationErrors.length > 0 && (
   186→   186→            <div className="p-3 rounded-xl bg-[#912D26]/10 border border-[#912D26]/30 text-[#912D26] text-sm space-y-1">
   187→   187→              <p className="font-semibold flex items-center gap-1.5"><AlertTriangle className="w-4 h-4" /> Campos obligatorios:</p>
   188→   188→              {validationErrors.map((e, i) => <p key={i}>- {e}</p>)}
   189→   189→            </div>
   190→   190→          )}
   191→   191→
   192→   192→          {/* 1. Cabecera - Fechas y Km */}
   193→   193→          <Card className="rounded-2xl border border-[#D6D6D6] bg-white">
   194→   194→            <CardContent className="p-4 space-y-3">
   195→   195→              <div className="grid grid-cols-2 gap-3">
   196→   196→                <div className="space-y-1.5">
   197→   197→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Fecha de Registro</Label>
   198→   198→                  <Input
   199→   199→                    type="text"
   200→   200→                    value={new Date().toLocaleDateString('es-EC', { day: '2-digit', month: '2-digit', year: 'numeric' })}
   201→   201→                    readOnly
   202→   202→                    className="rounded-xl h-11 bg-[#F5F5F5] text-[#3A3A3A]/50 cursor-not-allowed border-[#D6D6D6]"
   203→   203→                  />
   204→   204→                </div>
   205→   205→                <div className="space-y-1.5">
   206→   206→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Fecha de Operacion</Label>
   207→   207→                  <Input
   208→   208→                    type="date"
   209→   209→                    value={form.date}
   210→   210→                    onChange={(e) => updateField('date', e.target.value)}
   211→   211→                    className="rounded-xl h-11 border-[#D6D6D6]"
   212→   212→                  />
   213→   213→                </div>
   214→   214→              </div>
   215→   215→              <div className="space-y-1.5">
   216→   216→                <Label className="text-xs font-medium text-[#3A3A3A]/60 flex items-center gap-1">
   217→   217→                  Kilometraje <span className="text-[#912D26]">*</span>
   218→   218→                </Label>
   219→   219→                <Input
   220→   220→                  type="text"
   221→   221→                  inputMode="numeric"
   222→   222→                  placeholder="877604"
   223→   223→                  value={form.km}
   224→   224→                  onChange={(e) => updateField('km', e.target.value)}
   225→   225→                  className={`rounded-xl h-11 border-[#D6D6D6] ${!form.km.trim() && validationErrors.length > 0 ? 'border-[#912D26] bg-[#912D26]/5' : ''}`}
   226→   226→                />
   227→   227→              </div>
   228→   228→              <div className="grid grid-cols-2 gap-3">
   229→   229→                <div className="space-y-1.5">
   230→   230→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Conductor</Label>
   231→   231→                  <Input
   232→   232→                    placeholder="Nombre del conductor"
   233→   233→                    value={form.conductor}
   234→   234→                    onChange={(e) => updateField('conductor', e.target.value)}
   235→   235→                    className="rounded-xl h-11 border-[#D6D6D6]"
   236→   236→                  />
   237→   237→                </div>
   238→   238→                <div className="space-y-1.5">
   239→   239→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Ayudante</Label>
   240→   240→                  <Input
   241→   241→                    placeholder="Nombre del ayudante"
   242→   242→                    value={form.ayudanteNombre}
   243→   243→                    onChange={(e) => updateField('ayudanteNombre', e.target.value)}
   244→   244→                    className="rounded-xl h-11 border-[#D6D6D6]"
   245→   245→                  />
   246→   246→                </div>
   247→   247→              </div>
   248→   248→
   249→   249→              {/* Foto obligatoria */}
   250→   250→              <div>
   251→   251→                <Label className="text-xs font-medium text-[#3A3A3A]/60 flex items-center gap-1">
   252→   252→                  Foto del Cuaderno <span className="text-[#912D26]">*</span>
   253→   253→                </Label>
   254→   254→                <div className="mt-1.5">
   255→   255→                  {photoPreview ? (
   256→   256→                    <div className="relative w-20 h-20 rounded-xl overflow-hidden border-2 border-[#912D26]">
   257→   257→                      <img src={photoPreview} alt="Foto" className="w-full h-full object-cover" />
   258→   258→                      <button
   259→   259→                        onClick={() => setPhotoPreview(null)}
   260→   260→                        className="absolute top-0 right-0 w-5 h-5 bg-[#912D26] text-white rounded-bl-lg flex items-center justify-center"
   261→   261→                      >
   262→   262→                        <X className="w-3 h-3" />
   263→   263→                      </button>
   264→   264→                    </div>
   265→   265→                  ) : (
   266→   266→                    <Button
   267→   267→                      type="button"
   268→   268→                      variant="outline"
   269→   269→                      className="h-10 px-4 rounded-xl text-xs border-[#D6D6D6] text-[#3A3A3A]"
   270→   270→                      onClick={() => fileInputRef.current?.click()}
   271→   271→                    >
   272→   272→                      <Camera className="w-4 h-4 mr-1" />
   273→   273→                      Tomar/Adjuntar Foto
   274→   274→                    </Button>
   275→   275→                  )}
   276→   276→                </div>
   277→   277→              </div>
   278→   278→            </CardContent>
   279→   279→          </Card>
   280→   280→
   281→   281→          {/* 2. Frecuencias */}
   282→   282→          <Card className="rounded-2xl border border-[#D6D6D6] bg-white">
   283→   283→            <CardHeader className="pb-2 pt-4 px-4">
   284→   284→              <div className="flex items-center justify-between">
   285→   285→                <CardTitle className="text-base font-semibold text-[#3A3A3A]">Frecuencias del Dia</CardTitle>
   286→   286→                <Button onClick={addTrip} size="sm" variant="outline" className="h-8 rounded-lg text-xs border-dashed border-[#D6D6D6] text-[#3A3A3A]">
   287→   287→                  <Plus className="w-3.5 h-3.5 mr-1" /> Agregar
   288→   288→                </Button>
   289→   289→              </div>
   290→   290→            </CardHeader>
   291→   291→            <CardContent className="px-4 pb-4 space-y-3">
   292→   292→              {form.trips.length === 0 && (
   293→   293→                <div className="text-center py-6 text-[#3A3A3A]/40 text-sm">
   294→   294→                  <p>Toca &quot;Agregar&quot; para registrar frecuencias</p>
   295→   295→                </div>
   296→   296→              )}
   297→   297→              {form.trips.map((trip, index) => {
   298→   298→                const tripTimes = getTimesForRoute(trip.routeFrom, trip.routeTo);
   299→   299→                const odd = isOddTrip(index);
   300→   300→                return (
   301→   301→                  <div key={index} className={`rounded-xl p-3 space-y-2 border-l-4 ${odd ? 'border-l-[#912D26] bg-[#912D26]/5' : 'border-l-[#3A3A3A] bg-[#3A3A3A]/5'}`}>
   302→   302→                    <div className="flex items-center justify-between">
   303→   303→                      <span className={`text-xs font-bold ${odd ? 'text-[#912D26]' : 'text-[#3A3A3A]'}`}>
   304→   304→                        Frecuencia #{index + 1} {odd ? '(Loja → Vilca)' : '(Vilca → Loja)'}
   305→   305→                      </span>
   306→   306→                      <Button onClick={() => removeTrip(index)} variant="ghost" size="icon" className="h-7 w-7 text-[#3A3A3A]/40 hover:text-red-500">
   307→   307→                        <Trash2 className="w-3.5 h-3.5" />
   308→   308→                      </Button>
   309→   309→                    </div>
   310→   310→
   311→   311→                    <div className="space-y-1">
   312→   312→                      <Label className="text-[10px] text-[#3A3A3A]/50">Ruta</Label>
   313→   313→                      <select
   314→   314→                        value={getRouteLabel(trip.routeFrom, trip.routeTo)}
   315→   315→                        onChange={(e) => handleRouteSelect(index, e.target.value)}
   316→   316→                        className="w-full h-9 rounded-lg border border-[#D6D6D6] bg-white px-2 text-sm text-[#3A3A3A]"
   317→   317→                      >
   318→   318→                        {allRouteOptions.map((opt) => (
   319→   319→                          <option key={opt} value={opt}>{opt}</option>
   320→   320→                        ))}
   321→   321→                      </select>
   322→   322→                    </div>
   323→   323→
   324→   324→                    <div className="space-y-1">
   325→   325→                      <Label className="text-[10px] text-[#3A3A3A]/50">Hora de Salida</Label>
   326→   326→                      {tripTimes.length > 0 ? (
   327→   327→                        <select
   328→   328→                          value={trip.time}
   329→   329→                          onChange={(e) => updateTrip(index, 'time', e.target.value)}
   330→   330→                          className="w-full h-9 rounded-lg border border-[#D6D6D6] bg-white px-2 text-sm text-[#3A3A3A]"
   331→   331→                        >
   332→   332→                          <option value="">-- Seleccionar --</option>
   333→   333→                          {tripTimes.map((t) => (
   334→   334→                            <option key={t} value={t}>{t}</option>
   335→   335→                          ))}
   336→   336→                        </select>
   337→   337→                      ) : (
   338→   338→                        <Input
   339→   339→                          placeholder="HH:MM"
   340→   340→                          value={trip.time}
   341→   341→                          onChange={(e) => updateTrip(index, 'time', e.target.value)}
   342→   342→                          className="h-9 text-sm rounded-lg border-[#D6D6D6]"
   343→   343→                        />
   344→   344→                      )}
   345→   345→                    </div>
   346→   346→
   347→   347→                    <div className="grid grid-cols-2 gap-2">
   348→   348→                      <div className="space-y-1">
   349→   349→                        <Label className="text-[10px] text-[#3A3A3A]/50">Ingreso (Produccion)</Label>
   350→   350→                        <Input
   351→   351→                          type="number"
   352→   352→                          inputMode="decimal"
   353→   353→                          value={trip.income}
   354→   354→                          onChange={(e) => updateTrip(index, 'income', e.target.value)}
   355→   355→                          className="h-9 text-sm rounded-lg border-[#D6D6D6]"
   356→   356→                        />
   357→   357→                      </div>
   358→   358→                      <div className="space-y-1">
   359→   359→                        <Label className="text-[10px] text-[#3A3A3A]/50">Boletos (Caja Comun)</Label>
   360→   360→                        <Input
   361→   361→                          type="number"
   362→   362→                          inputMode="decimal"
   363→   363→                          value={trip.boletos}
   364→   364→                          onChange={(e) => updateTrip(index, 'boletos', e.target.value)}
   365→   365→                          className="h-9 text-sm rounded-lg border-[#D6D6D6]"
   366→   366→                        />
   367→   367→                      </div>
   368→   368→                    </div>
   369→   369→                  </div>
   370→   370→                );
   371→   371→              })}
   372→   372→
   373→   373→              {/* Sobrante / Ajuste */}
   374→   374→              <div className="pt-2 border-t border-[#D6D6D6]">
   375→   375→                <div className="space-y-1">
   376→   376→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Sobrante / Ajuste</Label>
   377→   377→                  <Input
   378→   378→                    type="number"
   379→   379→                    inputMode="decimal"
   380→   380→                    placeholder="0.00"
   381→   381→                    value={form.sobrante}
   382→   382→                    onChange={(e) => updateField('sobrante', e.target.value)}
   383→   383→                    className="h-10 text-sm rounded-lg border-[#D6D6D6]"
   384→   384→                  />
   385→   385→                  <p className="text-[10px] text-[#3A3A3A]/40">Dinero sobrante al contar (positivo o negativo)</p>
   386→   386→                </div>
   387→   387→              </div>
   388→   388→            </CardContent>
   389→   389→          </Card>
   390→   390→
   391→   391→          {/* 3. Gastos */}
   392→   392→          <Card className="rounded-2xl border border-[#D6D6D6] bg-white">
   393→   393→            <CardHeader className="pb-2 pt-4 px-4">
   394→   394→              <div className="flex items-center justify-between">
   395→   395→                <CardTitle className="text-base font-semibold text-red-600">Gastos</CardTitle>
   396→   396→                <Button onClick={addExpense} size="sm" variant="outline" className="h-8 rounded-lg text-xs border-dashed border-[#D6D6D6] text-[#3A3A3A]">
   397→   397→                  <Plus className="w-3.5 h-3.5 mr-1" /> Agregar
   398→   398→                </Button>
   399→   399→              </div>
   400→   400→            </CardHeader>
   401→   401→            <CardContent className="px-4 pb-4 space-y-2">
   402→   402→              {form.expenses.map((exp, index) => (
   403→   403→                <div key={index} className="flex items-center gap-2">
   404→   404→                  <Input
