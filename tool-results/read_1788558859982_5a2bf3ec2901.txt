   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→  return 'yellow';
   128→}
   129→
   130→export function isParadaPrincipal(parada: string): boolean {
   131→  // Paradas principales: no contienen → en el nombre
   132→  return !parada.includes('→');
   133→}

   713→// ─── Funciones de precio con soporte por dirección ───
   714→//
   715→// Lógica de precios (tipo semántico asignado por el API):
   716→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   717→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   718→//
   719→// El API asigna dirección según el origen:
   720→//   Si origen = Loja →  "ida" (alejándose)
   721→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   722→//
   723→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   724→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   725→
   726→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   727→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   728→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   729→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   730→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   731→  return false;
   732→}
   733→
   734→// Obtener precio considerando la dirección del viaje
   735→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   736→  if (viajaHaciaLoja(ruta, tipo)) {
   737→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   738→    if (ruta === 'Loja - El Tambo') {
   739→      const tamboVuelta = preciosVueltaElTambo[parada];
   740→      if (tamboVuelta) return tamboVuelta;
   741→    }
   742→    // Ruta Zahuayco tiene precios de vuelta propios (directos desde Zahuayco)
   743→    if (ruta === 'Loja - Zahuayco') {
   744→      const zahuaycoVuelta = preciosVueltaZahuayco[parada];
   745→      if (zahuaycoVuelta) return zahuaycoVuelta;
   746→    }
   747→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   748→    if (ruta === 'Loja - Yangana') {
   749→      const yangVuelta = preciosVueltaYangana[parada];
   750→      if (yangVuelta) return yangVuelta;
   751→    }
   752→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   753→    if (ruta === 'Loja - Vilcabamba') {
   754→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   755→      if (vilcVuelta) return vilcVuelta;
   756→    }
   757→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   758→    if (ruta === 'Loja - La Elvira') {
   759→      const elviraVuelta = preciosVueltaLaElvira[parada];
   760→      if (elviraVuelta) return elviraVuelta;
   761→    }
   762→    const vuelta = preciosVuelta[parada];
   763→    if (vuelta) return vuelta;
   764→    // Si la parada no tiene precio de vuelta, buscar en precios base
   765→  }
   766→  // Buscar primero en ida (base), luego en vuelta como fallback
   767→  const ida = preciosIda[parada];
   768→  if (ida) return ida;
   769→  const vuelta = preciosVuelta[parada];
   770→  return vuelta || null;
   771→}

   776→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   777→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   778→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   779→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   780→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   781→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   782→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   783→  let effectiveRuta = ruta;
   784→  let effectiveTipo = tipo;
   785→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   786→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   787→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   788→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   789→    const baseName = ruta.replace(' - Loja', '');
   790→    effectiveRuta = `Loja - ${baseName}`;
   791→    // NO invertir effectiveTipo — ya viene correcto del API
   792→  }
   793→
   794→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   795→  if (!rutaConfig) {
   796→    // Fallback: try original ruta
   797→    const origConfig = RUTA_PARADAS[ruta];
   798→    if (!origConfig) return [];
   799→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   800→    return paradas.map(parada => {
   801→      const precio = getPrecio(parada, ruta, tipo);
   802→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   803→    });
   804→  }
   805→
   806→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   807→  return paradas.map(parada => {
   808→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   809→    return {
   810→      parada,
   811→      normal: precio?.normal ?? 0,
   812→      media: precio?.media ?? 0,
   813→    };
   814→  });
   815→}

   817→// Normalizar ruta y tipo para rutas "X - Loja"
   818→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   819→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   820→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   821→    // El API ya asigna tipo correcto, NO invertir
   822→    const baseName = ruta.replace(' - Loja', '');
   823→    return { ruta: `Loja - ${baseName}`, tipo };
   824→  }
   825→  return { ruta, tipo };
   826→}

   828→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   829→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   830→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   831→  const precio = getPrecio(parada, nr, nt);
   832→  if (!precio) return TARIFA_MINIMA;
   833→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   834→}

   841→// Buscar la ruta más cercana dado un nombre de ruta parcial
   842→export function matchRuta(rutaNombre: string): string {
   843→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   844→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   845→  if (RUTA_PARADAS[normalized]) return normalized;
   846→
   847→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   848→  for (const key of Object.keys(RUTA_PARADAS)) {
   849→    const keyParts = key.split(' - ');
   850→    if (parts.length >= 2 && keyParts.length >= 2) {
   851→      const match = (
   852→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   853→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   854→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   855→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   856→      );
   857→      if (match) return key;
   858→    }
   859→  }
   860→  return rutaNombre;
   861→}
