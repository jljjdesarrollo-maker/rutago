     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo, useRef } from 'react';
     4→import { type VTSession, type FrecuenciaEstado } from './types-boletos';
     5→import { getVentasByFrecuencia } from '@/lib/indexeddb';
     6→import { type ConnectionInfo } from '@/hooks/use-connection';
     7→import {
     8→  ChevronLeft, DollarSign, Camera, X, Save, Loader2,
     9→  CheckCircle2, Send, AlertTriangle, Plus, Trash2, ImagePlus, Sparkles
    10→} from 'lucide-react';
    11→
    12→interface Props {
    13→  session: VTSession;
    14→  connection: ConnectionInfo;
    15→  onClose: () => void;
    16→  onGoToSync: () => void;
    17→  onSaved: () => void;
    18→}
    19→
    20→interface FrecuenciaResumen {
    21→  estadoId: string;
    22→  nombre: string;
    23→  ruta: string;
    24→  hora: string;
    25→  direccion: string;
    26→  ventasCount: number;
    27→  totalRecaudado: number;        // What the system says
    28→  efectivoContado: number;        // What the helper actually counted
    29→  diferencia: number;             // efectivoContado - totalRecaudado
    30→  boletosCaja: number;
    31→  cajaComunMonto?: number;
    32→  isIngresoEspecial?: boolean;
    33→  ingresoEspecialNota?: string;
    34→}
    35→
    36→interface GastoItem {
    37→  description: string;
    38→  amount: string;
    39→}
    40→
    41→const today = () => new Date().toISOString().split('T')[0];
    42→
    43→// Work date: use session.fecha (date when shift started, not necessarily today)
    44→function workDate(session: VTSession): string {
    45→  return session.fecha || today();
    46→}
    47→
    48→const GASTOS_DEFAULT: GastoItem[] = [
    49→  { description: 'Chofer', amount: '30' },
    50→  { description: 'Ayudante', amount: '20' },
    51→  { description: 'Diesel', amount: '0' },
    52→  { description: 'Plan Renova', amount: '68' },
    53→];
    54→
    55→export function ArqueoGeneralScreen({ session, connection, onClose, onGoToSync, onSaved }: Props) {
    56→  const [frecuencias, setFrecuencias] = useState<FrecuenciaResumen[]>([]);
    57→  const [loading, setLoading] = useState(true);
    58→  const [km, setKm] = useState('');
    59→  const [gastos, setGastos] = useState<GastoItem[]>(GASTOS_DEFAULT);
    60→  const [tickets, setTickets] = useState('');
    61→  const [sobrante, setSobrante] = useState('');
    62→  const [fotoPreview, setFotoPreview] = useState<string | null>(null);
    63→  const [saving, setSaving] = useState(false);
    64→  const [confirmSave, setConfirmSave] = useState(false);
    65→  const [saved, setSaved] = useState(false);
    66→  const [errors, setErrors] = useState<string[]>([]);
    67→  const [guardadoOffline, setGuardadoOffline] = useState(false);
    68→
    69→  const fileInputRef = useRef<HTMLInputElement>(null);
    70→  const loadData = useCallback(async () => {
    71→    try {
    72→      const fecha = workDate(session);
    73→      // Load estados from localStorage (synchronous, reliable)
    74→      const lsKey = `rg_estados_${session.vtCode}_${fecha}`;
    75→      const raw = localStorage.getItem(lsKey);
    76→      if (!raw) { setLoading(false); return; }
    77→      const estadosLS = JSON.parse(raw) as FrecuenciaEstado[];
    78→      const cerradas = estadosLS.filter(e => e.estado === 'cerrada');
    79→      // Ingresos especiales (no_realizada con monto > 0)
    80→      const especiales = estadosLS.filter(e => e.estado === 'no_realizada' && (e.ingresoEspecialMonto || 0) > 0);
    81→
    82→      const resumenes: FrecuenciaResumen[] = await Promise.all(
    83→        cerradas.map(async (e) => {
    84→          const ventas = await getVentasByFrecuencia(e.estadoId);
    85→          const sistemaTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
    86→          // Use arqueo data if available (what helper actually counted), else system total
    87→          const efectivo = (e as any).arqueoEfectivo ?? sistemaTotal;
    88→          const diff = efectivo - sistemaTotal;
    89→          const ccCount = (e as any).cajaComunCount || 0;
    90→          const ccMonto = (e as any).cajaComunMonto || 0;
    91→          return {
    92→            estadoId: e.estadoId,
    93→            nombre: e.nombre,
    94→            ruta: e.ruta,
    95→            hora: e.hora,
    96→            direccion: e.direccion,
    97→            ventasCount: ventas.length,
    98→            totalRecaudado: sistemaTotal,
    99→            efectivoContado: efectivo,
   100→            diferencia: diff,
   101→            boletosCaja: ccCount,
   102→            cajaComunMonto: ccMonto,
   103→          };
   104→        })
   105→      );
   106→
   107→      // Add ingresos especiales as virtual rows
   108→      especiales.forEach(e => {
   109→        resumenes.push({
   110→          estadoId: e.estadoId,
   111→          nombre: e.nombre,
   112→          ruta: e.ruta,
   113→          hora: e.hora,
   114→          direccion: e.direccion,
   115→          ventasCount: 0,
   116→          totalRecaudado: e.ingresoEspecialMonto || 0,
   117→          efectivoContado: e.ingresoEspecialMonto || 0,
   118→          diferencia: 0,
   119→          boletosCaja: 0,
   120→          cajaComunMonto: 0,
   121→          isIngresoEspecial: true,
   122→          ingresoEspecialNota: e.ingresoEspecialNota,
   123→        });
   124→      });
   125→      setFrecuencias(resumenes);
   126→    } catch (err) {
   127→      console.error('Error cargando resumen:', err);
   128→    } finally {
   129→      setLoading(false);
   130→    }
   131→  }, [session.vtCode]);
   132→
   133→  useEffect(() => { loadData(); }, [loadData]);
   134→
   135→  const totalIngresosAuto = useMemo(() =>
   136→    frecuencias.reduce((s, f) => s + f.totalRecaudado, 0),
   137→    [frecuencias]
   138→  );
   139→  const totalEfectivoReal = useMemo(() =>
   140→    frecuencias.reduce((s, f) => s + f.efectivoContado, 0),
   141→    [frecuencias]
   142→  );
   143→  const totalDiferencia = useMemo(() =>
   144→    frecuencias.reduce((s, f) => s + f.diferencia, 0),
   145→    [frecuencias]
   146→  );
   147→  const totalVentas = useMemo(() =>
   148→    frecuencias.reduce((s, f) => s + f.ventasCount, 0),
   149→    [frecuencias]
   150→  );
   151→  const totalCajaComunPasajeros = useMemo(() =>
   152→    frecuencias.reduce((s, f) => s + f.boletosCaja, 0),
   153→    [frecuencias]
   154→  );
   155→  const totalCajaComunMonto = useMemo(() =>
   156→    frecuencias.reduce((s, f) => s + (f.cajaComunMonto || 0), 0),
   157→    [frecuencias]
   158→  );
   159→  const totalGastos = useMemo(() =>
   160→    gastos.reduce((s, g) => s + (parseFloat(g.amount) || 0), 0),
   161→    [gastos]
   162→  );
   163→  const ticketsNum = parseFloat(tickets) || 0;
   164→  const sobranteNum = parseFloat(sobrante) || 0;
   165→  // PRODUCCION = efectivo real contado por el ayudante + caja comun + sobrante ajuste manual
   166→  const production = totalEfectivoReal + totalCajaComunMonto + sobranteNum;
   167→  // ENTREGA AYUDANTE = efectivo real + sobrante - gastos (la caja común NO pasa por el ayudante)
   168→  const entregaAyudante = totalEfectivoReal + sobranteNum - totalGastos;
   169→  // Lógica dual: si hay caja común se descuentan tickets, si no (pre-Jun 2026) entrega compañía = 0
   170→  const entregaCompania = totalCajaComunMonto > 0 ? totalCajaComunMonto - ticketsNum : 0;
   171→
   172→  const addGasto = () => setGastos(prev => [...prev, { description: '', amount: '0' }]);
   173→  const removeGasto = (i: number) => setGastos(prev => prev.filter((_, idx) => idx !== i));
   174→  const updateGasto = (i: number, field: keyof GastoItem, value: string) => {
   175→    setGastos(prev => {
   176→      const next = [...prev];
   177→      next[i] = { ...next[i], [field]: value };
   178→      return next;
   179→    });
   180→  };
   181→
   182→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   183→    const file = e.target.files?.[0];
   184→    if (!file || !file.type.startsWith('image/')) return;
   185→    e.target.value = '';
   186→    // Comprimir imagen antes de almacenar (max ~200KB base64)
   187→    const img = new Image();
   188→    const url = URL.createObjectURL(file);
   189→    img.onload = () => {
   190→      URL.revokeObjectURL(url);
   191→      const MAX_DIM = 1200;
   192→      const QUALITY = 0.6;
   193→      let w = img.width, h = img.height;
   194→      if (w > MAX_DIM || h > MAX_DIM) {
   195→        if (w > h) { h = Math.round(h * MAX_DIM / w); w = MAX_DIM; }
   196→        else { w = Math.round(w * MAX_DIM / h); h = MAX_DIM; }
   197→      }
   198→      const canvas = document.createElement('canvas');
   199→      canvas.width = w; canvas.height = h;
   200→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h);
   201→      setFotoPreview(canvas.toDataURL('image/jpeg', QUALITY));
   202→    };
   203→    img.onerror = () => URL.revokeObjectURL(url);
   204→    img.src = url;
   205→  };
   206→
   207→  const handleSaveOffline = () => {
   208→    const errs: string[] = [];
   209→    if (!km.trim()) errs.push('Kilometraje es obligatorio');
   210→    if (!fotoPreview) errs.push('Foto del cuaderno es obligatoria');
   211→    if (frecuencias.length === 0) errs.push('No hay frecuencias cerradas');
   212→    // Gastos pueden ser $0 si no hubo (ej. unidad parada)
   213→    if (errs.length > 0) {
   214→      setErrors(errs);
   215→      return;
   216→    }
   217→    setErrors([]);
   218→    setConfirmSave(true);
   219→  };
   220→
   221→  const handleConfirmSave = async () => {
   222→    setConfirmSave(false);
   223→    setSaving(true);
   224→
   225→    try {
   226→      // Build trips from frecuencias
   227→      const trips = frecuencias.map(f => ({
   228→        routeFrom: f.direccion === 'ida' ? 'Loja' : f.ruta.split(' - ')[1]?.trim() || 'Loja',
   229→        routeTo: f.direccion === 'ida' ? f.ruta.split(' - ')[1]?.trim() || 'Vilcabamba' : 'Loja',
   230→        time: f.hora,
   231→        income: f.totalRecaudado.toString(),
   232→        efectivoReal: f.efectivoContado.toString(),
   233→        boletos: '0',
   234→        cajaComunPasajeros: f.boletosCaja.toString(),
   235→        cajaComunMonto: (f.cajaComunMonto || 0).toString(),
   236→      }));
   237→
   238→      const body = {
   239→        date: workDate(session),
   240→        km: km,
   241→        conductor: '',
   242→        ayudanteNombre: session.ayudanteNombre,
   243→        vtCode: session.vtCode,
   244→        trips,
   245→        expenses: gastos,
   246→        tickets: tickets || '0',
   247→        cajaComun: totalCajaComunMonto,
   248→        sobrante: sobrante || '0',
   249→        photoUrl: fotoPreview,
   250→      };
   251→
   252→      const fechaTrabajo = workDate(session);
   253→      const isOnline = navigator.onLine;
   254→
   255→      if (isOnline) {
   256→        const res = await fetch('/api/records', {
   257→          method: 'POST',
   258→          headers: { 'Content-Type': 'application/json' },
   259→          body: JSON.stringify(body),
   260→        });
   261→        if (res.ok) {
   262→          setSaved(true);
   263→        } else {
   264→          const result = await res.json().catch(() => ({ error: 'Error desconocido' }));
   265→          console.error('Error guardando arqueo en servidor:', result.error);
   266→          // Save offline if server fails
   267→          localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   268→          setGuardadoOffline(true);
   269→          setSaved(true);
   270→        }
   271→      } else {
   272→        // Save to localStorage for later sync
   273→        localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify(body));
   274→        setGuardadoOffline(true);
   275→        setSaved(true);
   276→      }
   277→    } catch (err) {
   278→      console.error('Error guardando arqueo:', err);
   279→      localStorage.setItem(`arqueo_general_${session.vtCode}_${fechaTrabajo}`, JSON.stringify({
   280→        date: fechaTrabajo, km, vtCode: session.vtCode, ayudanteNombre: session.ayudanteNombre,
   281→        trips: frecuencias.map(f => ({ routeFrom: 'Loja', routeTo: 'Vilcabamba', time: f.hora, income: f.totalRecaudado.toString(), boletos: '0' })),
   282→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   283→      }));
   284→      setGuardadoOffline(true);
   285→      setSaved(true);
   286→    } finally {
   287→      setSaving(false);
   288→    }
   289→  };
   290→
   291→  if (loading) {
   292→    return (
   293→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   294→        <Loader2 className="w-8 h-8 text-[#912D26] animate-spin mb-3" />
   295→        <p className="text-[#3A3A3A]">Cargando arqueo general...</p>
   296→      </div>
   297→    );
   298→  }
   299→
   300→  // Saved confirmation screen
   301→  if (saved) {
   302→    return (
   303→      <div className="flex flex-col min-h-[100dvh] bg-[#912D26] text-white">
   304→        <div className="flex-1 flex flex-col items-center justify-center px-6">
   305→          <CheckCircle2 className="w-20 h-20 mb-4" />
   306→          <h1 className="text-2xl font-black mb-2">ARQUEO GENERAL GUARDADO</h1>
   307→          <p className="text-white/80 text-sm mb-1">{session.nombre} — {workDate(session)}</p>
   308→          <p className="text-white/80 text-sm mb-6">{totalVentas} ventas · ${totalIngresosAuto.toFixed(2)} ingresos</p>
   309→
   310→          {guardadoOffline && (
   311→            <div className="bg-yellow-500/20 border border-yellow-400/30 rounded-2xl px-5 py-3 mb-6 text-center">
   312→              <p className="text-yellow-200 text-sm font-bold">Guardado local (sin internet)</p>
   313→              <p className="text-yellow-200/70 text-xs mt-1">Se sincronizará cuando haya conexión</p>
   314→            </div>
   315→          )}
   316→
   317→          <div className="w-full max-w-xs space-y-3">
   318→            {connection.pendingCount > 0 && connection.status !== 'offline' && (
   319→              <button onClick={onGoToSync}
   320→                className="w-full py-4 rounded-2xl bg-white text-[#912D26] font-black text-lg flex items-center justify-center gap-2 active:scale-95">
   321→                <Send className="w-5 h-5" /> SINCRONIZAR {connection.pendingCount} VENTAS
   322→              </button>
   323→            )}
   324→            <button onClick={onSaved}
   325→              className="w-full py-4 rounded-2xl bg-white/20 text-white font-bold text-lg flex items-center justify-center gap-2 active:scale-95">
   326→              FINALIZAR
   327→            </button>
   328→          </div>
   329→        </div>
   330→      </div>
   331→    );
   332→  }
   333→
   334→  // Confirm save dialog
   335→  if (confirmSave) {
   336→    return (
   337→      <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   338→        <div className="flex-1 flex items-center justify-center px-4">
   339→          <div className="bg-white rounded-2xl border border-gray-200 p-6 max-w-md w-full shadow-lg">
   340→            <h2 className="text-lg font-bold text-[#3A3A3A] mb-4">Confirmar Arqueo General</h2>
   341→            <div className="bg-gray-50 rounded-xl p-4 space-y-2 mb-4">
   342→              <div className="flex justify-between text-sm">
   343→                <span className="text-gray-500">Producción</span>
   344→                <span className="font-bold text-[#3A3A3A]">${production.toFixed(2)}</span>
   345→              </div>
   346→              {totalCajaComunPasajeros > 0 && (
   347→                <div className="flex justify-between text-sm">
   348→                  <span className="text-purple-700">Caja Comun (Ofi.Loja): {totalCajaComunPasajeros} boletos</span>
   349→                  <span className="font-bold text-purple-700">${totalCajaComunMonto.toFixed(2)}</span>
   350→                </div>
   351→              )}
   352→              <div className="flex justify-between text-sm">
   353→                <span className="text-gray-500">Total Gastos</span>
   354→                <span className="font-bold text-red-600">${totalGastos.toFixed(2)}</span>
   355→              </div>
   356→              <hr className="border-gray-200" />
   357→              <div className="flex justify-between text-sm">
   358→                <span className="text-gray-600">Entrega Ayudante</span>
   359→                <span className={`font-bold ${entregaAyudante >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   360→                  ${entregaAyudante.toFixed(2)}
   361→                </span>
   362→              </div>
   363→              <div className="flex justify-between text-sm">
   364→                <span className="text-gray-600">Entrega Compañía</span>
   365→                <span className={`font-bold ${entregaCompania >= 0 ? 'text-green-600' : 'text-red-600'}`}>
   366→                  ${entregaCompania.toFixed(2)}
   367→                </span>
   368→              </div>
   369→            </div>
   370→            <p className="text-sm text-[#912D26] mb-4">
   371→              Entrega Ayudante: <strong>${entregaAyudante.toFixed(2)}</strong> —
   372→              Entrega Compañía: <strong>${entregaCompania.toFixed(2)}</strong>
   373→            </p>
   374→            <div className="flex gap-3">
   375→              <button onClick={() => setConfirmSave(false)}
   376→                className="flex-1 py-3 rounded-xl border-2 border-gray-200 text-gray-500 font-bold active:scale-95">
   377→                Cancelar
   378→              </button>
   379→              <button onClick={handleConfirmSave} disabled={saving}
   380→                className={`flex-1 py-3 rounded-xl font-black active:scale-95 ${saving ? 'bg-[#D6D6D6] text-gray-400 cursor-not-allowed' : 'bg-[#912D26] text-white'}`}>
   381→                {saving ? 'Guardando...' : 'CONFIRMAR'}
   382→              </button>
   383→            </div>
   384→          </div>
   385→        </div>
   386→      </div>
   387→    );
   388→  }
   389→
   390→  return (
   391→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   392→      {/* Barra conexion */}
   393→      <div className={`${connection.bgColor} px-4 py-1.5 flex items-center justify-center gap-2 text-xs font-medium ${connection.color}`}>
   394→        <span>{connection.icon}</span>
   395→        <span>{connection.label}</span>
   396→      </div>
   397→
   398→      {/* Header */}
   399→      <div className="bg-[#912D26] text-white px-4 py-3">
   400→        <div className="flex items-center justify-between">
   401→          <button onClick={onClose} className="text-red-100"><ChevronLeft className="w-6 h-6" /></button>
   402→          <div className="text-center">
   403→            <h2 className="text-lg font-bold">Arqueo General</h2>
   404→            <p className="text-red-100 text-xs">{session.nombre} — Turno completo</p>
   405→          </div>
   406→          <div className="w-6" />
   407→        </div>
   408→      </div>
   409→
   410→      <div className="flex-1 px-4 py-4 space-y-4 overflow-y-auto pb-8">
   411→        {/* Errores */}
   412→        {errors.length > 0 && (
   413→          <div className="p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm space-y-1">
   414→            <p className="font-semibold flex items-center gap-1.5"><AlertTriangle className="w-4 h-4" /> Campos obligatorios:</p>
   415→            {errors.map((e, i) => <p key={i}>- {e}</p>)}
   416→          </div>
   417→        )}
   418→
   419→        {/* 1. INGRESOS POR FRECUENCIA — Sistema vs Efectivo Real */}
   420→        <div className="bg-white rounded-2xl border border-gray-100 p-4 shadow-sm">
   421→          <h3 className="font-bold text-[#3A3A3A] mb-3 text-sm uppercase flex items-center gap-2">
   422→            <DollarSign className="w-4 h-4 text-green-600" /> Ingresos por Frecuencia
   423→          </h3>
   424→          {frecuencias.length === 0 ? (
   425→            <p className="text-gray-400 text-xs text-center py-4">No hay frecuencias cerradas</p>
   426→          ) : (
   427→            <div className="space-y-2">
   428→              {/* Header row */}
   429→              <div className="flex items-center justify-between px-3 py-1 text-[10px] font-bold text-gray-400 uppercase">
   430→                <div className="flex-1">Frecuencia</div>
   431→                <div className="w-20 text-right">Sistema</div>
   432→                <div className="w-20 text-right">Real</div>
   433→                <div className="w-16 text-right">Dif.</div>
   434→              </div>
   435→              {frecuencias.map((f, i) => {
   436→                const hasDiff = Math.abs(f.diferencia) >= 0.01;
   437→                if (f.isIngresoEspecial) {
   438→                  return (
   439→                    <div key={f.estadoId} className="flex items-center justify-between py-2 px-3 rounded-xl bg-amber-50 border border-amber-200">
   440→                      <div className="flex items-center gap-2 flex-1 min-w-0">
   441→                        <Sparkles className="w-4 h-4 text-amber-500 flex-shrink-0" />
   442→                        <div className="min-w-0">
   443→                          <div className="text-sm font-semibold text-amber-800 truncate">{f.hora} — {f.ingresoEspecialNota || 'Ingreso especial'}</div>
   444→                          <div className="text-[10px] text-amber-500">Viaje especial</div>
   445→                        </div>
   446→                      </div>
   447→                      <div className="w-20 text-right">
   448→                        <div className="font-black text-amber-700">${f.totalRecaudado.toFixed(2)}</div>
   449→                      </div>
   450→                      <div className="w-20 text-right">
   451→                        <div className="font-black text-amber-700">${f.efectivoContado.toFixed(2)}</div>
   452→                      </div>
   453→                      <div className="w-16 text-right">

   514→            </div>
