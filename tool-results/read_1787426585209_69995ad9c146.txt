   461→   560→   560→//
   462→   561→   561→// Lógica de precios (tipo semántico asignado por el API):
   463→   562→   562→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   464→   563→   563→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   465→   564→   564→//
   466→   565→   565→// El API asigna dirección según el origen:
   467→   566→   566→//   Si origen = Loja →  "ida" (alejándose)
   468→   567→   567→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   469→   568→   568→//
   470→   569→   569→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   471→   570→   570→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   472→   571→   571→
   473→   572→   572→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   474→   573→   573→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   475→   574→   574→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   476→   575→   575→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   477→   576→   576→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   478→   577→   577→  return false;
   479→   578→   578→}

   483→   582→   582→  if (viajaHaciaLoja(ruta, tipo)) {

   485→   584→   584→    if (ruta === 'Loja - El Tambo') {

   489→   588→   588→    const vuelta = preciosVuelta[parada];
   490→   589→   589→    if (vuelta) return vuelta;

   494→   593→   593→  const ida = preciosIda[parada];
   495→   594→   594→  if (ida) return ida;
   496→   595→   595→  const vuelta = preciosVuelta[parada];

   504→   603→   603→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {

   513→   612→   612→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   514→   613→   613→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   515→   614→   614→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   516→   615→   615→    const baseName = ruta.replace(' - Loja', '');
   517→   616→   616→    effectiveRuta = `Loja - ${baseName}`;
   518→   617→   617→    // NO invertir effectiveTipo — ya viene correcto del API
   519→   618→   618→  }
   520→   619→   619→
   521→   620→   620→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   522→   621→   621→  if (!rutaConfig) {
   523→   622→   622→    // Fallback: try original ruta
   524→   623→   623→    const origConfig = RUTA_PARADAS[ruta];
   525→   624→   624→    if (!origConfig) return [];
   526→   625→   625→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   527→   626→   626→    return paradas.map(parada => {
   528→   627→   627→      const precio = getPrecio(parada, ruta, tipo);
   529→   628→   628→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   530→   629→   629→    });
   531→   630→   630→  }
   532→   631→   631→
   533→   632→   632→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   534→   633→   633→  return paradas.map(parada => {
   535→   634→   634→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   536→   635→   635→    return {
   537→   636→   636→      parada,
   538→   637→   637→      normal: precio?.normal ?? 0,
   539→   638→   638→      media: precio?.media ?? 0,
   540→   639→   639→    };
   541→   640→   640→  });
   542→   641→   641→}

   544→   643→   643→// Normalizar ruta y tipo para rutas "X - Loja"
   545→   644→   644→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   546→   645→   645→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   547→   646→   646→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   548→   647→   647→    // El API ya asigna tipo correcto, NO invertir
   549→   648→   648→    const baseName = ruta.replace(' - Loja', '');
   550→   649→   649→    return { ruta: `Loja - ${baseName}`, tipo };
   551→   650→   650→  }
   552→   651→   651→  return { ruta, tipo };
   553→   652→   652→}

   556→   655→   655→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   557→   656→   656→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);

   561→   660→   660→}

   568→   667→   667→// Buscar la ruta más cercana dado un nombre de ruta parcial
   569→   668→   668→export function matchRuta(rutaNombre: string): string {
   570→   669→   669→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   571→   670→   670→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   572→   671→   671→  if (RUTA_PARADAS[normalized]) return normalized;
   573→   672→   672→
   574→   673→   673→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   575→   674→   674→  for (const key of Object.keys(RUTA_PARADAS)) {
   576→   675→   675→    const keyParts = key.split(' - ');
   577→   676→   676→    if (parts.length >= 2 && keyParts.length >= 2) {
   578→   677→   677→      const match = (
   579→   678→   678→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   580→   679→   679→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   581→   680→   680→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   582→   681→   681→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   583→   682→   682→      );
   584→   683→   683→      if (match) return key;
   585→   684→   684→    }
   586→   685→   685→  }
   587→   686→   686→  return rutaNombre;
   588→   687→   687→}
