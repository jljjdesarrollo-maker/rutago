    20→    20→interface FrecuenciaResumen {

    33→    33→interface GastoItem {

    37→    37→
    38→    38→const today = () => new Date().toISOString().split('T')[0];
    39→    39→
    40→    40→// Work date: use session.fecha (date when shift started, not necessarily today)
    41→    41→function workDate(session: VTSession): string {
    42→    42→  return session.fecha || today();
    43→    43→}
    44→    44→

    88→    88→            ruta: e.ruta,

    90→    90→            direccion: e.direccion,
    91→    91→            ventasCount: ventas.length,
    92→    92→            totalRecaudado: sistemaTotal,
    93→    93→            efectivoContado: efectivo,
    94→    94→            diferencia: diff,

   228→   228→        expenses: gastos, tickets: tickets || '0', sobrante: sobrante || '0', photoUrl: fotoPreview,
   229→   229→      }));
