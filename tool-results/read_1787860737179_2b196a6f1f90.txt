    56→function getVTRouteType(frecuencias?: { routeFrom: string; routeTo: string }[]): VTRouteType {
    57→  if (!frecuencias || frecuencias.length === 0) return 'vilcabamba';
    58→  const allDests = [...frecuencias.map(f => f.routeTo), ...frecuencias.map(f => f.routeFrom)];
    59→  if (allDests.some(d => d === 'El Tambo')) return 'el_tambo';
    60→  if (allDests.some(d => d === 'La Elvira')) return 'la_elvira';
    61→  if (allDests.some(d => d === 'Yangana')) return 'yangana';
    62→  if (allDests.some(d => d === 'Zahuayco')) return 'zahuayco';
    63→  return 'vilcabamba';
    64→}

   476→                }`}>{l}</p>
   477→              ))}
   478→            </div>
   479→          </div>
   480→        )}
   481→      </div>
   482→
   483→      <div className="flex-1 px-5 py-5 space-y-4">
   484→        {/* Ayudante activo — compacto */}
   485→        {ayudante ? (
   486→          <div className="flex items-center gap-3 bg-green-50 border border-green-200 rounded-xl px-4 py-3">
   487→            <CheckCircle className="w-5 h-5 text-green-600 flex-shrink-0" />
   488→            <div>
   489→              <p className="font-semibold text-green-800 text-sm">{ayudante.nombre}</p>
   490→              <p className="text-green-600 text-[10px]">Turno activo</p>
   491→            </div>
   492→          </div>
   493→        ) : (
   494→          <div className="bg-yellow-50 border border-yellow-200 rounded-xl px-4 py-3">
   495→            <p className="text-yellow-700 text-sm">No hay ayudante activo asignado</p>
   496→            <p className="text-yellow-600 text-xs mt-1">Configure un ayudante desde el panel de administracion</p>
   497→          </div>
   498→        )}
   499→
   500→        {/* Selector de VT — grid grande */}
   501→        {vts.length === 0 ? (
   502→          <div className="bg-yellow-50 border border-yellow-200 rounded-2xl p-5 text-center">
   503→            <p className="text-yellow-700 font-medium">No hay unidades activas</p>
   504→            <p className="text-yellow-600 text-sm mt-1">Ejecuta el seed: POST /api/seed-vts</p>
   505→          </div>
   506→        ) : (
   507→          <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-100">
   508→            <h2 className="text-base font-bold text-[#3A3A3A] mb-3 flex items-center gap-2">
   509→              <Bus className="w-4 h-4 text-[#912D26]" /> Unidad
   510→            </h2>
   511→            <div className={`grid gap-3 ${vts.length <= 2 ? 'grid-cols-2' : vts.length <= 4 ? 'grid-cols-2' : 'grid-cols-3'}`}>
   512→              {vts.map(vt => {
   513→                const routeType = getVTRouteType(vt.frecuencias);
   514→                const style = ROUTE_STYLES[routeType];
   515→                const isSelected = selectedVT === vt.codigo;
   516→                return (
   517→                  <button key={vt.codigo}
   518→                    onClick={() => vts.length === 1 ? startSession(vt.codigo) : setSelectedVT(vt.codigo)}
   519→                    className={`relative py-5 px-3 rounded-xl text-center transition-all active:scale-95 border ${
   520→                      isSelected
   521→                        ? `${style.selectedBg} text-white shadow-md ring-2 ${style.selectedRing}`
   522→                        : `${style.bg} ${style.border} ${style.text} hover:opacity-80 active:opacity-70`
   523→                    }`}
   524→                  >
   525→                    <div className="text-base font-bold">{vt.codigo}</div>
   526→                    <div className={`text-[10px] mt-0.5 ${isSelected ? 'text-white/70' : style.subtext}`}>{vt.nombre}</div>
   527→                    {style.label && !isSelected && (
   528→                      <span className={`absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full ${style.badgeBg} ${style.badgeText} shadow-sm`}>
   529→                        {style.label}
   530→                      </span>
   531→                    )}
   532→                    {style.label && isSelected && (
   533→                      <span className="absolute -top-1.5 -right-1.5 text-[8px] font-bold px-1.5 py-0.5 rounded-full bg-white/25 text-white shadow-sm">
   534→                        {style.label}
   535→                      </span>
   536→                    )}
   537→                  </button>
   538→                );
   539→              })}
   540→            </div>
   541→          </div>
   542→        )}
   543→
   544→        {/* Boton Iniciar — grande y prominente */}
   545→        <button onClick={handleStart} disabled={!selectedVT || !ayudante || pendingVentasCount > 0}
   546→          className={`w-full py-5 rounded-2xl font-black text-xl flex items-center justify-center gap-3 transition-all ${
   547→            selectedVT && ayudante && pendingVentasCount === 0
   548→              ? 'bg-[#912D26] text-white shadow-xl shadow-red-300 active:scale-[0.97]'
   549→              : 'bg-[#D6D6D6] text-gray-400 cursor-not-allowed'
   550→          }`}
   551→        >
   552→          {pendingVentasCount > 0
   553→            ? `ESPERANDO SYNC — ${pendingVentasCount} VENTAS`
   554→            : 'Iniciar Turno'
   555→          } <ArrowRight className="w-6 h-6" />
   556→        </button>
   557→      </div>
   558→
   559→      {/* Ventas pendientes — bloquear inicio de nuevo VT */}
   560→      {pendingVentasCount > 0 && pendingCheckDone && (
   561→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-[#912D26] to-[#b33d34] p-4 shadow-2xl z-40">
   562→          <div className="max-w-lg mx-auto">
   563→            <div className="flex items-center gap-2 mb-2">
   564→              <AlertTriangle className="w-5 h-5 text-white" />
   565→              <p className="text-white font-bold text-sm">Ventas sin sincronizar</p>
   566→            </div>
   567→            <p className="text-white/90 text-xs mb-3">
   568→              Tienes <strong>{pendingVentasCount} venta{pendingVentasCount !== 1 ? 's' : ''}</strong> pendiente{pendingVentasCount !== 1 ? 's' : ''} de un turno anterior. Debes sincronizar antes de iniciar uno nuevo.
   569→            </p>
   570→            {navigator.onLine ? (
   571→              <button onClick={handleForceSync} disabled={syncingPending}
   572→                className="w-full py-3 rounded-xl bg-white text-[#912D26] font-black text-sm flex items-center justify-center gap-2 active:scale-[0.98] disabled:opacity-60">
   573→                {syncingPending ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Wifi className="w-4 h-4" />}
   574→                {syncingPending ? 'SINCRONIZANDO...' : `SINCRONIZAR ${pendingVentasCount} VENTAS`}
   575→              </button>
   576→            ) : (
   577→              <div className="w-full py-3 rounded-xl bg-white/20 text-white/80 text-sm flex items-center justify-center gap-2">
   578→                <WifiOff className="w-4 h-4" /> SIN INTERNET — CONECTA PARA SINCRONIZAR
   579→              </div>
   580→            )}
   581→          </div>
   582→        </div>
   583→      )}
   584→
   585→      {/* Existing unfinished session banner */}
   586→      {existingUnfinished && existingSession && (
   587→        <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-orange-500 to-red-500 p-4 shadow-2xl z-40">
   588→          <div className="max-w-lg mx-auto">
   589→            <div className="flex items-center gap-2 mb-2">
   590→              <AlertTriangle className="w-5 h-5 text-white" />
   591→              <p className="text-white font-bold text-sm">Turno sin terminar</p>
   592→            </div>
   593→            <p className="text-white/90 text-xs mb-3">
   594→              Tienes un turno pendiente en <strong>{existingSession.nombre}</strong> ({existingSession.vtCode}) con frecuencias abiertas.
   595→            </p>
   596→            <div className="flex gap-2">
   597→              <button onClick={handleResumeOld}
   598→                className="flex-1 py-2.5 rounded-xl bg-white text-orange-600 font-bold text-sm active:scale-[0.98]">
   599→                Continuar VT {existingSession.vtCode}
   600→              </button>
   601→              <button onClick={() => setConfirmNewSession(true)}
   602→                className="flex-1 py-2.5 rounded-xl bg-white/20 text-white font-semibold text-sm active:scale-[0.98]">
   603→                Iniciar nuevo
   604→              </button>
   605→            </div>
   606→          </div>
   607→        </div>
   608→      )}
   609→
   610→      {/* Confirm new session modal */}
   611→      {confirmNewSession && existingSession && (
   612→        <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
   613→          <div className="bg-white rounded-2xl w-full max-w-sm p-6 text-center">
   614→            <div className="w-16 h-16 bg-red-100 rounded-full flex items-center justify-center mx-auto mb-4">
   615→              <AlertTriangle className="w-8 h-8 text-[#912D26]" />
   616→            </div>
   617→            <h3 className="text-lg font-bold text-[#3A3A3A] mb-2">Abandonar turno anterior?</h3>
   618→            <div className="bg-red-50 rounded-xl p-3 mb-3">
   619→              <p className="text-sm text-[#912D26] font-semibold">
   620→                {existingSession.nombre} ({existingSession.vtCode}) — frecuencias sin completar.
   621→              </p>
   622→            </div>
   623→            <div className="bg-yellow-50 border border-yellow-200 rounded-xl p-3 mb-4">
   624→              <p className="text-xs text-yellow-800 font-bold">
   625→                Se ELIMINARAN todos los datos del turno anterior. Solo puede haber un turno activo por dia.
   626→              </p>
   627→            </div>
   628→            <div className="flex gap-2">
   629→              <button onClick={() => setConfirmNewSession(false)}
   630→                className="flex-1 py-3 rounded-xl bg-[#912D26] text-white font-bold text-sm active:scale-[0.98] shadow-lg shadow-red-200">
   631→                Cancelar
   632→              </button>
   633→              <button onClick={handleForceNew}
   634→                className="flex-1 py-3 rounded-xl border-2 border-red-200 text-[#912D26] font-bold text-sm active:scale-[0.98]">
   635→                Nuevo Turno
   636→              </button>
   637→            </div>
   638→          </div>
   639→        </div>
   640→      )}
   641→    </div>
   642→  );
