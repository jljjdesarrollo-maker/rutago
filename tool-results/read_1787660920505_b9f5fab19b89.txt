   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→  return 'yellow';
   128→}
   129→
   130→export function isParadaPrincipal(parada: string): boolean {
   131→  // Paradas principales: no contienen → en el nombre
   132→  return !parada.includes('→');
   133→}

   732→// ─── Funciones de precio con soporte por dirección ───
   733→//
   734→// Lógica de precios (tipo semántico asignado por el API):
   735→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   736→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   737→//
   738→// El API asigna dirección según el origen:
   739→//   Si origen = Loja →  "ida" (alejándose)
   740→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   741→//
   742→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   743→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   744→
   745→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   746→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   747→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   748→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   749→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   750→  return false;
   751→}
   752→
   753→// Obtener precio considerando la dirección del viaje
   754→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   755→  if (viajaHaciaLoja(ruta, tipo)) {
   756→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   757→    if (ruta === 'Loja - El Tambo') {
   758→      const tamboVuelta = preciosVueltaElTambo[parada];
   759→      if (tamboVuelta) return tamboVuelta;
   760→    }
   761→    // Ruta Zahuayco tiene precios de vuelta propios (directos desde Zahuayco)
   762→    if (ruta === 'Loja - Zahuayco') {
   763→      const zahuaycoVuelta = preciosVueltaZahuayco[parada];
   764→      if (zahuaycoVuelta) return zahuaycoVuelta;
   765→    }
   766→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   767→    if (ruta === 'Loja - Yangana') {
   768→      const yangVuelta = preciosVueltaYangana[parada];
   769→      if (yangVuelta) return yangVuelta;
   770→    }
   771→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   772→    if (ruta === 'Loja - Vilcabamba') {
   773→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   774→      if (vilcVuelta) return vilcVuelta;
   775→    }
   776→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   777→    if (ruta === 'Loja - La Elvira') {
   778→      const elviraVuelta = preciosVueltaLaElvira[parada];
   779→      if (elviraVuelta) return elviraVuelta;
   780→    }
   781→    const vuelta = preciosVuelta[parada];
   782→    if (vuelta) return vuelta;
   783→    // Si la parada no tiene precio de vuelta, buscar en precios base
   784→  }
   785→  // Buscar primero en ida (base), luego en vuelta como fallback
   786→  const ida = preciosIda[parada];
   787→  if (ida) return ida;
   788→  const vuelta = preciosVuelta[parada];
   789→  return vuelta || null;
   790→}

   795→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   796→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   797→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   798→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   799→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   800→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   801→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   802→  let effectiveRuta = ruta;
   803→  let effectiveTipo = tipo;
   804→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   805→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   806→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   807→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   808→    const baseName = ruta.replace(' - Loja', '');
   809→    effectiveRuta = `Loja - ${baseName}`;
   810→    // NO invertir effectiveTipo — ya viene correcto del API
   811→  }
   812→
   813→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   814→  if (!rutaConfig) {
   815→    // Fallback: try original ruta
   816→    const origConfig = RUTA_PARADAS[ruta];
   817→    if (!origConfig) return [];
   818→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   819→    return paradas.map(parada => {
   820→      const precio = getPrecio(parada, ruta, tipo);
   821→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   822→    });
   823→  }
   824→
   825→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   826→  return paradas.map(parada => {
   827→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   828→    return {
   829→      parada,
   830→      normal: precio?.normal ?? 0,
   831→      media: precio?.media ?? 0,
   832→    };
   833→  });
   834→}

   836→// Normalizar ruta y tipo para rutas "X - Loja"
   837→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   838→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   839→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   840→    // El API ya asigna tipo correcto, NO invertir
   841→    const baseName = ruta.replace(' - Loja', '');
   842→    return { ruta: `Loja - ${baseName}`, tipo };
   843→  }
   844→  return { ruta, tipo };
   845→}

   847→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   848→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   849→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   850→  const precio = getPrecio(parada, nr, nt);
   851→  if (!precio) return TARIFA_MINIMA;
   852→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   853→}

   860→// Buscar la ruta más cercana dado un nombre de ruta parcial
   861→export function matchRuta(rutaNombre: string): string {
   862→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   863→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   864→  if (RUTA_PARADAS[normalized]) return normalized;
   865→
   866→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   867→  for (const key of Object.keys(RUTA_PARADAS)) {
   868→    const keyParts = key.split(' - ');
   869→    if (parts.length >= 2 && keyParts.length >= 2) {
   870→      const match = (
   871→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   872→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   873→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   874→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   875→      );
   876→      if (match) return key;
   877→    }
   878→  }
   879→  return rutaNombre;
   880→}
