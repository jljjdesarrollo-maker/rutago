   121→export function getZonaParada(parada: string): ZonaColor {
   122→  if (PARADA_ZONA[parada]) return PARADA_ZONA[parada];
   123→  // Intermedios: color por hub de origen
   124→  if (parada.startsWith('Vilc→')) return 'purple';
   125→  if (parada.endsWith('→Vilc')) return 'purple';
   126→  if (parada.startsWith('Mal→') || parada.endsWith('→Mal')) return 'orange';
   127→  return 'yellow';
   128→}
   129→
   130→export function isParadaPrincipal(parada: string): boolean {
   131→  // Paradas principales: no contienen → en el nombre
   132→  return !parada.includes('→');
   133→}

   759→// ─── Funciones de precio con soporte por dirección ───
   760→//
   761→// Lógica de precios (tipo semántico asignado por el API):
   762→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   763→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   764→//
   765→// El API asigna dirección según el origen:
   766→//   Si origen = Loja →  "ida" (alejándose)
   767→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   768→//
   769→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   770→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   771→
   772→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   773→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   774→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   775→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   776→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   777→  return false;
   778→}
   779→
   780→// Obtener precio considerando la dirección del viaje
   781→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   782→  if (viajaHaciaLoja(ruta, tipo)) {
   783→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   784→    if (ruta === 'Loja - El Tambo') {
   785→      const tamboVuelta = preciosVueltaElTambo[parada];
   786→      if (tamboVuelta) return tamboVuelta;
   787→    }
   788→    // Ruta Zahuayco tiene precios de vuelta propios (directos desde Zahuayco)
   789→    if (ruta === 'Loja - Zahuayco') {
   790→      const zahuaycoVuelta = preciosVueltaZahuayco[parada];
   791→      if (zahuaycoVuelta) return zahuaycoVuelta;
   792→    }
   793→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   794→    if (ruta === 'Loja - Yangana') {
   795→      const yangVuelta = preciosVueltaYangana[parada];
   796→      if (yangVuelta) return yangVuelta;
   797→    }
   798→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   799→    if (ruta === 'Loja - Vilcabamba') {
   800→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   801→      if (vilcVuelta) return vilcVuelta;
   802→    }
   803→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   804→    if (ruta === 'Loja - La Elvira') {
   805→      const elviraVuelta = preciosVueltaLaElvira[parada];
   806→      if (elviraVuelta) return elviraVuelta;
   807→    }
   808→    const vuelta = preciosVuelta[parada];
   809→    if (vuelta) return vuelta;
   810→    // Si la parada no tiene precio de vuelta, buscar en precios base
   811→  }
   812→  // Buscar primero en ida (base), luego en vuelta como fallback
   813→  const ida = preciosIda[parada];
   814→  if (ida) return ida;
   815→  const vuelta = preciosVuelta[parada];
   816→  return vuelta || null;
   817→}

   822→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   823→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   824→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   825→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   826→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   827→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   828→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   829→  let effectiveRuta = ruta;
   830→  let effectiveTipo = tipo;
   831→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   832→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   833→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   834→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   835→    const baseName = ruta.replace(' - Loja', '');
   836→    effectiveRuta = `Loja - ${baseName}`;
   837→    // NO invertir effectiveTipo — ya viene correcto del API
   838→  }
   839→
   840→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   841→  if (!rutaConfig) {
   842→    // Fallback: try original ruta
   843→    const origConfig = RUTA_PARADAS[ruta];
   844→    if (!origConfig) return [];
   845→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   846→    return paradas.map(parada => {
   847→      const precio = getPrecio(parada, ruta, tipo);
   848→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   849→    });
   850→  }
   851→
   852→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   853→  return paradas.map(parada => {
   854→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   855→    return {
   856→      parada,
   857→      normal: precio?.normal ?? 0,
   858→      media: precio?.media ?? 0,
   859→    };
   860→  });
   861→}

   863→// Normalizar ruta y tipo para rutas "X - Loja"
   864→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   865→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   866→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   867→    // El API ya asigna tipo correcto, NO invertir
   868→    const baseName = ruta.replace(' - Loja', '');
   869→    return { ruta: `Loja - ${baseName}`, tipo };
   870→  }
   871→  return { ruta, tipo };
   872→}

   874→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   875→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   876→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   877→  const precio = getPrecio(parada, nr, nt);
   878→  if (!precio) return TARIFA_MINIMA;
   879→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   880→}

   887→// Buscar la ruta más cercana dado un nombre de ruta parcial
   888→export function matchRuta(rutaNombre: string): string {
   889→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   890→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   891→  if (RUTA_PARADAS[normalized]) return normalized;
   892→
   893→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   894→  for (const key of Object.keys(RUTA_PARADAS)) {
   895→    const keyParts = key.split(' - ');
   896→    if (parts.length >= 2 && keyParts.length >= 2) {
   897→      const match = (
   898→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   899→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   900→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   901→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   902→      );
   903→      if (match) return key;
   904→    }
   905→  }
   906→  return rutaNombre;
   907→}
