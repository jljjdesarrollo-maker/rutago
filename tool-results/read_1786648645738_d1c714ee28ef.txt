    17→const today = () => new Date().toISOString().split('T')[0];
    18→
    19→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    20→
    21→function estadosKey(vtCode: string, fecha: string) {
    22→  return `rg_estados_${vtCode}_${fecha}`;
    23→}
    24→
    25→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    26→  try {
    27→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    28→  } catch (e) { console.error('Error saving estados to LS:', e); }
    29→}
    30→
    31→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    32→  try {
    33→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    34→    return raw ? JSON.parse(raw) : null;
    35→  } catch { return null; }
    36→}
    37→
    38→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    39→  const all = loadEstadosFromLS(vtCode, fecha);
    40→  if (!all) return;
    41→  const idx = all.findIndex(e => e.estadoId === estadoId);
    42→  if (idx >= 0) {
    43→    all[idx] = { ...all[idx], ...updates };
    44→    saveEstadosToLS(vtCode, fecha, all);
    45→  }
    46→}

    85→            estadoId: `${fecha}_${f.id}`,

    87→            nombre: f.nombre,

   104→    } finally {

   108→

   112→      setPendingCount(count);
   113→    } catch { /* ignore */ }
   114→  }, []);

   116→  // Reload ventas counts from IndexedDB periodically

   120→        const ventas = await getVentasByFrecuencia(estado.estadoId);

   123→        const newTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   124→        if (newCount !== estado.ventasCount || Math.abs(newTotal - estado.totalRecaudado) > 0.01) {
   125→          updateEstadoInLS(session.vtCode, fecha, estado.estadoId, { ventasCount: newCount, totalRecaudado: newTotal });

   128→          ));
   129→        }
   130→      } catch { /* ignore */ }
   131→    }

   133→
   134→  useEffect(() => {

   144→      refreshVentasCounts();

   257→      case 'abierta': return 'bg-green-50 border-green-200';

   259→      case 'no_realizada': return 'bg-orange-50 border-orange-200';

   270→    { id: 'orden_superior', label: 'Orden superior', icon: <Ban className="w-4 h-4" />, color: 'text-gray-600' },
   271→    { id: 'otro', label: 'Otro motivo', icon: <FileText className="w-4 h-4" />, color: 'text-gray-500' },
   272→  ];
   273→

   275→    switch (estado) {
   276→      case 'pendiente': return <span className="px-2 py-0.5 rounded-full text-xs bg-gray-200 text-gray-600">Pendiente</span>;
   277→      case 'abierta': return <span className="px-2 py-0.5 rounded-full text-xs bg-green-200 text-green-700">Abierta</span>;
   278→      case 'cerrada': return <span className="px-2 py-0.5 rounded-full text-xs bg-blue-200 text-blue-700">Cerrada</span>;

   280→      default: return null;

   282→  };
   283→
   284→  if (loading) {
   285→    return (
   286→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">
   287→        <RefreshCw className="w-8 h-8 text-[#912D26] animate-spin mb-3" />

   295→

   297→    if (hasUnfinishedFrecuencias) {
   298→      setExitModal(true);
   299→    } else {
   300→      onBack();
   301→    }

   303→
   304→  // Check if ALL frequencies are done (cerrada or no_realizada)
   305→  const allFrecuenciasDone = estados.length > 0 && estados.every(e => e.estado === 'cerrada' || e.estado === 'no_realizada');

   307→  // Totals across all frequencies
   308→  const totalVentasAll = estados.reduce((s, e) => s + e.ventasCount, 0);
   309→  const totalRecaudadoAll = estados.reduce((s, e) => s + e.totalRecaudado, 0);
   310→  const cerradasCount = estados.filter(e => e.estado === 'cerrada').length;
   311→
   312→  // SYNC control: if online with pending ventas, block arqueo
