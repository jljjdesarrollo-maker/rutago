     1→     1→'use client';
     2→     2→
     3→     3→import { useState, useMemo, useRef, useEffect } from 'react';
     4→     4→import { ArrowLeft, Save, Plus, Trash2, Camera, X, Loader2, AlertTriangle } from 'lucide-react';
     5→     5→import { Button } from '@/components/ui/button';
     6→     6→import { Input } from '@/components/ui/input';
     7→     7→import { Label } from '@/components/ui/label';
     8→     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     9→     9→import { Separator } from '@/components/ui/separator';
    10→    10→import {
    11→    11→  AlertDialog,
    12→    12→  AlertDialogContent,
    13→    13→  AlertDialogHeader,
    14→    14→  AlertDialogTitle,
    15→    15→  AlertDialogDescription,
    16→    16→  AlertDialogFooter,
    17→    17→  AlertDialogCancel,
    18→    18→  AlertDialogAction,
    19→    19→} from '@/components/ui/alert-dialog';
    20→    20→import { type RecordFormData, type TripData, type ExpenseData, createEmptyFormData, num, getCurrentConductor, getCurrentAyudante } from './types';
    21→    21→import { routes, getRouteLabel, getTimesForRoute } from '@/lib/routes';
    22→    22→
    23→    23→interface RecordFormProps {
    24→    24→  saving: boolean;
    25→    25→  error: string | null;
    26→    26→  onBack: () => void;
    27→    27→  onSave: (data: RecordFormData, photoBase64: string | null) => void;
    28→    28→}
    29→    29→
    30→    30→export function RecordForm({ saving, error, onBack, onSave }: RecordFormProps) {
    31→    31→  const [form, setForm] = useState<RecordFormData>(createEmptyFormData());
    32→    32→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
    33→    33→  const [validationErrors, setValidationErrors] = useState<string[]>([]);
    34→    34→  const [showConfirm, setShowConfirm] = useState(false);
    35→    35→  const [loadedDefaults, setLoadedDefaults] = useState(false);
    36→    36→  const [vts, setVts] = useState<Array<{id: string; codigo: string; nombre: string; frecuencias: any[]}>>([]);
    37→    37→  const [selectedVtId, setSelectedVtId] = useState('');
    38→    38→  const [vtsLoaded, setVtsLoaded] = useState(false);
    39→    39→
    40→    40→  // Load VTs from API - always run seed to ensure data is up to date
    41→    41→  useEffect(() => {
    42→    42→    if (vtsLoaded) return;
    43→    43→    (async () => {
    44→    44→      try {
    45→    45→        // Always run seed (upsert - safe to run multiple times)
    46→    46→        await fetch('/api/seed-vts', { method: 'POST' });
    47→    47→        const res = await fetch('/api/bus-vts');
    48→    48→        const data = await res.json();
    49→    49→        if (Array.isArray(data)) setVts(data);
    50→    50→      } catch { /* ignore */ }
    51→    51→      setVtsLoaded(true);
    52→    52→    })();
    53→    53→  }, [vtsLoaded]);
    54→    54→
    55→    55→  // Load current conductor and ayudante from DB
    56→    56→  useEffect(() => {
    57→    57→    if (loadedDefaults) return;
    58→    58→    Promise.all([getCurrentConductor(), getCurrentAyudante()]).then(([conductor, ayudante]) => {
    59→    59→      if (conductor || ayudante) {
    60→    60→        setForm(prev => ({
    61→    61→          ...prev,
    62→    62→          ...(conductor ? { conductor } : {}),
    63→    63→          ...(ayudante ? { ayudanteNombre: ayudante } : {}),
    64→    64→        }));
    65→    65→      }
    66→    66→      setLoadedDefaults(true);
    67→    67→    });
    68→    68→  }, [loadedDefaults]);
    69→    69→  const fileInputRef = useRef<HTMLInputElement>(null);
    70→    70→
    71→    71→  // Handle VT selection - load frequencies into form
    72→    72→  const handleVtSelect = (vtId: string) => {
    73→    73→    setSelectedVtId(vtId);
    74→    74→    const vt = vts.find(v => v.id === vtId);
    75→    75→    if (vt && Array.isArray(vt.frecuencias)) {
    76→    76→      const newTrips = vt.frecuencias.map((f: any) => ({
    77→    77→        routeFrom: f.routeFrom || 'Loja',
    78→    78→        routeTo: f.routeTo || 'Vilcabamba',
    79→    79→        time: f.time || '',
    80→    80→        income: '0',
    81→    81→        boletos: '0',
    82→    82→      }));
    83→    83→      setForm(prev => ({ ...prev, vtCode: vt.codigo, trips: newTrips }));
    84→    84→    } else {
    85→    85→      setForm(prev => ({ ...prev, vtCode: '', trips: [] }));
    86→    86→    }
    87→    87→  };
    88→    88→
    89→    89→  const updateField = (field: keyof RecordFormData, value: string) => {
    90→    90→    setForm((prev) => ({ ...prev, [field]: value }));
    91→    91→  };
    92→    92→
    93→    93→  const addTrip = () => {
    94→    94→    setForm((prev) => ({
    95→    95→      ...prev,
    96→    96→      trips: [...prev.trips, { routeFrom: 'Loja', routeTo: 'Vilcabamba', time: '', income: '0', boletos: '0' }],
    97→    97→    }));
    98→    98→  };
    99→    99→
   100→   100→  const removeTrip = (index: number) => {
   101→   101→    setForm((prev) => ({
   102→   102→      ...prev,
   103→   103→      trips: prev.trips.filter((_, i) => i !== index),
   104→   104→    }));
   105→   105→  };
   106→   106→
   107→   107→  const updateTrip = (index: number, field: keyof TripData, value: string) => {
   108→   108→    setForm((prev) => {
   109→   109→      const newTrips = [...prev.trips];
   110→   110→      newTrips[index] = { ...newTrips[index], [field]: value };
   111→   111→      return { ...prev, trips: newTrips };
   112→   112→    });
   113→   113→  };
   114→   114→
   115→   115→  const handleRouteSelect = (index: number, routeLabel: string) => {
   116→   116→    const route = routes.find((r) => getRouteLabel(r.from, r.to) === routeLabel);
   117→   117→    if (route) {
   118→   118→      setForm((prev) => {
   119→   119→        const newTrips = [...prev.trips];
   120→   120→        newTrips[index] = { ...newTrips[index], routeFrom: route.from, routeTo: route.to, time: '' };
   121→   121→        return { ...prev, trips: newTrips };
   122→   122→      });
   123→   123→    }
   124→   124→  };
   125→   125→
   126→   126→  const addExpense = () => {
   127→   127→    setForm((prev) => ({
   128→   128→      ...prev,
   129→   129→      expenses: [...prev.expenses, { description: '', amount: '0' }],
   130→   130→    }));
   131→   131→  };
   132→   132→
   133→   133→  const removeExpense = (index: number) => {
   134→   134→    setForm((prev) => ({
   135→   135→      ...prev,
   136→   136→      expenses: prev.expenses.filter((_, i) => i !== index),
   137→   137→    }));
   138→   138→  };
   139→   139→
   140→   140→  const updateExpense = (index: number, field: keyof ExpenseData, value: string) => {
   141→   141→    setForm((prev) => {
   142→   142→      const newExp = [...prev.expenses];
   143→   143→      newExp[index] = { ...newExp[index], [field]: value };
   144→   144→      return { ...prev, expenses: newExp };
   145→   145→    });
   146→   146→  };
   147→   147→
   148→   148→  const handlePhotoChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   149→   149→    const file = e.target.files?.[0];
   150→   150→    if (!file || !file.type.startsWith('image/')) return;
   151→   151→    const reader = new FileReader();
   152→   152→    reader.onload = (ev) => setPhotoPreview(ev.target?.result as string);
   153→   153→    reader.readAsDataURL(file);
   154→   154→    e.target.value = '';
   155→   155→  };
   156→   156→
   157→   157→  const totals = useMemo(() => {
   158→   158→    const tripIncome = form.trips.reduce((s, t) => s + num(t.income), 0);
   159→   159→    const cajaComun = form.trips.reduce((s, t) => s + num(t.boletos), 0);
   160→   160→    const sobrante = num(form.sobrante);
   161→   161→    const production = tripIncome + sobrante;
   162→   162→    const totalGastos = form.expenses.reduce((s, e) => s + num(e.amount), 0);
   163→   163→    const tickets = num(form.tickets);
   164→   164→    const entregaAyudante = production - totalGastos;
   165→   165→    const entregaCompania = cajaComun - tickets;
   166→   166→    return { production, cajaComun, sobrante, totalGastos, tickets, entregaAyudante, entregaCompania };
   167→   167→  }, [form]);
   168→   168→
   169→   169→  const handleSave = () => {
   170→   170→    // Validaciones obligatorias
   171→   171→    const errors: string[] = [];
   172→   172→    if (!form.vtCode.trim()) errors.push('Selecciona un Vehiculo Tipo (VT)');
   173→   173→    if (!form.km.trim()) errors.push('Kilometraje es obligatorio');
   174→   174→    if (!photoPreview) errors.push('Foto del cuaderno es obligatoria');
   175→   175→    if (form.trips.length === 0) errors.push('No hay frecuencias cargadas');
   176→   176→    if (totals.totalGastos <= 0) errors.push('Total de gastos debe ser mayor a 0');
   177→   177→    if (totals.production <= 0) errors.push('Produccion total debe ser mayor a 0');
   178→   178→    if (errors.length > 0) {
   179→   179→      setValidationErrors(errors);
   180→   180→      return;
   181→   181→    }
   182→   182→    setValidationErrors([]);
   183→   183→    setShowConfirm(true);
   184→   184→  };
   185→   185→
   186→   186→  const handleConfirmSave = () => {
   187→   187→    setShowConfirm(false);
   188→   188→    onSave(form, photoPreview);
   189→   189→  };
   190→   190→
   191→   191→  const allRouteOptions = routes.map((r) => getRouteLabel(r.from, r.to));
   192→   192→
   193→   193→  const isOddTrip = (index: number) => (index + 1) % 2 !== 0;
   194→   194→
   195→   195→  return (
   196→   196→    <div className="flex flex-col min-h-[100dvh] bg-[#FAFAFA]">
   197→   197→      <header className="sticky top-0 z-10 bg-white border-b border-[#D6D6D6] px-4 py-3 flex items-center justify-between">
   198→   198→        <div className="flex items-center gap-3">
   199→   199→          <Button variant="ghost" size="icon" onClick={onBack} className="rounded-xl text-[#3A3A3A]">
   200→   200→            <ArrowLeft className="w-5 h-5" />
   201→   201→          </Button>
   202→   202→          <div>
   203→   203→            <h1 className="text-lg font-semibold text-[#3A3A3A]">Registro del Dia</h1>
   204→   204→          </div>
   205→   205→        </div>
   206→   206→        <Button
   207→   207→          onClick={handleSave}
   208→   208→          disabled={saving}
   209→   209→          className="h-10 px-4 rounded-xl bg-[#912D26] hover:bg-[#7A2520] text-white font-semibold text-sm"
   210→   210→        >
   211→   211→          {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Save className="w-4 h-4 mr-1" />}
   212→   212→          {saving ? 'Guardando...' : 'Guardar'}
   213→   213→        </Button>
   214→   214→      </header>
   215→   215→
   216→   216→      <main className="flex-1 overflow-y-auto pb-8">
   217→   217→        <div className="px-4 py-4 space-y-4">
   218→   218→          {error && (
   219→   219→            <div className="p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm">
   220→   220→              {error}
   221→   221→            </div>
   222→   222→          )}
   223→   223→          {validationErrors.length > 0 && (
   224→   224→            <div className="p-3 rounded-xl bg-[#912D26]/10 border border-[#912D26]/30 text-[#912D26] text-sm space-y-1">
   225→   225→              <p className="font-semibold flex items-center gap-1.5"><AlertTriangle className="w-4 h-4" /> Campos obligatorios:</p>
   226→   226→              {validationErrors.map((e, i) => <p key={i}>- {e}</p>)}
   227→   227→            </div>
   228→   228→          )}
   229→   229→
   230→   230→          {/* 0. Seleccion VT */}
   231→   231→          <Card className="rounded-2xl border-2 border-[#912D26]/30 bg-[#912D26]/5">
   232→   232→            <CardContent className="p-4 space-y-2">
   233→   233→              <Label className="text-sm font-semibold text-[#3A3A3A] flex items-center gap-1.5">
   234→   234→                Vehiculo Tipo (VT) del dia <span className="text-[#912D26]">*</span>
   235→   235→              </Label>
   236→   236→              {!vtsLoaded ? (
   237→   237→                <p className="text-sm text-[#3A3A3A]/50 mt-1">Cargando VTs...</p>
   238→   238→              ) : vts.length === 0 ? (
   239→   239→                <div className="text-center py-3">
   240→   240→                  <p className="text-sm text-[#912D26]">No hay VTs configurados</p>
   241→   241→                  <p className="text-xs text-[#3A3A3A]/50 mt-1">Ejecuta el seed: POST /api/seed-vts</p>
   242→   242→                </div>
   243→   243→              ) : (
   244→   244→                <div className="grid grid-cols-3 gap-2 mt-2">
   245→   245→                  {vts.map(vt => (
   246→   246→                    <button
   247→   247→                      key={vt.id}
   248→   248→                      type="button"
   249→   249→                      onClick={() => handleVtSelect(vt.id)}
   250→   250→                      className={`py-2.5 px-2 rounded-xl text-sm font-semibold transition-all border ${
   251→   251→                        selectedVtId === vt.id
   252→   252→                          ? 'bg-[#912D26] text-white border-[#912D26] shadow-md'
   253→   253→                          : 'bg-white text-[#3A3A3A] border-[#D6D6D6] hover:border-[#912D26]/40'
   254→   254→                      }`}
   255→   255→                    >
   256→   256→                      {vt.codigo}
   257→   257→                    </button>
   258→   258→                  ))}
   259→   259→                </div>
   260→   260→              )}
   261→   261→              {selectedVtId && (
   262→   262→                <p className="text-xs text-[#912D26]/70 mt-1">
   263→   263→                  {form.trips.length} frecuencias cargadas - puedes editarlas abajo
   264→   264→                </p>
   265→   265→              )}
   266→   266→            </CardContent>
   267→   267→          </Card>
   268→   268→
   269→   269→          {/* 1. Cabecera - Fechas y Km */}
   270→   270→          <Card className="rounded-2xl border border-[#D6D6D6] bg-white">
   271→   271→            <CardContent className="p-4 space-y-3">
   272→   272→              <div className="grid grid-cols-2 gap-3">
   273→   273→                <div className="space-y-1.5">
   274→   274→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Fecha de Registro</Label>
   275→   275→                  <Input
   276→   276→                    type="text"
   277→   277→                    value={new Date().toLocaleDateString('es-EC', { day: '2-digit', month: '2-digit', year: 'numeric' })}
   278→   278→                    readOnly
   279→   279→                    className="rounded-xl h-11 bg-[#F5F5F5] text-[#3A3A3A]/50 cursor-not-allowed border-[#D6D6D6]"
   280→   280→                  />
   281→   281→                </div>
   282→   282→                <div className="space-y-1.5">
   283→   283→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Fecha de Operacion</Label>
   284→   284→                  <Input
   285→   285→                    type="date"
   286→   286→                    value={form.date}
   287→   287→                    onChange={(e) => updateField('date', e.target.value)}
   288→   288→                    className="rounded-xl h-11 border-[#D6D6D6]"
   289→   289→                  />
   290→   290→                </div>
   291→   291→              </div>
   292→   292→              <div className="space-y-1.5">
   293→   293→                <Label className="text-xs font-medium text-[#3A3A3A]/60 flex items-center gap-1">
   294→   294→                  Kilometraje <span className="text-[#912D26]">*</span>
   295→   295→                </Label>
   296→   296→                <Input
   297→   297→                  type="text"
   298→   298→                  inputMode="numeric"
   299→   299→                  placeholder="877604"
   300→   300→                  value={form.km}
   301→   301→                  onChange={(e) => updateField('km', e.target.value)}
   302→   302→                  className={`rounded-xl h-11 border-[#D6D6D6] ${!form.km.trim() && validationErrors.length > 0 ? 'border-[#912D26] bg-[#912D26]/5' : ''}`}
   303→   303→                />
   304→   304→              </div>
   305→   305→              <div className="grid grid-cols-2 gap-3">
   306→   306→                <div className="space-y-1.5">
   307→   307→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Conductor</Label>
   308→   308→                  <Input
   309→   309→                    placeholder="Nombre del conductor"
   310→   310→                    value={form.conductor}
   311→   311→                    onChange={(e) => updateField('conductor', e.target.value)}
   312→   312→                    className="rounded-xl h-11 border-[#D6D6D6]"
   313→   313→                  />
   314→   314→                </div>
   315→   315→                <div className="space-y-1.5">
   316→   316→                  <Label className="text-xs font-medium text-[#3A3A3A]/60">Ayudante</Label>
   317→   317→                  <Input
   318→   318→                    placeholder="Nombre del ayudante"
   319→   319→                    value={form.ayudanteNombre}
   320→   320→                    onChange={(e) => updateField('ayudanteNombre', e.target.value)}
   321→   321→                    className="rounded-xl h-11 border-[#D6D6D6]"
   322→   322→                  />
   323→   323→                </div>
   324→   324→              </div>
   325→   325→
   326→   326→              {/* Foto obligatoria */}
   327→   327→              <div>
   328→   328→                <Label className="text-xs font-medium text-[#3A3A3A]/60 flex items-center gap-1">
   329→   329→                  Foto del Cuaderno <span className="text-[#912D26]">*</span>
   330→   330→                </Label>
   331→   331→                <div className="mt-1.5">
   332→   332→                  {photoPreview ? (
   333→   333→                    <div className="relative w-20 h-20 rounded-xl overflow-hidden border-2 border-[#912D26]">
   334→   334→                      <img src={photoPreview} alt="Foto" className="w-full h-full object-cover" />
   335→   335→                      <button
   336→   336→                        onClick={() => setPhotoPreview(null)}
   337→   337→                        className="absolute top-0 right-0 w-5 h-5 bg-[#912D26] text-white rounded-bl-lg flex items-center justify-center"
   338→   338→                      >
   339→   339→                        <X className="w-3 h-3" />
   340→   340→                      </button>
   341→   341→                    </div>
   342→   342→                  ) : (
   343→   343→                    <Button
   344→   344→                      type="button"
   345→   345→                      variant="outline"
   346→   346→                      className="h-10 px-4 rounded-xl text-xs border-[#D6D6D6] text-[#3A3A3A]"
   347→   347→                      onClick={() => fileInputRef.current?.click()}
   348→   348→                    >
   349→   349→                      <Camera className="w-4 h-4 mr-1" />
   350→   350→                      Tomar/Adjuntar Foto
   351→   351→                    </Button>
   352→   352→                  )}
   353→   353→                </div>
   354→   354→              </div>
   355→   355→            </CardContent>
   356→   356→          </Card>
   357→   357→
   358→   358→          {/* 2. Frecuencias */}
   359→   359→          <Card className="rounded-2xl border border-[#D6D6D6] bg-white">
   360→   360→            <CardHeader className="pb-2 pt-4 px-4">
   361→   361→              <div className="flex items-center justify-between">
   362→   362→                <CardTitle className="text-base font-semibold text-[#3A3A3A]">Frecuencias del Dia</CardTitle>
   363→   363→                <Button onClick={addTrip} size="sm" variant="outline" className="h-8 rounded-lg text-xs border-dashed border-[#D6D6D6] text-[#3A3A3A]">
   364→   364→                  <Plus className="w-3.5 h-3.5 mr-1" /> Agregar
   365→   365→                </Button>
   366→   366→              </div>
   367→   367→            </CardHeader>
   368→   368→            <CardContent className="px-4 pb-4 space-y-3">
   369→   369→              {form.trips.length === 0 && (
   370→   370→                <div className="text-center py-6 text-[#3A3A3A]/40 text-sm">
   371→   371→                  <p>Toca &quot;Agregar&quot; para registrar frecuencias</p>
   372→   372→                </div>
   373→   373→              )}
   374→   374→              {form.trips.map((trip, index) => {
   375→   375→                const tripTimes = getTimesForRoute(trip.routeFrom, trip.routeTo);
   376→   376→                const odd = isOddTrip(index);
   377→   377→                return (
   378→   378→                  <div key={index} className={`rounded-xl p-3 space-y-2 border-l-4 ${odd ? 'border-l-[#912D26] bg-[#912D26]/5' : 'border-l-[#3A3A3A] bg-[#3A3A3A]/5'}`}>
   379→   379→                    <div className="flex items-center justify-between">
   380→   380→                      <span className={`text-xs font-bold ${odd ? 'text-[#912D26]' : 'text-[#3A3A3A]'}`}>
   381→   381→                        Frecuencia #{index + 1} {odd ? '(Loja → Vilca)' : '(Vilca → Loja)'}
   382→   382→                      </span>
   383→   383→                      <Button onClick={() => removeTrip(index)} variant="ghost" size="icon" className="h-7 w-7 text-[#3A3A3A]/40 hover:text-red-500">
   384→   384→                        <Trash2 className="w-3.5 h-3.5" />
   385→   385→                      </Button>
   386→   386→                    </div>
   387→   387→
   388→   388→                    <div className="space-y-1">
   389→   389→                      <Label className="text-[10px] text-[#3A3A3A]/50">Ruta</Label>
   390→   390→                      <select
   391→   391→                        value={getRouteLabel(trip.routeFrom, trip.routeTo)}
   392→   392→                        onChange={(e) => handleRouteSelect(index, e.target.value)}
   393→   393→                        className="w-full h-9 rounded-lg border border-[#D6D6D6] bg-white px-2 text-sm text-[#3A3A3A]"
   394→   394→                      >
   395→   395→                        {allRouteOptions.map((opt) => (
   396→   396→                          <option key={opt} value={opt}>{opt}</option>
   397→   397→                        ))}
   398→   398→                      </select>
   399→   399→                    </div>
