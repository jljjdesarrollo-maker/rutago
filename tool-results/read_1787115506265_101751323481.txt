   592→   592→// ─── Funciones de precio con soporte por dirección ───
   593→   593→//
   594→   594→// Lógica de precios (tipo semántico asignado por el API):
   595→   595→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   596→   596→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   597→   597→//
   598→   598→// El API asigna dirección según el origen:
   599→   599→//   Si origen = Loja →  "ida" (alejándose)
   600→   600→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   601→   601→//
   602→   602→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   603→   603→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   604→   604→
   605→   605→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   606→   606→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   607→   607→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   608→   608→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   609→   609→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   610→   610→  return false;
   611→   611→}

   613→   613→// Obtener precio considerando la dirección del viaje
   614→   614→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   615→   615→  if (viajaHaciaLoja(ruta, tipo)) {
   616→   616→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   617→   617→    if (ruta === 'Loja - El Tambo') {
   618→   618→      const tamboVuelta = preciosVueltaElTambo[parada];
   619→   619→      if (tamboVuelta) return tamboVuelta;
   620→   620→    }
   621→   621→    const vuelta = preciosVuelta[parada];
   622→   622→    if (vuelta) return vuelta;
   623→   623→    // Si la parada no tiene precio de vuelta, buscar en precios base
   624→   624→  }
   625→   625→  // Buscar primero en ida (base), luego en vuelta como fallback
   626→   626→  const ida = preciosIda[parada];
   627→   627→  if (ida) return ida;
   628→   628→  const vuelta = preciosVuelta[parada];
   629→   629→  return vuelta || null;
   630→   630→}

   635→   635→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   636→   636→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   637→   637→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   638→   638→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   639→   639→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   640→   640→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   641→   641→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   642→   642→  let effectiveRuta = ruta;
   643→   643→  let effectiveTipo = tipo;
   644→   644→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   645→   645→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   646→   646→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   647→   647→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   648→   648→    const baseName = ruta.replace(' - Loja', '');
   649→   649→    effectiveRuta = `Loja - ${baseName}`;
   650→   650→    // NO invertir effectiveTipo — ya viene correcto del API
   651→   651→  }
   652→   652→
   653→   653→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   654→   654→  if (!rutaConfig) {
   655→   655→    // Fallback: try original ruta
   656→   656→    const origConfig = RUTA_PARADAS[ruta];
   657→   657→    if (!origConfig) return [];
   658→   658→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   659→   659→    return paradas.map(parada => {
   660→   660→      const precio = getPrecio(parada, ruta, tipo);
   661→   661→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   662→   662→    });
   663→   663→  }
   664→   664→
   665→   665→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   666→   666→  return paradas.map(parada => {
   667→   667→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   668→   668→    return {
   669→   669→      parada,
   670→   670→      normal: precio?.normal ?? 0,
   671→   671→      media: precio?.media ?? 0,
   672→   672→    };
   673→   673→  });
   674→   674→}

   676→   676→// Normalizar ruta y tipo para rutas "X - Loja"
   677→   677→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   678→   678→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   679→   679→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   680→   680→    // El API ya asigna tipo correcto, NO invertir
   681→   681→    const baseName = ruta.replace(' - Loja', '');
   682→   682→    return { ruta: `Loja - ${baseName}`, tipo };
   683→   683→  }
   684→   684→  return { ruta, tipo };
   685→   685→}

   688→   688→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   689→   689→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);

   692→   692→  return pasajeroTipo === 'media' ? precio.media : precio.normal;

   700→   700→// Buscar la ruta más cercana dado un nombre de ruta parcial
   701→   701→export function matchRuta(rutaNombre: string): string {
   702→   702→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   703→   703→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   704→   704→  if (RUTA_PARADAS[normalized]) return normalized;
   705→   705→
   706→   706→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   707→   707→  for (const key of Object.keys(RUTA_PARADAS)) {
   708→   708→    const keyParts = key.split(' - ');
   709→   709→    if (parts.length >= 2 && keyParts.length >= 2) {
   710→   710→      const match = (
   711→   711→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   712→   712→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   713→   713→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   714→   714→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   715→   715→      );
   716→   716→      if (match) return key;
   717→   717→    }
   718→   718→  }
   719→   719→  return rutaNombre;
   720→   720→}
