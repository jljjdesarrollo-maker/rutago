   462→// ─── Funciones de precio con soporte por dirección ───
   463→//
   464→// Lógica de precios (tipo semántico asignado por el API):
   465→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   466→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   467→//
   468→// El API asigna dirección según el origen:
   469→//   Si origen = Loja →  "ida" (alejándose)
   470→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   471→//
   472→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   473→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   474→
   475→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   476→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   477→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   478→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   479→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   480→  return false;
   481→}
   482→
   483→// Obtener precio considerando la dirección del viaje
   484→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   485→  if (viajaHaciaLoja(ruta, tipo)) {
   486→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   487→    if (ruta === 'Loja - El Tambo') {
   488→      const tamboVuelta = preciosVueltaElTambo[parada];
   489→      if (tamboVuelta) return tamboVuelta;
   490→    }
   491→    const vuelta = preciosVuelta[parada];
   492→    if (vuelta) return vuelta;
   493→    // Si la parada no tiene precio de vuelta, buscar en precios base
   494→  }
   495→  // Buscar primero en ida (base), luego en vuelta como fallback
   496→  const ida = preciosIda[parada];
   497→  if (ida) return ida;
   498→  const vuelta = preciosVuelta[parada];
   499→  return vuelta || null;
   500→}

   505→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   506→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   507→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   508→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   509→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   510→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   511→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   512→  let effectiveRuta = ruta;
   513→  let effectiveTipo = tipo;
   514→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   515→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   516→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   517→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   518→    const baseName = ruta.replace(' - Loja', '');
   519→    effectiveRuta = `Loja - ${baseName}`;
   520→    // NO invertir effectiveTipo — ya viene correcto del API
   521→  }
   522→
   523→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   524→  if (!rutaConfig) {
   525→    // Fallback: try original ruta
   526→    const origConfig = RUTA_PARADAS[ruta];
   527→    if (!origConfig) return [];
   528→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   529→    return paradas.map(parada => {
   530→      const precio = getPrecio(parada, ruta, tipo);
   531→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   532→    });
   533→  }
   534→
   535→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   536→  return paradas.map(parada => {
   537→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   538→    return {
   539→      parada,
   540→      normal: precio?.normal ?? 0,
   541→      media: precio?.media ?? 0,
   542→    };
   543→  });
   544→}

   546→// Normalizar ruta y tipo para rutas "X - Loja"
   547→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   548→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   549→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   550→    // El API ya asigna tipo correcto, NO invertir
   551→    const baseName = ruta.replace(' - Loja', '');
   552→    return { ruta: `Loja - ${baseName}`, tipo };
   553→  }
   554→  return { ruta, tipo };
   555→}

   557→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   558→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   559→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   560→  const precio = getPrecio(parada, nr, nt);
   561→  if (!precio) return TARIFA_MINIMA;
   562→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   563→}

   570→// Buscar la ruta más cercana dado un nombre de ruta parcial
   571→export function matchRuta(rutaNombre: string): string {
   572→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   573→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   574→  if (RUTA_PARADAS[normalized]) return normalized;
   575→
   576→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   577→  for (const key of Object.keys(RUTA_PARADAS)) {
   578→    const keyParts = key.split(' - ');
   579→    if (parts.length >= 2 && keyParts.length >= 2) {
   580→      const match = (
   581→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   582→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   583→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   584→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   585→      );
   586→      if (match) return key;
   587→    }
   588→  }
   589→  return rutaNombre;
   590→}
