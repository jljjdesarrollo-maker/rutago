   758→// ─── Funciones de precio con soporte por dirección ───
   759→//
   760→// Lógica de precios (tipo semántico asignado por el API):
   761→// - "ida"   = el bus se ALEJA de Loja → usar preciosIda
   762→// - "vuelta" = el bus VIAJA HACIA Loja → usar preciosVuelta
   763→//
   764→// El API asigna dirección según el origen:
   765→//   Si origen = Loja →  "ida" (alejándose)
   766→//   Si origen ≠ Loja → "vuelta" (hacia Loja)
   767→//
   768→// normalizeRutaTipo solo cambia el nombre de ruta ("Vilcabamba-Loja" → "Loja-Vilcabamba")
   769→// SIN invertir el tipo, porque el API ya lo asigna semánticamente correcto.
   770→
   771→function viajaHaciaLoja(ruta: string, tipo: string): boolean {
   772→  // Ruta "Loja - X": ida se aleja de Loja, vuelta regresa a Loja
   773→  if (ruta.startsWith('Loja')) return tipo === 'vuelta';
   774→  // Ruta "X - Loja": ida va hacia Loja (usar preciosVuelta), vuelta se aleja (usar preciosIda)
   775→  if (ruta.endsWith('Loja')) return tipo === 'ida';
   776→  return false;
   777→}
   778→
   779→// Obtener precio considerando la dirección del viaje
   780→function getPrecio(parada: string, ruta: string, tipo: string): { normal: number; media: number } | null {
   781→  if (viajaHaciaLoja(ruta, tipo)) {
   782→    // Ruta El Tambo tiene precios de vuelta propios (no comparte troncal con Vilcabamba)
   783→    if (ruta === 'Loja - El Tambo') {
   784→      const tamboVuelta = preciosVueltaElTambo[parada];
   785→      if (tamboVuelta) return tamboVuelta;
   786→    }
   787→    // Ruta Zahuayco tiene precios de vuelta propios (directos desde Zahuayco)
   788→    if (ruta === 'Loja - Zahuayco') {
   789→      const zahuaycoVuelta = preciosVueltaZahuayco[parada];
   790→      if (zahuaycoVuelta) return zahuaycoVuelta;
   791→    }
   792→    // Ruta Yangana tiene precios de vuelta propios (directos desde Yangana)
   793→    if (ruta === 'Loja - Yangana') {
   794→      const yangVuelta = preciosVueltaYangana[parada];
   795→      if (yangVuelta) return yangVuelta;
   796→    }
   797→    // Ruta Vilcabamba tiene precios de vuelta propios (directos desde Vilcabamba)
   798→    if (ruta === 'Loja - Vilcabamba') {
   799→      const vilcVuelta = preciosVueltaVilcabamba[parada];
   800→      if (vilcVuelta) return vilcVuelta;
   801→    }
   802→    // Ruta La Elvira tiene precios de vuelta propios (directos desde La Elvira)
   803→    if (ruta === 'Loja - La Elvira') {
   804→      const elviraVuelta = preciosVueltaLaElvira[parada];
   805→      if (elviraVuelta) return elviraVuelta;
   806→    }
   807→    const vuelta = preciosVuelta[parada];
   808→    if (vuelta) return vuelta;
   809→    // Si la parada no tiene precio de vuelta, buscar en precios base
   810→  }
   811→  // Buscar primero en ida (base), luego en vuelta como fallback
   812→  const ida = preciosIda[parada];
   813→  if (ida) return ida;
   814→  const vuelta = preciosVuelta[parada];
   815→  return vuelta || null;
   816→}

   821→// Obtener paradas con tarifas para una ruta y dirección (USA PRECIOS POR DIRECCIÓN)
   822→export function getParadasByRutaAndTipo(ruta: string, tipo: string): { parada: string; normal: number; media: number }[] {
   823→  // Normalize: para rutas "X - Loja", mapear a "Loja - X" pero mantener tipo semántico
   824→  // "Vilcabamba - Loja" tipo="vuelta" (API) → "Loja - Vilcabamba" tipo="vuelta"
   825→  //   → usa paradas vuelta = [Vilcabamba, San Pedro, ...] y preciosVuelta ✓
   826→  // "Vilcabamba - Loja" tipo="ida" (API) → "Loja - Vilcabamba" tipo="ida"
   827→  //   → usa paradas ida = [Dos Puentes, Cajánuma, ...] y preciosIda ✓
   828→  let effectiveRuta = ruta;
   829→  let effectiveTipo = tipo;
   830→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   831→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo el tipo semántico
   832→    // El API ya asigna el tipo correcto: "ida"=alejándose de Loja, "vuelta"=hacia Loja
   833→    // Solo necesitamos el nombre de ruta base, SIN invertir el tipo
   834→    const baseName = ruta.replace(' - Loja', '');
   835→    effectiveRuta = `Loja - ${baseName}`;
   836→    // NO invertir effectiveTipo — ya viene correcto del API
   837→  }
   838→
   839→  const rutaConfig = RUTA_PARADAS[effectiveRuta];
   840→  if (!rutaConfig) {
   841→    // Fallback: try original ruta
   842→    const origConfig = RUTA_PARADAS[ruta];
   843→    if (!origConfig) return [];
   844→    const paradas = tipo === 'ida' ? origConfig.ida : origConfig.vuelta;
   845→    return paradas.map(parada => {
   846→      const precio = getPrecio(parada, ruta, tipo);
   847→      return { parada, normal: precio?.normal ?? 0, media: precio?.media ?? 0 };
   848→    });
   849→  }
   850→
   851→  const paradas = effectiveTipo === 'ida' ? rutaConfig.ida : rutaConfig.vuelta;
   852→  return paradas.map(parada => {
   853→    const precio = getPrecio(parada, effectiveRuta, effectiveTipo);
   854→    return {
   855→      parada,
   856→      normal: precio?.normal ?? 0,
   857→      media: precio?.media ?? 0,
   858→    };
   859→  });
   860→}

   862→// Normalizar ruta y tipo para rutas "X - Loja"
   863→function normalizeRutaTipo(ruta: string, tipo: string): { ruta: string; tipo: string } {
   864→  if (ruta.endsWith('Loja') && !ruta.startsWith('Loja')) {
   865→    // "Vilcabamba - Loja" → mapear a "Loja - Vilcabamba" manteniendo tipo semántico
   866→    // El API ya asigna tipo correcto, NO invertir
   867→    const baseName = ruta.replace(' - Loja', '');
   868→    return { ruta: `Loja - ${baseName}`, tipo };
   869→  }
   870→  return { ruta, tipo };
   871→}

   873→// Obtener tarifa oficial (USA PRECIOS POR DIRECCIÓN)
   874→export function getTarifa(ruta: string, parada: string, tipo: string, pasajeroTipo: TipoPasajero = 'normal'): number {
   875→  const { ruta: nr, tipo: nt } = normalizeRutaTipo(ruta, tipo);
   876→  const precio = getPrecio(parada, nr, nt);
   877→  if (!precio) return TARIFA_MINIMA;
   878→  return pasajeroTipo === 'media' ? precio.media : precio.normal;
   879→}

   886→// Buscar la ruta más cercana dado un nombre de ruta parcial
   887→export function matchRuta(rutaNombre: string): string {
   888→  if (RUTA_PARADAS[rutaNombre]) return rutaNombre;
   889→  const normalized = rutaNombre.replace(/\s*-\s*/g, ' - ').trim();
   890→  if (RUTA_PARADAS[normalized]) return normalized;
   891→
   892→  const parts = rutaNombre.split(/[-–]/).map(s => s.trim());
   893→  for (const key of Object.keys(RUTA_PARADAS)) {
   894→    const keyParts = key.split(' - ');
   895→    if (parts.length >= 2 && keyParts.length >= 2) {
   896→      const match = (
   897→        (keyParts[0].toLowerCase().includes(parts[0].toLowerCase()) ||
   898→         parts[0].toLowerCase().includes(keyParts[0].toLowerCase())) &&
   899→        (keyParts[1].toLowerCase().includes(parts[1].toLowerCase()) ||
   900→         parts[1].toLowerCase().includes(keyParts[1].toLowerCase()))
   901→      );
   902→      if (match) return key;
   903→    }
   904→  }
   905→  return rutaNombre;
   906→}
