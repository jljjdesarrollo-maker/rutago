    66→function getRutaColorKey(rutaNombre: string): RutaColorKey {
    67→  const ruta = matchRuta(rutaNombre).toLowerCase();
    68→  if (ruta.includes('el tambo')) return 'el_tambo';
    69→  if (ruta.includes('la elvira')) return 'la_elvira';
    70→  if (ruta.includes('yangana')) return 'yangana';
    71→  if (ruta.includes('zahuayco')) return 'zahuayco';
    72→  return 'vilcabamba';
    73→}
    74→
    75→// ─── LocalStorage helpers for estados (synchronous, reliable) ───
    76→
    77→function estadosKey(vtCode: string, fecha: string) {
    78→  return `rg_estados_${vtCode}_${fecha}`;
    79→}
    80→
    81→function saveEstadosToLS(vtCode: string, fecha: string, estados: FrecuenciaEstado[]) {
    82→  try {
    83→    localStorage.setItem(estadosKey(vtCode, fecha), JSON.stringify(estados));
    84→  } catch (e) { console.error('Error saving estados to LS:', e); }
    85→}
    86→
    87→function loadEstadosFromLS(vtCode: string, fecha: string): FrecuenciaEstado[] | null {
    88→  try {
    89→    const raw = localStorage.getItem(estadosKey(vtCode, fecha));
    90→    return raw ? JSON.parse(raw) : null;
    91→  } catch { return null; }
    92→}
    93→
    94→function updateEstadoInLS(vtCode: string, fecha: string, estadoId: string, updates: Partial<FrecuenciaEstado>) {
    95→  const all = loadEstadosFromLS(vtCode, fecha);
    96→  if (!all) return;
    97→  const idx = all.findIndex(e => e.estadoId === estadoId);
    98→  if (idx >= 0) {
    99→    all[idx] = { ...all[idx], ...updates };
   100→    saveEstadosToLS(vtCode, fecha, all);
   101→  }
   102→}

   104→export function FrecuenciaSelector({ session, onOpenFrequency, onGoToArqueo, onGoToArqueoGeneral, onBack, onGoToSync }: Props) {

   108→  const [reassigning, setReassigning] = useState<string | null>(null);
   109→  const [pendingCount, setPendingCount] = useState(0);

   116→  const [confirmNoRealizada, setConfirmNoRealizada] = useState(false);
   117→  const [allFrecuencias, setAllFrecuencias] = useState<FrecuenciaData[]>([]);
   118→  const [exitModal, setExitModal] = useState(false);
   119→  const [cajaComunModal, setCajaComunModal] = useState<FrecuenciaEstado | null>(null);
   120→  const [cajaComunCount, setCajaComunCount] = useState(0);
   121→  const [cajaComunMonto, setCajaComunMonto] = useState('');

   125→  useEffect(() => {
   126→    const handle = () => setIsOnline(navigator.onLine);
   127→    window.addEventListener('online', handle);

   130→  }, []);
   131→

   133→    try {
   134→      const res = await fetch(`/api/frecuencias?vtCode=${session.vtCode}`);
   135→      if (res.ok) {
   136→        const data = await res.json();
   137→        setFrecuencias(data);
   138→
   139→        // Try loading saved estados from localStorage first
   140→        const saved = loadEstadosFromLS(session.vtCode, fecha);
   141→        if (saved && saved.length === data.length) {
   142→          setEstados(saved);
   143→        } else {
   144→          // Create fresh pendiente estados
   145→          const newEstados: FrecuenciaEstado[] = data.map((f: FrecuenciaData) => ({
   146→            id: f.id,
   147→            estadoId: `${fecha}_${f.id}`,
   148→            frecuenciaId: f.id,
   149→            nombre: f.nombre,
   150→            ruta: f.ruta,
   151→            hora: f.hora,
   152→            direccion: f.direccion,
   153→            estado: 'pendiente' as const,
   154→            ventasCount: 0,
   155→            totalRecaudado: 0,
   156→          }));
   157→          setEstados(newEstados);
   158→          saveEstadosToLS(session.vtCode, fecha, newEstados);
   159→        }
   160→      }
   161→    } catch (error) {
   162→      console.error('Error cargando frecuencias:', error);
   163→      // Fallback: try to restore from localStorage only
   164→      const saved = loadEstadosFromLS(session.vtCode, fecha);
   165→      if (saved) setEstados(saved);
   166→    } finally {
   167→      setLoading(false);
   168→    }
   169→  }, [session.vtCode, fecha]);
   170→
   171→  const loadPendingCount = useCallback(async () => {
   172→    try {
   173→      const count = await countVentasPendientes();
   174→      setPendingCount(count);
   175→    } catch { /* ignore */ }
   176→  }, []);
   177→
   178→  // Reload ventas counts from IndexedDB periodically
   179→  const refreshVentasCounts = useCallback(async () => {
   180→    for (const estado of estados) {
   181→      try {
   182→        const ventas = await getVentasByFrecuencia(estado.estadoId);
   183→        // Count ALL ventas (including synced) for display in arqueo
   184→        const newCount = ventas.length;
   185→        const newTotal = ventas.reduce((s, v) => s + v.cobrado, 0);
   186→        if (newCount !== estado.ventasCount || Math.abs(newTotal - estado.totalRecaudado) > 0.01) {
   187→          updateEstadoInLS(session.vtCode, fecha, estado.estadoId, { ventasCount: newCount, totalRecaudado: newTotal });
   188→          setEstados(prev => prev.map(e =>
   189→            e.estadoId === estado.estadoId ? { ...e, ventasCount: newCount, totalRecaudado: newTotal } : e
   190→          ));
   191→        }
   192→      } catch { /* ignore */ }
   193→    }
   194→  }, [estados, session.vtCode, fecha]);
   195→
   196→  useEffect(() => {
   197→    loadFrecuencias();
   198→    loadPendingCount();
   199→    const interval = setInterval(() => { loadPendingCount(); }, 5000);
   200→    return () => clearInterval(interval);
   201→  }, [loadFrecuencias, loadPendingCount]);
   202→
   203→  // Refresh counts after initial load and periodically
   204→  useEffect(() => {
   205→    if (!loading && estados.length > 0) {
   206→      refreshVentasCounts();
   207→      const interval = setInterval(refreshVentasCounts, 5000);
   208→      return () => clearInterval(interval);
   209→    }
   210→  }, [loading, estados.length, refreshVentasCounts]);
   211→
   212→  const updateEstado = (estadoId: string, updates: Partial<FrecuenciaEstado>) => {
   213→    // Update in-memory state
   214→    setEstados(prev => prev.map(e => e.estadoId === estadoId ? { ...e, ...updates } : e));
   215→    // Persist to localStorage (synchronous, always reliable)
   216→    updateEstadoInLS(session.vtCode, fecha, estadoId, updates);
   217→  };
   218→
   219→  // Sequential: frequency available if previous are all cerrada/no_realizada
   220→  const isFrecuenciaDisponible = (index: number): boolean => {
   221→    const estado = estados[index];
   222→    if (!estado) return false;
   223→    if (estado.estado === 'cerrada' || estado.estado === 'no_realizada') return false;
   224→    for (let i = 0; i < index; i++) {
   225→      const prev = estados[i];
   226→      if (prev && prev.estado !== 'cerrada' && prev.estado !== 'no_realizada') {
   227→        return false;
   228→      }
   229→    }
   230→    return true;
   231→  };

   234→    // GPS invisible: capture coordinates when frequency opens

   236→    const gps = await getGPSPosition();

   238→    // Generate random winner position for Viaje Gratis
   239→    const promoConfig = loadPromoConfig();

   241→    if (promoConfig.activa) {
   242→      const min = promoConfig.rangoMin || 3;
   243→      const max = promoConfig.rangoMax || 30;
   244→      ganadorPosicion = Math.floor(Math.random() * (max - min + 1)) + min;
   245→    }
   246→
   247→    const updatedEstado = { ...estado, estado: 'abierta' as const, ganadorPosicion };
   248→    updateEstado(estado.estadoId, {
   249→      estado: 'abierta',
   250→      ganadorPosicion,
   251→      ...(gps ? { gpsLatStart: gps.lat, gpsLngStart: gps.lng } : {}),
   252→    });
   253→    onOpenFrequency(updatedEstado);
   254→  };
   255→
   256→  const handleNoRealizada = (estado: FrecuenciaEstado) => {
   257→    setNoRealizadaModal(estado);
   258→    setMotivoSeleccionado('');
   259→    setMotivoPersonalizado('');
   260→    setConfirmNoRealizada(false);
   261→  };

   263→  const confirmNoRealizadaAction = () => {

   266→    // Eliminar ventas de esta frecuencia de IndexedDB (no se realizaron)

   269→      estado: 'no_realizada',
   270→      ventasCount: 0,
   271→      totalRecaudado: 0,
   272→      motivoNoRealizada: motivo || 'Sin especificar',

   274→    setNoRealizadaModal(null);

   277→  // ─── Caja Común (Oficina Loja) ───
   278→  const handleGoToArqueo = (estado: FrecuenciaEstado, esUltima: boolean) => {
   279→    // Open caja común modal first, then proceed to arqueo
   280→    setCajaComunModal(estado);
   281→    setCajaComunEsUltima(esUltima);
   282→    // Restore previous values if reopening
   283→    setCajaComunCount(estado.cajaComunCount || 0);
   284→    setCajaComunMonto(estado.cajaComunMonto ? estado.cajaComunMonto.toString() : '');
   285→  };

   287→  const confirmCajaComun = () => {

   292→      cajaComunCount: cajaComunCount,
   293→      cajaComunMonto: montoNum,

   295→    setCajaComunModal(null);
   296→    // Proceed to arqueo with the updated estado

   299→  };
   300→
   301→  // Filtrar frecuencias por origen para reasignación (usa TODAS las frecuencias del sistema)
   302→  const getFrecuenciasParaReasignar = (estadoId: string): FrecuenciaData[] => {
   303→    const estadoActual = estados.find(e => e.estadoId === estadoId);
   304→    if (!estadoActual) return allFrecuencias;
   305→    // Extraer origen de la ruta: "Vilcabamba - Loja" → "Vilcabamba"
   306→    const partes = estadoActual.ruta.split(' - ');
   307→    const origen = partes[0]?.trim().toLowerCase();
   308→    if (!origen) return allFrecuencias;
   309→    // Filtrar frecuencias que tengan el mismo origen y ordenar por hora
   310→    return allFrecuencias
   311→      .filter(f => {
   312→        const fPartes = f.ruta.split(' - ');
   313→        const fOrigen = fPartes[0]?.trim().toLowerCase();
   314→        return fOrigen === origen;
   315→      })
   316→      .sort((a, b) => a.hora.localeCompare(b.hora));
   317→  };
   318→
   319→  const handleReassign = async (estado: FrecuenciaEstado) => {
   320→    // Cargar todas las frecuencias del sistema si no están cargadas

   322→      try {
   323→        const res = await fetch('/api/frecuencias?all=true');
   324→        if (res.ok) {
   325→          const data = await res.json();
   326→          setAllFrecuencias(data);

   333→  const handleSelectReassignment = (targetFrec: FrecuenciaData) => {

   336→      id: targetFrec.id, frecuenciaId: targetFrec.id,

   338→      hora: targetFrec.hora, direccion: targetFrec.direccion,
   339→      estado: 'pendiente', ventasCount: 0, totalRecaudado: 0,
   340→    });
   341→    setReassigning(null);
   342→  };
   343→
   344→  const getStateIcon = (estado: string) => {
   345→    switch (estado) {
   346→      case 'pendiente': return <Clock className="w-4 h-4 text-gray-400" />;
   347→      case 'abierta': return <Play className="w-4 h-4 text-green-500" />;

   349→      case 'no_realizada': return <XCircle className="w-4 h-4 text-orange-500" />;
   350→      default: return null;

   353→

   357→      case 'abierta': return 'bg-green-50 border-green-200';
   358→      case 'cerrada': return 'bg-blue-50 border-blue-200';
   359→      case 'no_realizada': return 'bg-orange-50 border-orange-200';
   360→      default: return 'bg-gray-100 border-gray-200';
   361→    }
   362→  };
   363→
   364→  const MOTIVOS_NO_REALIZADA = [
   365→    { id: 'daño_unidad', label: 'Daño en la unidad', icon: <Truck className="w-4 h-4" />, color: 'text-red-500' },
   366→    { id: 'mantenimiento', label: 'Mantenimiento', icon: <Wrench className="w-4 h-4" />, color: 'text-blue-500' },
   367→    { id: 'clima', label: 'Clima / Lluvia', icon: <Droplets className="w-4 h-4" />, color: 'text-cyan-500' },
   368→    { id: 'sin_pasajeros', label: 'Sin pasajeros', icon: <UserX className="w-4 h-4" />, color: 'text-purple-500' },
   369→    { id: 'problema_ruta', label: 'Problema en la ruta', icon: <AlertTriangle className="w-4 h-4" />, color: 'text-yellow-500' },
   370→    { id: 'orden_superior', label: 'Orden superior', icon: <Ban className="w-4 h-4" />, color: 'text-gray-600' },

   373→

   375→    switch (estado) {

   377→      case 'abierta': return <span className="px-2 py-0.5 rounded-full text-xs bg-green-200 text-green-700">Abierta</span>;
   378→      case 'cerrada': return <span className="px-2 py-0.5 rounded-full text-xs bg-blue-200 text-blue-700">Cerrada</span>;

   380→      default: return null;
   381→    }
   382→  };
   383→
   384→  if (loading) {
   385→    return (
   386→      <div className="flex flex-col min-h-[100dvh] bg-gray-50 items-center justify-center">

   392→
   393→  // Check if there are unfinished frequencies (for exit guard)
   394→  const hasUnfinishedFrecuencias = estados.some(e => e.estado === 'pendiente' || e.estado === 'abierta');
   395→
   396→  const handleBack = () => {
   397→    if (hasUnfinishedFrecuencias) {
   398→      setExitModal(true);
   399→    } else {

   401→    }
   402→  };
   403→
   404→  // Check if ALL frequencies are done (cerrada or no_realizada)
   405→  const allFrecuenciasDone = estados.length > 0 && estados.every(e => e.estado === 'cerrada' || e.estado === 'no_realizada');
   406→
   407→  // Totals across all frequencies

   412→  // SYNC control: ALWAYS block arqueo general if there are pending ventas
   413→  const mustSyncBeforeArqueo = pendingCount > 0;
   414→

   419→      const result = await syncVentasSilencioso();
   420→      const remaining = await countVentasPendientes();
   421→      setPendingCount(remaining);
   422→      if (remaining === 0) {
   423→        setSyncDone(true);
   424→        setTimeout(() => setSyncDone(false), 2000);
   425→      }
   426→    } catch { /* ignore */ }
   427→    setSyncingForArqueo(false);
   428→  };
   429→
   430→  return (
   431→    <div className="flex flex-col min-h-[100dvh] bg-gray-50">
   432→      {/* Header */}
   433→      <div className="bg-[#912D26] text-white px-4 py-3">
   434→        <div className="flex items-center justify-between mb-2">
   435→          <div className="flex items-center gap-2">
   436→            <button onClick={handleBack} className="text-red-100"><ArrowLeft className="w-5 h-5" /></button>
   437→            <div>
   438→              <h1 className="text-lg font-bold">Frecuencias</h1>
   439→              <p className="text-red-100 text-xs">{session.nombre} - {session.ayudanteNombre}</p>
   440→            </div>
   441→          </div>
   442→          <div className="flex items-center gap-2">
   443→            {isOnline ? <Wifi className="w-4 h-4 text-green-300" /> : <WifiOff className="w-4 h-4 text-red-300" />}
   444→          </div>
   445→        </div>
   446→        <div className="flex items-center gap-2">
   447→          <div className="flex-1 bg-white/15 rounded-xl px-3 py-1.5 text-sm flex items-center gap-2">
